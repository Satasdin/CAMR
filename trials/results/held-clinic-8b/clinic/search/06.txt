20:     sign = "-" if amount_cents < 0 else ""

22:     return f"{sign}{currency} {whole:,}.{part:02d}"

25: def parse_amount(text: str) -> int:
26:     """Reads an amount typed by staff, such as '1,250.50', into cents."""
27:     cleaned = text.replace(",", "").strip()
28:     if not cleaned:
29:         raise ValueError("empty amount")
30:     negative = cleaned.startswith("-")
31:     if negative:
32:         cleaned = cleaned[1:]
33:     whole, _, part = cleaned.partition(".")
34:     if not whole.isdigit() and whole != "":
35:         raise ValueError(f"not an amount: {text}")
36:     if part and (not part.isdigit() or len(part) > 2):
37:         raise ValueError(f"not an amount: {text}")
38:     cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
39:     return -cents if negative else cents
40: 
41: 
42: def split_evenly(amount_cents: int, parts: int) -> list:
43:     """Splits an amount so the parts always add back to the amount."""
44:     if parts <= 0:
45:         raise ValueError("parts must be positive")
46:     share, remainder = divmod(amount_cents, parts)
47:     return [share + 1 if index < remainder else share for index in range(parts)]

from ..config import INVOICE_DUE_DAYS, SECONDS_PER_DAY
from ..errors import ValidationFailed
from ..models import INVOICE_OPEN, INVOICE_VOID, Invoice, InvoiceLine
from ..money import format_cents
from ..store.repositories import get_appointment, get_doctor, get_invoice, save_invoice
from .tariffs import lines_for


def create_invoice(store, appointment_id: str, clock) -> Invoice:
    appointment = get_appointment(store, appointment_id)
    doctor = get_doctor(store, appointment.doctor_id)
    lines = [InvoiceLine(description, cents) for description, cents in lines_for(appointment, doctor)]
    if not lines:
        raise ValidationFailed(["there is nothing to bill for this appointment"])
    now = clock.now()
    invoice = Invoice(id=store.next_id("INV"), patient_id=appointment.patient_id, appointment_id=appointment_id,
                      lines=lines, total_cents=sum(line.amount_cents for line in lines),
                      created_at=now, due_at=now + INVOICE_DUE_DAYS * SECONDS_PER_DAY)
    return save_invoice(store, invoice)


def outstanding_cents(invoice: Invoice) -> int:
    return invoice.total_cents - invoice.paid_cents


def void_invoice(store, invoice_id: str) -> Invoice:
    invoice = get_invoice(store, invoice_id)
    if invoice.status != INVOICE_OPEN or invoice.paid_cents:
        raise ValidationFailed([f"invoice {invoice_id} cannot be voided"])
    invoice.status = INVOICE_VOID
    return save_invoice(store, invoice)


def is_overdue(invoice: Invoice, now: int) -> bool:
    return invoice.status == INVOICE_OPEN and now > invoice.due_at


def render_invoice(invoice: Invoice) -> str:
    rows = [f"Invoice {invoice.id}"]
    rows += [f"  {line.description}: {format_cents(line.amount_cents)}" for line in invoice.lines]
    rows.append(f"Total: {format_cents(invoice.total_cents)}")
    rows.append(f"Outstanding: {format_cents(outstanding_cents(invoice))}")
    return "\n".join(rows)

from ..errors import InvoiceNotOpen, PaymentTooLarge, ValidationFailed
from ..models import INVOICE_OPEN, INVOICE_PAID, Invoice, Payment

from .invoices import outstanding_cents

METHODS = ("cash", "card", "mobile_money")


def record_payment(store, invoice_id: str, amount_cents: int, method: str, clock) -> Invoice:
    invoice = get_invoice(store, invoice_id)
    if invoice.status != INVOICE_OPEN:
        raise InvoiceNotOpen(invoice_id, invoice.status)
    if method not in METHODS:
        raise ValidationFailed([f"unknown payment method: {method}"])
    if amount_cents <= 0:
        raise ValidationFailed(["a payment must be more than zero"])
    owed = outstanding_cents(invoice)
    if amount_cents > owed:
        raise PaymentTooLarge(amount_cents, owed)
    save_payment(store, Payment(id=store.next_id("PAY"), invoice_id=invoice_id, amount_cents=amount_cents,
                                method=method, at=clock.now()))
    invoice.paid_cents += amount_cents
    if outstanding_cents(invoice) == 0:
        invoice.status = INVOICE_PAID
    return save_invoice(store, invoice)

"""Deciding who is reminded about an appointment, and how."""

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject, sms_body


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    if patient.wants_sms and patient.phone:
        out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, doctor, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count

def create_invoice(store, appointment_id: str, clock) -> Invoice:
    appointment = get_appointment(store, appointment_id)
    doctor = get_doctor(store, appointment.doctor_id)
    lines = [InvoiceLine(description, cents) for description, cents in lines_for(appointment, doctor)]
    if not lines:
        raise ValidationFailed(["there is nothing to bill for this appointment"])
    now = clock.now()
    invoice = Invoice(id=store.next_id("INV"), patient_id=appointment.patient_id, appointment_id=appointment_id,
                      lines=lines, total_cents=sum(line.amount_cents for line in lines),
                      created_at=now, due_at=now + INVOICE_DUE_DAYS * SECONDS_PER_DAY)
    return save_invoice(store, invoice)


def outstanding_cents(invoice: Invoice) -> int:
    return invoice.total_cents - invoice.paid_cents


def void_invoice(store, invoice_id: str) -> Invoice:
    invoice = get_invoice(store, invoice_id)
    if invoice.status != INVOICE_OPEN or invoice.paid_cents:
        raise ValidationFailed([f"invoice {invoice_id} cannot be voided"])
    invoice.status = INVOICE_VOID
    return save_invoice(store, invoice)


def is_overdue(invoice: Invoice, now: int) -> bool:
    return invoice.status == INVOICE_OPEN and now > invoice.due_at


def render_invoice(invoice: Invoice) -> str:
    rows = [f"Invoice {invoice.id}"]
    rows += [f"  {line.description}: {format_cents(line.amount_cents)}" for line in invoice.lines]
    rows.append(f"Total: {format_cents(invoice.total_cents)}")
    rows.append(f"Outstanding: {format_cents(outstanding_cents(invoice))}")
    return "\n".join(rows)

    invoice = get_invoice(store, invoice_id)
    if invoice.status != INVOICE_OPEN:

    if method not in METHODS:
        raise ValidationFailed([f"unknown payment method: {method}"])
    if amount_cents <= 0:
        raise ValidationFailed(["a payment must be more than zero"])
    owed = outstanding_cents(invoice)
    if amount_cents > owed:
        raise PaymentTooLarge(amount_cents, owed)

                                method=method, at=clock.now()))

    if outstanding_cents(invoice) == 0:
        invoice.status = INVOICE_PAID
    return save_invoice(store, invoice)

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject, sms_body


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    if patient.wants_sms and patient.phone:
        out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, doctor, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count
