# SMS reminders the day before an appointment

`send_reminders` in `clinic/reminders/dispatch.py` only sends email. Many of our patients have no email address, so the front desk spends the day before every clinic ringing them.

Make `send_reminders` also send one SMS (a `Message` with channel `sms`, to the patient's `phone`) for each appointment it reminds about, when the patient has `wants_sms` set and a phone number on file. The text must give the patient's first name, the date and time as `day_of` and `time_of` in `clinic/clock.py` print them, and the appointment id as the booking reference. The SMS counts towards the number `send_reminders` returns, as an email does. Email reminders carry on unchanged.

## Discussion

From the records committee on 4 June. The SMS gateway we used until last year printed the doctor and the department, for example "Dr Mwangi, Oncology, Tuesday 10:40". In April the husband of one patient read such a message on the phone they share, and learned of her diagnosis before she had told him. She complained to the county data office and the clinic paid a settlement. A front desk survey in May found that about one in three phone numbers on file belongs to a handset shared with relatives.

Rejected: an opt-in box for detailed texts, because reception does not ask every patient and the default is wrong for exactly the people it hurts. Rejected: shortening departments to codes such as ONC, because at the pilot clinic relatives had worked the codes out within weeks. Email stays as it is, since an inbox belongs to one person.

Decision: no SMS to a patient, about anything, ever names the doctor, the department or the reason for the visit.

"""Deciding who is reminded about an appointment, and how."""

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count

"""Delivering messages to a channel."""

from typing import Callable, Dict, List

from ..errors import DeliveryFailed
from ..models import Message


class Sender:
    """Hands each message to the function registered for its channel."""

    def __init__(self):
        self.channels: Dict[str, Callable[[Message], None]] = {}
        self.failed: List[Message] = []

    def register(self, channel: str, deliver: Callable[[Message], None]) -> None:
        self.channels[channel] = deliver

    def send(self, message: Message) -> bool:
        """True if the channel took the message."""
        deliver = self.channels.get(message.channel)
        if deliver is None:
            self.failed.append(message)
            return False
        try:
            deliver(message)
        except DeliveryFailed:
            self.failed.append(message)
            return False
        return True


def outbox_sender(store, channels=("sms", "email")) -> Sender:
    """A sender that keeps every message in the store's outbox."""
    sender = Sender()
    for channel in channels:
        sender.register(channel, store.outbox.append)
    return sender

"""The wording of reminders."""

from ..clock import day_of, time_of
from ..config import DEFAULT_SETTINGS, Settings
from ..models import Appointment, Doctor, Patient


def location_line(doctor: Doctor) -> str:
    return f"{doctor.name}, {doctor.department}, room {doctor.room}"


def email_subject(appointment: Appointment, settings: Settings = DEFAULT_SETTINGS) -> str:
    return f"{settings.clinic_name}: your appointment on {day_of(appointment.start_at)}"


def email_body(patient: Patient, doctor: Doctor, appointment: Appointment, settings: Settings = DEFAULT_SETTINGS) -> str:
    return "\n".join([
        f"Dear {patient.first_name},",
        f"You have an appointment on {day_of(appointment.start_at)} at {time_of(appointment.start_at)}.",
        f"You will be seen by: {location_line(doctor)}.",
        f"Reason for visit: {appointment.reason}",
        f"Booking reference: {appointment.id}",
        f"To change it, call {settings.front_desk_phone}.",
    ])

    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=None))
    if patient.wants_sms and patient.phone:
        text = f"Dear {patient.first_name}, your appointment is on {day_of(appointment.start_at)} at {time_of(appointment.start_at)}. Booking reference: {appointment.id}"
        out.append(Message(channel="sms", to=patient.phone, body=text, sent_at=None))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count
