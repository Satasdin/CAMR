1: """Helpers for working with whole cents."""
2: 
3: from typing import Iterable
4: 
5: from .config import CURRENCY
6: 
7: 
8: def percent_of(amount_cents: int, percent: int) -> int:
9:     """Rounds half up, so a share of an amount never comes out a cent short."""
10:     if amount_cents < 0:
11:         raise ValueError("amount must not be negative")
12:     return (amount_cents * percent + 50) // 100
13: 
14: 
15: def add(amounts: Iterable[int]) -> int:
16:     return sum(amounts)
17: 
18: 
19: def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
20:     sign = "-" if amount_cents < 0 else ""
21:     whole, part = divmod(abs(amount_cents), 100)
22:     return f"{sign}{currency} {whole:,}.{part:02d}"
23: 
24: 
25: def parse_amount(text: str) -> int:
26:     """Reads an amount typed by staff, such as '1,250.50', into cents."""
27:     cleaned = text.replace(",", "").strip()
28:     if not cleaned:
29:         raise ValueError("empty amount")
30:     negative = cleaned.startswith("-")
31:     if negative:
32:         cleaned = cleaned[1:]
33:     whole, _, part = cleaned.partition(".")
34:     if not whole.isdigit() and whole != "":
35:         raise ValueError(f"not an amount: {text}")
36:     if part and (not part.isdigit() or len(part) > 2):
37:         raise ValueError(f"not an amount: {text}")
38:     cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
39:     return -cents if negative else cents
40: 
41: 
42: def split_evenly(amount_cents: int, parts: int) -> list:
43:     """Splits an amount so the parts always add back to the amount."""
44:     if parts <= 0:
45:         raise ValueError("parts must be positive")
46:     share, remainder = divmod(amount_cents, parts)
47:     return [share + 1 if index < remainder else share for index in range(parts)]

13:     for threshold, percent in LOYALTY_TIERS:
14:         if points >= threshold:
15:             return percent
16:     return 0
17:

20:     return PROMO_CODES.get(code.strip().upper(), 0)
21: 
22: 
23: def discounts_for(customer: Customer, subtotal_cents: int, promo: str = "") -> List[Tuple[str, int]]:
24:     """Only the larger of a loyalty and a promotional discount is applied."""
25:     loyalty = loyalty_percent(customer.loyalty_points)
26:     code = promo_percent(promo) if promo else 0
27:     if not loyalty and not code:
28:         return []
29:     if code > loyalty:
30:         return [(f"promo {promo.strip().upper()}", -percent_of(subtotal_cents, code))]
31:     return [("loyalty", -percent_of(subtotal_cents, loyalty))]

"""Extra charges added on top of the band and weight prices."""

from typing import List, Tuple

from ..config import DEFAULT_SETTINGS, Settings
from ..models import Parcel
from ..money import percent_of
from .zones import is_remote


def fuel_surcharge(subtotal_cents: int, settings: Settings) -> int:
    return percent_of(subtotal_cents, settings.fuel_percent)


def remote_area_surcharge(parcel: Parcel, settings: Settings) -> int:
    if is_remote(parcel.origin_zone) or is_remote(parcel.destination_zone):
        return settings.remote_area_cents
    return 0


def insurance_surcharge(parcel: Parcel, settings: Settings) -> int:
    return percent_of(parcel.insured_value_cents, settings.insurance_percent)


def surcharges_for(parcel: Parcel, subtotal_cents: int, settings: Settings = DEFAULT_SETTINGS) -> List[Tuple[str, int]]:
    """Every surcharge that applies, as (name, cents) pairs."""
    found = [("fuel", fuel_surcharge(subtotal_cents, settings))]
    remote = remote_area_surcharge(parcel, settings)
    if remote:
        found.append(("remote_area", remote))
    insurance = insurance_surcharge(parcel, settings)
    if insurance:
        found.append(("insurance", insurance))
    return found

"""Combines every part of a price into one quote."""

from dataclasses import dataclass
from typing import List, Optional, Tuple

from ..config import DEFAULT_SETTINGS, Settings, service_level
from ..models import Customer, Parcel
from ..money import percent_of
from .discounts import discounts_for
from .rates import check_weight, weight_charge
from .surcharges import surcharges_for
from .zones import band, band_price


@dataclass
class Quote:
    lines: List[Tuple[str, int]]
    total_cents: int
    band: str

    def amount(self, name: str) -> int:
        return sum(cents for line, cents in self.lines if line == name)


def quote_parcel(parcel: Parcel, customer: Optional[Customer] = None, promo: str = "", settings: Settings = DEFAULT_SETTINGS) -> Quote:
    check_weight(parcel.weight_kg)
    which = band(parcel.origin_zone, parcel.destination_zone)
    level = service_level(parcel.service)

    lines: List[Tuple[str, int]] = [("base", band_price(parcel.origin_zone, parcel.destination_zone))]
    weight = weight_charge(parcel, which)
    if weight:
        lines.append(("weight", weight))

    subtotal = sum(cents for _, cents in lines)
    if level.multiplier_percent != 100:
        change = percent_of(subtotal, level.multiplier_percent) - subtotal
        lines.append((f"service {level.code}", change))
        subtotal += change

    lines.extend(surcharges_for(parcel, subtotal, settings))
    if customer is not None:
        lines.extend(discounts_for(customer, subtotal, promo))
    return Quote(lines=lines, total_cents=sum(cents for _, cents in lines), band=which)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up, so a share of an amount never comes out a cent short."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Discounts that reduce a quote."""

from typing import List, Tuple

from ..models import Customer
from ..money import percent_of

LOYALTY_TIERS = ((5_000, 10), (2_000, 5), (500, 2))
PROMO_CODES = {"WELCOME10": 10, "RAINY5": 5}


def loyalty_percent(points: int) -> int:
    for threshold, percent in LOYALTY_TIERS:
        if points >= threshold:
            return percent
    return 0


def promo_percent(code: str) -> int:
    return PROMO_CODES.get(code.strip().upper(), 0)


def discounts_for(customer: Customer, subtotal_cents: int, promo: str = "") -> List[Tuple[str, int]]:
    """Only the larger of a loyalty and a promotional discount is applied."""
    loyalty = loyalty_percent(customer.loyalty_points)
    code = promo_percent(promo) if promo else 0
    if not loyalty and not code:
        return []
    if code > loyalty:
        return [(f"promo {promo.strip().upper()}", -percent_of(subtotal_cents, code))]
    return [("loyalty", -percent_of(subtotal_cents, loyalty))]

"""Delivery zones and the bands between them."""

from typing import Dict, Tuple

ZONE_NAMES: Dict[str, str] = {
    "NBO": "Nairobi",
    "MBA": "Mombasa",
    "KSM": "Kisumu",
    "NKR": "Nakuru",
    "ELD": "Eldoret",
    "NYR": "Nyeri",
    "LAM": "Lamu",
    "GAR": "Garissa",
    "LOD": "Lodwar",
}

REMOTE_ZONES = frozenset({"LAM", "GAR", "LOD"})

BAND_PRICES_CENTS: Dict[str, int] = {"local": 25_000, "near": 42_000, "far": 68_000, "remote": 110_000}

_NEAR: Tuple[frozenset, ...] = (
    frozenset({"NBO", "NKR"}),
    frozenset({"NBO", "NYR"}),
    frozenset({"NKR", "ELD"}),
    frozenset({"NKR", "KSM"}),
    frozenset({"ELD", "KSM"}),
)


def is_known(zone: str) -> bool:
    return zone in ZONE_NAMES


def is_remote(zone: str) -> bool:
    return zone in REMOTE_ZONES


def band(origin: str, destination: str) -> str:
    """The price band for a journey. The order of the two zones does not matter."""
    for zone in (origin, destination):
        if not is_known(zone):
            raise ValueError(f"unknown zone: {zone}")
    if origin == destination:
        return "local"
    if is_remote(origin) or is_remote(destination):
        return "remote"
    if frozenset({origin, destination}) in _NEAR:
        return "near"
    return "far"


def band_price(origin: str, destination: str) -> int:
    return BAND_PRICES_CENTS[band(origin, destination)]
