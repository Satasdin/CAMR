1: """Helpers for working with whole shillings."""
2: 
3: from typing import Iterable
4: 
5: from .config import CURRENCY
6: 
7: 
8: def percent_of(amount_kes: int, percent: int) -> int:
9:     """Rounds half up."""
10:     if amount_kes < 0:
11:         raise ValueError("amount must not be negative")
12:     return (amount_kes * percent + 50) // 100
13: 
14: 
15: def add(amounts: Iterable[int]) -> int:
16:     return sum(amounts)
17: 
18: 
19: def round_up_to(amount_kes: int, step: int) -> int:
20:     if step <= 0:
21:         raise ValueError("step must be positive")
22:     return -(-amount_kes // step) * step
23: 
24: 
25: def format_kes(amount_kes: int, currency: str = CURRENCY) -> str:
26:     sign = "-" if amount_kes < 0 else ""
27:     return f"{sign}{currency} {abs(amount_kes):,}"
28: 
29: 
30: def parse_amount(text: str) -> int:
31:     """Reads an amount from a statement or typed by staff, such as '1,250' or '1,250.00'."""
32:     cleaned = text.replace(",", "").strip()
33:     if not cleaned:
34:         raise ValueError("empty amount")
35:     whole, _, part = cleaned.partition(".")
36:     if not whole.isdigit():
37:         raise ValueError(f"not an amount: {text}")
38:     if part.strip("0"):
39:         raise ValueError(f"not a whole number of shillings: {text}")
40:     return int(whole)
41: 
42: 
43: def split_evenly(amount_kes: int, parts: int) -> list:
44:     """Splits an amount so the parts always add back to the amount."""
45:     if parts <= 0:
46:         raise ValueError("parts must be positive")
47:     share, remainder = divmod(amount_kes, parts)
48:     return [share + 1 if index < remainder else share for index in range(parts)]

1: # Matatu

3: The back office of a Nairobi minibus operator. It keeps the routes and stages, works

5: to tickets and prepares the daily figures.
6: 
7: ## Layout
8: 
9: - `matatu/routes`: stages and the order they come in along a route.

13: - `matatu/crews`: drivers and conductors, shifts and commission.
14: - `matatu/payments`: the mobile-money statement, matching lines to tickets, suspense and the provider.
15: - `matatu/reports`: figures for finance and the routes table.
16: - `matatu/notify`: message templates and delivery channels.
17: - `matatu/api`: request validation and the handlers the web layer calls.
18: - `matatu/store`: an in-memory store with small repository helpers.
19: 
20: Money is held as whole Kenya shillings. Times are whole seconds since the epoch, taken from
21: `matatu.clock` so tests can control them.

"""The people who own the vehicles."""

from typing import List

from ..models import Vehicle
from ..money import percent_of
from ..errors import NotFound


def vehicles_of(store, owner_id: str) -> List[Vehicle]:
    return [vehicle for vehicle in store.vehicles.values() if vehicle.owner_id == owner_id]


def fleet_size(store, owner_id: str) -> int:
    return sum(1 for vehicle in vehicles_of(store, owner_id) if vehicle.active)


def owner_payout(store, owner_id: str, date: str) -> dict:
    try:
        vehicles = vehicles_of(store, owner_id)
    except NotFound:
        raise NotFound("owner", owner_id) from None

    gross_kes = 0
    for vehicle in vehicles:
        for ticket in store.tickets.values():
            if ticket.status == TICKET_PAID and day_of(ticket.issued_at) == date and ticket.vehicle_id == vehicle.id:
                gross_kes += ticket.fare_kes

    levy_kes = percent_of(gross_kes, SACCO_LEVY_PERCENT)
    net_kes = gross_kes - levy_kes

    return {
        "owner_id": owner_id,
        "date": date,
        "gross_kes": gross_kes,
        "levy_kes": levy_kes,
        "net_kes": net_kes
    }
