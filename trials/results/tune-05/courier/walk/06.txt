21:     frozenset({"NBO", "MBA"}): 9,
22:     frozenset({"NBO", "GAR"}): 7,
23:     frozenset({"NKR", "ELD"}): 3,

25:     frozenset({"ELD", "KSM"}): 3,
26:     frozenset({"ELD", "LOD"}): 9,
27:     frozenset({"MBA", "LAM"}): 6,
28: }
29: 
30: 
31: def route_between(origin: str, destination: str) -> List[str]:
32:     """The route with the fewest stops, found breadth first. Raises ValueError if none exists."""
33:     if origin not in LINKS or destination not in LINKS:
34:         raise ValueError("unknown depot")
35:     queue = deque([[origin]])
36:     visited = {origin}
37:     while queue:
38:         path = queue.popleft()
39:         if path[-1] == destination:
40:             return path
41:         for neighbour in sorted(LINKS[path[-1]]):
42:             if neighbour not in visited:
43:                 visited.add(neighbour)
44:                 queue.append(path + [neighbour])
45:     raise ValueError(f"no route from {origin} to {destination}")

31: def route_between(origin: str, destination: str) -> List[str]:
32:     """The route with the fewest stops, found breadth first. Raises ValueError if none exists."""
33:     if origin not in LINKS or destination not in LINKS:
34:         raise ValueError("unknown depot")
35:     queue = deque([[origin]])
36:     visited = {origin}
37:     while queue:
38:         path = queue.popleft()
39:         if path[-1] == destination:
40:             return path
41:         for neighbour in sorted(LINKS[path[-1]]):
42:             if neighbour not in visited:
43:                 visited.add(neighbour)
44:                 queue.append(path + [neighbour])
