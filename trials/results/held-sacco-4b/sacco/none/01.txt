# Undo a mistaken posting

Tellers have no way to undo a deposit or withdrawal that was posted by mistake.

Add `sacco/savings/reversals.py` with `reverse_entry(store, entry_id, reason, clock)`. It undoes the ledger entry so that the account's balance is back to what it was without that entry, and returns the account's new balance in cents. An unknown entry raises `NotFound`, as `find_entry` in `sacco/savings/ledger.py` does. Reversing the same entry a second time raises `AlreadyReversed`.

Add a handler `reverse_posting(request, context)` to `sacco/api/handlers.py`. It takes `{"entry_id": ..., "reason": ...}` and returns `{"entry_id": ..., "balance_cents": ...}`, using the `handled` decorator like the other handlers.

## Discussion

From the audit committee on 18 June. For years tellers fixed a mistyped or wrong posting by deleting the ledger line and entering it again. At the 2023 SASRA inspection the inspector found 214 deposits with no ledger line for money we had clearly received, and a member walked in with a printed statement showing a deposit the ledger no longer held. Members keep their printed statements, and the auditors tick statements against the ledger line by line.

The treasurer wanted to keep the line but set a `reversed` flag on it and leave it out of balances. Rejected: a flag changes what an old line means after the fact, so an old printed statement and the ledger disagree with no line to explain why. Deleting the line is rejected for the reasons above.

Decision: a reversal posts a new ledger line for the opposite amount, and the original line is never changed or removed. Both lines stay on the statement.

"""Every movement of money on a savings account."""

from typing import List

from ..errors import NotFound
from ..models import LedgerEntry


def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)

"""Handlers take a request dictionary and a context, and return a response dictionary."""

from dataclasses import dataclass
from functools import wraps

from ..errors import SaccoError
from ..loans.applications import apply_for_loan, approve_loan
from ..loans.disbursement import disburse_loan
from ..members.registry import register_member
from ..notify.dispatch import notify_deposit, notify_loan_sent, notify_repayment
from ..repayments.payments import apply_repayment
from ..savings.deposits import deposit
from ..savings.withdrawals import withdraw
from ..statements.loan import loan_statement
from ..statements.member import member_statement
from ..store.repositories import get_loan
from .validators import require, require_amount


@dataclass
class Context:
    store: object
    gateway: object
    clock: object
    sender: object


def handled(function):
    """Turns expected errors into a response the web layer can send."""

    @wraps(function)
    def wrapper(request, context):
        try:
            return function(request, context)
        except SaccoError as problem:
            return {"error": problem.code, "message": str(problem)}

    return wrapper


@handled
def join(request, context):
    require(request, "name", "phone", "national_id")
    member = register_member(context.store, request["name"], request["phone"], request.get("email", ""), request["national_id"], context.clock)
    return {"member_id": member.id}


@handled
def deposit_cash(request, context):
    require(request, "member_id")
    amount = require_amount(request)
    entry = deposit(context.store, request["member_id"], amount, context.clock, reference=request.get("reference", ""))
    notify_deposit(context.store, request["member_id"], amount, entry.id, context.sender, context.clock)
    return {"entry_id": entry.id}


@handled
def withdraw_cash(request, context):
    require(request, "member_id")
    amount = require_amount(request)
    entry = withdraw(context.store, request["member_id"], amount, context.clock)
    return {"entry_id": entry.id}


@handled
def request_loan(request, context):
    require(request, "member_id", "product", "term_months")
    amount = require_amount(request)
    loan = apply_for_loan(context.store, request["member_id"], request["product"], amount, request["term_months"], context.clock)
    return {"loan_id": loan.id, "status": loan.status}


@handled
def approve(request, context):
    require(request, "loan_id")
    loan = approve_loan(context.store, request["loan_id"], context.clock)
    return {"loan_id": loan.id, "status": loan.status}


@handled
def send_out(request, context):
    require(request, "loan_id")
    done = disburse_loan(context.store, request["loan_id"], context.gateway, context.clock)
    loan = get_loan(context.store, request["loan_id"])
    notify_loan_sent(context.store, loan.member_id, done.sent_cents, done.receipt_id, context.sender, context.clock)
    return {"loan_id": done.loan_id, "sent_cents": done.sent_cents, "fee_cents": done.fee_cents, "receipt_id": done.receipt_id}


@handled
def repay(request, context):
    require(request, "loan_id")
    amount = require_amount(request)
    repayment = apply_repayment(context.store, request["loan_id"], amount, context.clock, reference=request.get("reference", ""))
    loan = get_loan(context.store, request["loan_id"])
    notify_repayment(context.store, loan.member_id, amount, repayment.id, context.sender, context.clock)
    return {"repayment_id": repayment.id, "loan_status": loan.status}


@handled
def savings_statement(request, context):
    require(request, "member_id", "start", "end")
    return member_statement(context.store, request["member_id"], request["start"], request["end"])


@handled
def loan_position(request, context):
    require(request, "loan_id")
    return loan_statement(context.store, request["loan_id"])

"""Every movement of money on a savings account."""

from typing import List

from ..errors import NotFound
from ..models import LedgerEntry


def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)

"""Reverse a ledger entry by posting a new entry with the opposite amount."""

from typing import int
from ..errors import NotFound, AlreadyReversed
from ..models import LedgerEntry
from .ledger import find_entry, balance_of


def reverse_entry(store, entry_id: str, reason: str, clock):
    """
    Undoes a ledger entry by posting a new entry with the opposite amount.
    The original entry remains in the ledger, and the account's balance is adjusted.
    
    Args:
        store: The store object containing the ledger.
        entry_id: The ID of the entry to reverse.
        reason: A string explaining why the reversal is being made.
        clock: The clock object for timestamping the new entry.
    
    Returns:
        The new balance in cents after the reversal.
    
    Raises:
        NotFound: If the entry is not found in the ledger.
        AlreadyReversed: If the entry has already been reversed.
    """
    # Find the original entry
    entry = find_entry(store, entry_id)
    if not entry:
        raise NotFound("entry", entry_id)
    
    # Check if the entry has already been reversed
    if entry.reversed:
        raise AlreadyReversed(entry_id)
    
    # Create a new entry with the opposite amount
    reversed_entry = LedgerEntry(
        id=store.next_id("ENT"),
        account_id=entry.account_id,
        kind=entry.kind,
        amount_cents=-entry.amount_cents,
        at=clock.now(),
        reference=entry.reference,
        note=f"Reversed: {reason}",
        reversed=True
    )
    
    # Append the reversed entry to the ledger
    store.ledger.append(reversed_entry)
    
    # Return the new balance
    return balance_of(store, entry.account_id)
