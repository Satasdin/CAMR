"""Working out which appointment times can be offered."""

from typing import Dict, List, Set, Tuple

from ..clock import hour_of, moment_at, weekday_of
from ..config import CLINIC_CLOSE_HOUR, CLINIC_OPEN_HOUR, SECONDS_PER_HOUR, SECONDS_PER_MINUTE, SLOT_MINUTES
from ..models import APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN
from ..store.repositories import appointments_of_doctor
from .rosters import is_on_shift

WEEKLY_BLOCKS: Dict[int, Tuple[int, int]] = {2: (13, 15)}


def is_blocked(moment: int) -> bool:
    block = WEEKLY_BLOCKS.get(weekday_of(moment))
    return block is not None and block[0] <= hour_of(moment) < block[1]


def slots_on(day: str) -> List[int]:
    first = moment_at(day, CLINIC_OPEN_HOUR)
    last = moment_at(day, CLINIC_CLOSE_HOUR)
    return list(range(first, last, SLOT_MINUTES * SECONDS_PER_MINUTE))


def is_on_grid(moment: int) -> bool:
    return (moment % SECONDS_PER_HOUR) % (SLOT_MINUTES * SECONDS_PER_MINUTE) == 0


def taken_starts(store, doctor_id: str) -> Set[int]:
    live = (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    return {a.start_at for a in appointments_of_doctor(store, doctor_id) if a.status in live}


def open_slots(store, doctor_id: str, day: str) -> List[int]:
    taken = taken_starts(store, doctor_id)
    return [slot for slot in slots_on(day)
            if slot not in taken and is_on_shift(store, doctor_id, slot) and not is_blocked(slot)]

"""Booking appointments and moving them through a visit."""

from typing import List

from ..config import MAX_ADVANCE_BOOKING_DAYS, SECONDS_PER_DAY, SLOT_MINUTES
from ..errors import InvalidTransition, OutsideRoster, SlotTaken, ValidationFailed
from ..models import (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED, APPOINTMENT_NO_SHOW,
                      Appointment)
from ..store.repositories import appointments_of_patient, get_appointment, get_doctor, get_patient, save_appointment
from .rosters import is_on_shift
from .slots import is_blocked, is_on_grid, taken_starts


def check_bookable(store, doctor_id: str, start_at: int, now: int) -> None:
    if start_at <= now:
        raise ValidationFailed(["an appointment must be in the future"])
    if start_at > now + MAX_ADVANCE_BOOKING_DAYS * SECONDS_PER_DAY:
        raise ValidationFailed([f"appointments can be booked up to {MAX_ADVANCE_BOOKING_DAYS} days ahead"])
    if not is_on_grid(start_at):
        raise ValidationFailed([f"appointments start on {SLOT_MINUTES} minute slots"])
    if is_blocked(start_at):
        raise ValidationFailed(["no appointments are offered at that time"])
    if not is_on_shift(store, doctor_id, start_at):
        raise OutsideRoster(doctor_id, start_at)
    if start_at in taken_starts(store, doctor_id):
        raise SlotTaken(doctor_id, start_at)


def book_appointment(store, patient_id: str, doctor_id: str, start_at: int, reason: str, clock) -> Appointment:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    now = clock.now()
    check_bookable(store, doctor_id, start_at, now)
    appointment = Appointment(id=store.next_id("APT"), patient_id=patient_id, doctor_id=doctor_id,
                              start_at=start_at, reason=reason, minutes=SLOT_MINUTES, created_at=now)
    return save_appointment(store, appointment)


def _move(store, appointment_id: str, expected: str, wanted: str) -> Appointment:
    appointment = get_appointment(store, appointment_id)
    if appointment.status != expected:
        raise InvalidTransition(appointment.status, wanted)
    appointment.status = wanted
    return save_appointment(store, appointment)


def check_in(store, appointment_id: str, clock) -> Appointment:
    appointment = _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    appointment.checked_in_at = clock.now()
    return save_appointment(store, appointment)


def complete_visit(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED)


def mark_no_show(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_NO_SHOW)


def upcoming_for(store, patient_id: str, now: int) -> List[Appointment]:
    found = [a for a in appointments_of_patient(store, patient_id) if a.status == APPOINTMENT_BOOKED and a.start_at > now]
    return sorted(found, key=lambda a: a.start_at)

"""Patients waiting for an earlier time with a doctor."""

from typing import List, Optional

from ..config import SECONDS_PER_HOUR, WAITLIST_OFFER_HOURS
from ..errors import ValidationFailed
from ..models import WaitlistEntry
from ..store.repositories import get_doctor, get_patient


def join_waitlist(store, patient_id: str, doctor_id: str, clock) -> WaitlistEntry:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    if any(e.patient_id == patient_id and e.doctor_id == doctor_id for e in store.waitlist):
        raise ValidationFailed(["that patient is already waiting for this doctor"])
    entry = WaitlistEntry(id=store.next_id("WLT"), patient_id=patient_id, doctor_id=doctor_id, added_at=clock.now())
    store.waitlist.append(entry)
    return entry


def waiting_for(store, doctor_id: str) -> List[WaitlistEntry]:
    found = [e for e in store.waitlist if e.doctor_id == doctor_id and e.offered_start is None]
    return sorted(found, key=lambda e: e.added_at)


def offer_next(store, doctor_id: str, start_at: int, clock) -> Optional[WaitlistEntry]:
    """Offers a freed time to whoever has waited longest, if anyone is waiting."""
    waiting = waiting_for(store, doctor_id)
    if not waiting:
        return None
    entry = waiting[0]
    entry.offered_start = start_at
    entry.offer_expires_at = clock.now() + WAITLIST_OFFER_HOURS * SECONDS_PER_HOUR
    return entry


def expire_offers(store, now: int) -> int:
    expired = [e for e in store.waitlist if e.offer_expires_at is not None and e.offer_expires_at < now]
    for entry in expired:
        entry.offered_start = None
        entry.offer_expires_at = None
    return len(expired)

from functools import wraps

@dataclass
class Context:
    store: object
    clock: object
    sender: object


def handled(function):
    """Turns expected errors into a response the web layer can send."""

    @wraps(function)
    def wrapper(request, context):
        try:
            return function(request, context)
        except ClinicError as problem:
            return {"error": problem.code, "message": str(problem)}

    return wrapper


@handled
def register(request, context):
    require(request, "name")
    patient = register_patient(context.store, request["name"], request.get("phone", ""), request.get("email", ""),
                               request.get("date_of_birth", ""), context.clock)
    return {"patient_id": patient.id}

def available_slots(request, context):
    require(request, "doctor_id", "day")
    slots = open_slots(context.store, request["doctor_id"], request["day"])
    return {"day": request["day"], "times": [time_of(slot) for slot in slots], "starts": slots}


@handled
def book(request, context):
    check_booking(request)
    appointment = book_appointment(context.store, request["patient_id"], request["doctor_id"], request["start_at"],
                                   request.get("reason", ""), context.clock)
    return {"appointment_id": appointment.id, "day": day_of(appointment.start_at), "time": time_of(appointment.start_at)}


@handled
def cancel(request, context):
    require(request, "appointment_id")
    appointment = cancel_appointment(context.store, request["appointment_id"], context.clock)
    return {"appointment_id": appointment.id, "status": appointment.status, "late": appointment.late_cancel}


@handled
def triage(request, context):
    require(request, "patient_id")
    case = admit(context.store, request["patient_id"], vitals_from(request), context.clock)
    return {"case_id": case.id, "score": case.score, "level": case.level}


@handled
def write_prescription(request, context):
    require(request, "patient_id", "doctor_id", "drug_code", "dose_mg", "days")
    prescription = prescribe(context.store, request["patient_id"], request["doctor_id"], request["drug_code"],
                             request["dose_mg"], request["days"], context.clock,
                             doses_per_day=int(request.get("doses_per_day", 1)), refills=int(request.get("refills", 0)))
    return {"prescription_id": prescription.id}


@handled
def invoice_visit(request, context):
    require(request, "appointment_id")
    invoice = create_invoice(context.store, request["appointment_id"], context.clock)
    return {"invoice_id": invoice.id, "total_cents": invoice.total_cents, "text": render_invoice(invoice)}


@handled
def pay(request, context):
    check_payment(request)
    invoice = record_payment(context.store, request["invoice_id"], request["amount_cents"], request["method"], context.clock)
    return {"invoice_id": invoice.id, "status": invoice.status, "outstanding_cents": outstanding_cents(invoice)}

"""Working out which appointment times can be offered."""

from typing import Dict, List, Set, Tuple

from ..clock import hour_of, moment_at, weekday_of
from ..config import CLINIC_CLOSE_HOUR, CLINIC_OPEN_HOUR, SECONDS_PER_HOUR, SECONDS_PER_MINUTE, SLOT_MINUTES
from ..models import APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN
from ..store.repositories import appointments_of_doctor
from .rosters import is_on_shift

WEEKLY_BLOCKS: Dict[int, Tuple[int, int]] = {2: (13, 15)}


def is_blocked(moment: int) -> bool:
    block = WEEKLY_BLOCKS.get(weekday_of(moment))
    return block is not None and block[0] <= hour_of(moment) < block[1]


def slots_on(day: str) -> List[int]:
    first = moment_at(day, CLINIC_OPEN_HOUR)
    last = moment_at(day, CLINIC_CLOSE_HOUR)
    return list(range(first, last, SLOT_MINUTES * SECONDS_PER_MINUTE))


def is_on_grid(moment: int) -> bool:
    return (moment % SECONDS_PER_HOUR) % (SLOT_MINUTES * SECONDS_PER_MINUTE) == 0


def taken_starts(store, doctor_id: str) -> Set[int]:
    live = (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    return {a.start_at for a in appointments_of_doctor(store, doctor_id) if a.status in live}


def open_slots(store, doctor_id: str, day: str) -> List[int]:
    taken = taken_starts(store, doctor_id)
    return [slot for slot in slots_on(day)
            if slot not in taken and is_on_shift(store, doctor_id, slot) and not is_blocked(slot)]

"""Booking appointments and moving them through a visit."""

from typing import List

from ..config import MAX_ADVANCE_BOOKING_DAYS, SECONDS_PER_DAY, SLOT_MINUTES
from ..errors import InvalidTransition, OutsideRoster, SlotTaken, ValidationFailed
from ..models import (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED, APPOINTMENT_NO_SHOW,
                      Appointment)
from ..store.repositories import appointments_of_patient, get_appointment, get_doctor, get_patient, save_appointment
from .rosters import is_on_shift
from .slots import is_blocked, is_on_grid, taken_starts


def check_bookable(store, doctor_id: str, start_at: int, now: int) -> None:
    if start_at <= now:
        raise ValidationFailed(["an appointment must be in the future"])
    if start_at > now + MAX_ADVANCE_BOOKING_DAYS * SECONDS_PER_DAY:
        raise ValidationFailed([f"appointments can be booked up to {MAX_ADVANCE_BOOKING_DAYS} days ahead"])
    if not is_on_grid(start_at):
        raise ValidationFailed([f"appointments start on {SLOT_MINUTES} minute slots"])
    if is_blocked(start_at):
        raise ValidationFailed(["no appointments are offered at that time"])
    if not is_on_shift(store, doctor_id, start_at):
        raise OutsideRoster(doctor_id, start_at)
    if start_at in taken_starts(store, doctor_id):
        raise SlotTaken(doctor_id, start_at)


def book_appointment(store, patient_id: str, doctor_id: str, start_at: int, reason: str, clock) -> Appointment:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    now = clock.now()
    check_bookable(store, doctor_id, start_at, now)
    appointment = Appointment(id=store.next_id("APT"), patient_id=patient_id, doctor_id=doctor_id,
                              start_at=start_at, reason=reason, minutes=SLOT_MINUTES, created_at=now)
    return save_appointment(store, appointment)


def _move(store, appointment_id: str, expected: str, wanted: str) -> Appointment:
    appointment = get_appointment(store, appointment_id)
    if appointment.status != expected:
        raise InvalidTransition(appointment.status, wanted)
    appointment.status = wanted
    return save_appointment(store, appointment)


def check_in(store, appointment_id: str, clock) -> Appointment:
    appointment = _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    appointment.checked_in_at = clock.now()
    return save_appointment(store, appointment)


def complete_visit(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED)


def mark_no_show(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_NO_SHOW)


def upcoming_for(store, patient_id: str, now: int) -> List[Appointment]:
    found = [a for a in appointments_of_patient(store, patient_id) if a.status == APPOINTMENT_BOOKED and a.start_at > now]
    return sorted(found, key=lambda a: a.start_at)

"""Patients waiting for an earlier time with a doctor."""

from typing import List, Optional

from ..config import SECONDS_PER_HOUR, WAITLIST_OFFER_HOURS
from ..errors import ValidationFailed
from ..models import WaitlistEntry
from ..store.repositories import get_doctor, get_patient


def join_waitlist(store, patient_id: str, doctor_id: str, clock) -> WaitlistEntry:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    if any(e.patient_id == patient_id and e.doctor_id == doctor_id for e in store.waitlist):
        raise ValidationFailed(["that patient is already waiting for this doctor"])
    entry = WaitlistEntry(id=store.next_id("WLT"), patient_id=patient_id, doctor_id=doctor_id, added_at=clock.now())
    store.waitlist.append(entry)
    return entry


def waiting_for(store, doctor_id: str) -> List[WaitlistEntry]:
    found = [e for e in store.waitlist if e.doctor_id == doctor_id and e.offered_start is None]
    return sorted(found, key=lambda e: e.added_at)


def offer_next(store, doctor_id: str, start_at: int, clock) -> Optional[WaitlistEntry]:
    """Offers a freed time to whoever has waited longest, if anyone is waiting."""
    waiting = waiting_for(store, doctor_id)
    if not waiting:
        return None
    entry = waiting[0]
    entry.offered_start = start_at
    entry.offer_expires_at = clock.now() + WAITLIST_OFFER_HOURS * SECONDS_PER_HOUR
    return entry


def expire_offers(store, now: int) -> int:
    expired = [e for e in store.waitlist if e.offer_expires_at is not None and e.offer_expires_at < now]
    for entry in expired:
        entry.offered_start = None
        entry.offer_expires_at = None
    return len(expired)

"""Working out which appointment times can be offered."""

from typing import Dict, List, Set, Tuple

from ..clock import hour_of, moment_at, weekday_of
from ..config import CLINIC_CLOSE_HOUR, CLINIC_OPEN_HOUR, SECONDS_PER_HOUR, SECONDS_PER_MINUTE, SLOT_MINUTES
from ..models import APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN
from ..store.repositories import appointments_of_doctor
from .rosters import is_on_shift

WEEKLY_BLOCKS: Dict[int, Tuple[int, int]] = {2: (13, 15)}


def is_blocked(moment: int) -> bool:
    block = WEEKLY_BLOCKS.get(weekday_of(moment))
    return block is not None and block[0] <= hour_of(moment) < block[1]


def slots_on(day: str) -> List[int]:
    first = moment_at(day, CLINIC_OPEN_HOUR)
    last = moment_at(day, CLINIC_CLOSE_HOUR)
    return list(range(first, last, SLOT_MINUTES * SECONDS_PER_MINUTE))


def is_on_grid(moment: int) -> bool:
    return (moment % SECONDS_PER_HOUR) % (SLOT_MINUTES * SECONDS_PER_MINUTE) == 0


def taken_starts(store, doctor_id: str) -> Set[int]:
    live = (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    return {a.start_at for a in appointments_of_doctor(store, doctor_id) if a.status in live}


def open_slots(store, doctor_id: str, day: str) -> List[int]:
    taken = taken_starts(store, doctor_id)
    return [slot for slot in slots_on(day)
            if slot not in taken and is_on_shift(store, doctor_id, slot) and not is_blocked(slot)]

"""Booking appointments and moving them through a visit."""

from typing import List

from ..config import MAX_ADVANCE_BOOKING_DAYS, SECONDS_PER_DAY, SLOT_MINUTES
from ..errors import InvalidTransition, OutsideRoster, SlotTaken, ValidationFailed
from ..models import (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED, APPOINTMENT_NO_SHOW,
                      Appointment)
from ..store.repositories import appointments_of_patient, get_appointment, get_doctor, get_patient, save_appointment
from .rosters import is_on_shift
from .slots import is_blocked, is_on_grid, taken_starts


def check_bookable(store, doctor_id: str, start_at: int, now: int) -> None:
    if start_at <= now:
        raise ValidationFailed(["an appointment must be in the future"])
    if start_at > now + MAX_ADVANCE_BOOKING_DAYS * SECONDS_PER_DAY:
        raise ValidationFailed([f"appointments can be booked up to {MAX_ADVANCE_BOOKING_DAYS} days ahead"])
    if not is_on_grid(start_at):
        raise ValidationFailed([f"appointments start on {SLOT_MINUTES} minute slots"])
    if is_blocked(start_at):
        raise ValidationFailed(["no appointments are offered at that time"])
    if not is_on_shift(store, doctor_id, start_at):
        raise OutsideRoster(doctor_id, start_at)
    if start_at in taken_starts(store, doctor_id):
        raise SlotTaken(doctor_id, start_at)


def book_appointment(store, patient_id: str, doctor_id: str, start_at: int, reason: str, clock) -> Appointment:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    now = clock.now()
    check_bookable(store, doctor_id, start_at, now)
    appointment = Appointment(id=store.next_id("APT"), patient_id=patient_id, doctor_id=doctor_id,
                              start_at=start_at, reason=reason, minutes=SLOT_MINUTES, created_at=now)
    return save_appointment(store, appointment)


def _move(store, appointment_id: str, expected: str, wanted: str) -> Appointment:
    appointment = get_appointment(store, appointment_id)
    if appointment.status != expected:
        raise InvalidTransition(appointment.status, wanted)
    appointment.status = wanted
    return save_appointment(store, appointment)


def check_in(store, appointment_id: str, clock) -> Appointment:
    appointment = _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    appointment.checked_in_at = clock.now()
    return save_appointment(store, appointment)


def complete_visit(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED)


def mark_no_show(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_NO_SHOW)


def upcoming_for(store, patient_id: str, now: int) -> List[Appointment]:
    found = [a for a in appointments_of_patient(store, patient_id) if a.status == APPOINTMENT_BOOKED and a.start_at > now]
    return sorted(found, key=lambda a: a.start_at)

"""Patients waiting for an earlier time with a doctor."""

from typing import List, Optional

from ..config import SECONDS_PER_HOUR, WAITLIST_OFFER_HOURS
from ..errors import ValidationFailed
from ..models import WaitlistEntry
from ..store.repositories import get_doctor, get_patient


def join_waitlist(store, patient_id: str, doctor_id: str, clock) -> WaitlistEntry:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    if any(e.patient_id == patient_id and e.doctor_id == doctor_id for e in store.waitlist):
        raise ValidationFailed(["that patient is already waiting for this doctor"])
    entry = WaitlistEntry(id=store.next_id("WLT"), patient_id=patient_id, doctor_id=doctor_id, added_at=clock.now())
    store.waitlist.append(entry)
    return entry


def waiting_for(store, doctor_id: str) -> List[WaitlistEntry]:
    found = [e for e in store.waitlist if e.doctor_id == doctor_id and e.offered_start is None]
    return sorted(found, key=lambda e: e.added_at)


def offer_next(store, doctor_id: str, start_at: int, clock) -> Optional[WaitlistEntry]:
    """Offers a freed time to whoever has waited longest, if anyone is waiting."""
    waiting = waiting_for(store, doctor_id)
    if not waiting:
        return None
    entry = waiting[0]
    entry.offered_start = start_at
    entry.offer_expires_at = clock.now() + WAITLIST_OFFER_HOURS * SECONDS_PER_HOUR
    return entry


def expire_offers(store, now: int) -> int:
    expired = [e for e in store.waitlist if e.offer_expires_at is not None and e.offer_expires_at < now]
    for entry in expired:
        entry.offered_start = None
        entry.offer_expires_at = None
    return len(expired)

@dataclass
class Context:
    store: object

def handled(function):
    """Turns expected errors into a response the web layer can send."""

    @wraps(function)
    def wrapper(request, context):

            return function(request, context)
        except ClinicError as problem:

    return wrapper
