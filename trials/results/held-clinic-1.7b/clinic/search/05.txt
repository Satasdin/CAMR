8: def percent_of(amount_cents: int, percent: int) -> int:
9:     """A percentage of an amount, half a cent going up."""
10:     if amount_cents < 0:
11:         raise ValueError("amount must not be negative")
12:     return (amount_cents * percent + 50) // 100
13: 
14: 
15: def add(amounts: Iterable[int]) -> int:
16:     return sum(amounts)
17: 
18: 
19: def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
20:     sign = "-" if amount_cents < 0 else ""
21:     whole, part = divmod(abs(amount_cents), 100)
22:     return f"{sign}{currency} {whole:,}.{part:02d}"
23: 
24: 
25: def parse_amount(text: str) -> int:
26:     """Reads an amount typed by staff, such as '1,250.50', into cents."""
27:     cleaned = text.replace(",", "").strip()
28:     if not cleaned:
29:         raise ValueError("empty amount")
30:     negative = cleaned.startswith("-")
31:     if negative:
32:         cleaned = cleaned[1:]
33:     whole, _, part = cleaned.partition(".")
34:     if not whole.isdigit() and whole != "":
35:         raise ValueError(f"not an amount: {text}")
36:     if part and (not part.isdigit() or len(part) > 2):
37:         raise ValueError(f"not an amount: {text}")
38:     cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
39:     return -cents if negative else cents
40: 
41: 
42: def split_evenly(amount_cents: int, parts: int) -> list:
43:     """Splits an amount so the parts always add back to the amount."""
44:     if parts <= 0:
45:         raise ValueError("parts must be positive")
46:     share, remainder = divmod(amount_cents, parts)
47:     return [share + 1 if index < remainder else share for index in range(parts)]

9:     """A percentage of an amount, half a cent going up."""
10:     if amount_cents < 0:
11:         raise ValueError("amount must not be negative")
12:     return (amount_cents * percent + 50) // 100
13:

15: def add(amounts: Iterable[int]) -> int:
16:     return sum(amounts)
17: 
18: 
19: def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
20:     sign = "-" if amount_cents < 0 else ""
21:     whole, part = divmod(abs(amount_cents), 100)
22:     return f"{sign}{currency} {whole:,}.{part:02d}"
23: 
24: 
25: def parse_amount(text: str) -> int:
26:     """Reads an amount typed by staff, such as '1,250.50', into cents."""
27:     cleaned = text.replace(",", "").strip()
28:     if not cleaned:
29:         raise ValueError("empty amount")
30:     negative = cleaned.startswith("-")
31:     if negative:
32:         cleaned = cleaned[1:]
33:     whole, _, part = cleaned.partition(".")
34:     if not whole.isdigit() and whole != "":
35:         raise ValueError(f"not an amount: {text}")
36:     if part and (not part.isdigit() or len(part) > 2):
37:         raise ValueError(f"not an amount: {text}")
38:     cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
39:     return -cents if negative else cents
40: 
41: 
42: def split_evenly(amount_cents: int, parts: int) -> list:
43:     """Splits an amount so the parts always add back to the amount."""
44:     if parts <= 0:
45:         raise ValueError("parts must be positive")
46:     share, remainder = divmod(amount_cents, parts)
47:     return [share + 1 if index < remainder else share for index in range(parts)]

"""Checking how much of a drug is prescribed."""

import math

from ..errors import ValidationFailed
from ..models import Prescription
from .formulary import Drug, get_drug


def check_single_dose(drug: Drug, dose_mg: float) -> None:
    if dose_mg <= 0:
        raise ValidationFailed(["a dose must be more than zero"])
    if drug.max_single_mg and dose_mg > drug.max_single_mg:
        raise ValidationFailed([f"{drug.name} is limited to {drug.max_single_mg} mg at a time"])


def daily_total(dose_mg: float, doses_per_day: int) -> float:
    return dose_mg * doses_per_day


def course_total(prescription: Prescription) -> float:
    return daily_total(prescription.dose_mg, prescription.doses_per_day) * prescription.days


def tablets_per_dose(drug: Drug, dose_mg: float) -> float:
    if drug.form != "tablet":
        raise ValidationFailed([f"{drug.name} is not a tablet"])
    return dose_mg / drug.strength_mg


def paediatric_dose(drug_code: str, weight_kg: float) -> float:
    """The amount of a drug for one dose, worked out from a child's weight."""
    drug = get_drug(drug_code)
    if weight_kg <= 0:
        raise ValidationFailed(["a weight must be more than zero"])
    amount = min(drug.mg_per_kg * weight_kg, drug.max_single_mg)
    return math.floor(amount / drug.step_mg) * drug.step_mg

    if prescription.refills_left <= 0:
        raise RefillsExhausted(prescription_id)
    prescription.refills_left -= 1
    return save_prescription(store, prescription)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """A percentage of an amount, half a cent going up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

from typing import FrozenSet, Iterable, List, Optional

BAD_PAIRS: List[FrozenSet[str]] = [
    frozenset(("merixin", "varenol")),
    frozenset(("bezorin", "coldaxine")),
]


def clash(first: str, second: str) -> bool:
    return frozenset((first.lower(), second.lower())) in BAD_PAIRS


def first_clash(ingredient: str, others: Iterable[str]) -> Optional[str]:
    for other in others:
        if clash(ingredient, other):
            return other
    return None

"""A week's visits by department."""

from collections import Counter
from typing import Dict, List

from ..clock import day_of
from ..config import SECONDS_PER_DAY
from ..models import APPOINTMENT_COMPLETED
from ..store.repositories import get_doctor


def days_of_week(first_moment: int) -> List[str]:
    return [day_of(first_moment + offset * SECONDS_PER_DAY) for offset in range(7)]


def visits_by_department(store, first_moment: int) -> Dict[str, int]:
    days = set(days_of_week(first_moment))
    counted: Counter = Counter()
    for appointment in store.appointments.values():
        if appointment.status == APPOINTMENT_COMPLETED and day_of(appointment.start_at) in days:
            counted[get_doctor(store, appointment.doctor_id).department] += 1
    return dict(counted)


def render_weekly(counts: Dict[str, int]) -> str:
    rows = ["Visits by department"]
    rows += [f"  {department}: {counts[department]}" for department in sorted(counts)]
    rows.append(f"Total: {sum(counts.values())}")
    return "\n".join(rows)

"""Checking how much of a drug is prescribed."""

import math

from ..errors import ValidationFailed
from ..models import Prescription
from .formulary import Drug, get_drug


def check_single_dose(drug: Drug, dose_mg: float) -> None:
    if dose_mg <= 0:
        raise ValidationFailed(["a dose must be more than zero"])
    if drug.max_single_mg and dose_mg > drug.max_single_mg:
        raise ValidationFailed([f"{drug.name} is limited to {drug.max_single_mg} mg at a time"])


def daily_total(dose_mg: float, doses_per_day: int) -> float:
    return dose_mg * doses_per_day


def course_total(prescription: Prescription) -> float:
    return daily_total(prescription.dose_mg, prescription.doses_per_day) * prescription.days


def tablets_per_dose(drug: Drug, dose_mg: float) -> float:
    if drug.form != "tablet":
        raise ValidationFailed([f"{drug.name} is not a tablet"])
    return dose_mg / drug.strength_mg


def paediatric_dose(drug_code: str, weight_kg: float) -> float:
    """The amount of a drug for one dose, worked out from a child's weight."""
    drug = get_drug(drug_code)
    if weight_kg <= 0:
        raise ValidationFailed(["a weight must be more than zero"])
    amount = min(drug.mg_per_kg * weight_kg, drug.max_single_mg)
    return math.floor(amount / drug.step_mg) * drug.step_mg

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """A percentage of an amount, half a cent going up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

        super().__init__("; ".join(problems))

class InvalidTransition(ClinicError):
    code = "invalid_transition"

    def __init__(self, current, wanted):
        super().__init__(f"cannot move from {current} to {wanted}")
        self.current = current
        self.wanted = wanted


class SlotTaken(ClinicError):

    def __init__(self, doctor_id, start_at):

        self.doctor_id = doctor_id
        self.start_at = start_at


class OutsideRoster(ClinicError):
    code = "outside_roster"

        self.doctor_id = doctor_id
        self.start_at = start_at


class AllergyConflict(ClinicError):

"""Pairs of ingredients that must not be taken together."""

from typing import FrozenSet, Iterable, List, Optional

BAD_PAIRS: List[FrozenSet[str]] = [
    frozenset(("merixin", "varenol")),
    frozenset(("bezorin", "coldaxine")),
]


def clash(first: str, second: str) -> bool:
    return frozenset((first.lower(), second.lower())) in BAD_PAIRS


def first_clash(ingredient: str, others: Iterable[str]) -> Optional[str]:
    for other in others:
        if clash(ingredient, other):
            return other
    return None

"""A week's visits by department."""

from collections import Counter
from typing import Dict, List

from ..clock import day_of
from ..config import SECONDS_PER_DAY
from ..models import APPOINTMENT_COMPLETED
from ..store.repositories import get_doctor


def days_of_week(first_moment: int) -> List[str]:
    return [day_of(first_moment + offset * SECONDS_PER_DAY) for offset in range(7)]


def visits_by_department(store, first_moment: int) -> Dict[str, int]:
    days = set(days_of_week(first_moment))
    counted: Counter = Counter()
    for appointment in store.appointments.values():
        if appointment.status == APPOINTMENT_COMPLETED and day_of(appointment.start_at) in days:
            counted[get_doctor(store, appointment.doctor_id).department] += 1
    return dict(counted)


def render_weekly(counts: Dict[str, int]) -> str:
    rows = ["Visits by department"]
    rows += [f"  {department}: {counts[department]}" for department in sorted(counts)]
    rows.append(f"Total: {sum(counts.values())}")
    return "\n".join(rows)

def volume_ml(prescription: Prescription, drug: Drug) -> float:
    """How many millilitres one dose comes to."""

"""Checking how much of a drug is prescribed."""

import math

from ..errors import ValidationFailed
from ..models import Prescription
from .formulary import Drug, get_drug


def check_single_dose(drug: Drug, dose_mg: float) -> None:
    if dose_mg <= 0:
        raise ValidationFailed(["a dose must be more than zero"])
    if drug.max_single_mg and dose_mg > drug.max_single_mg:
        raise ValidationFailed([f"{drug.name} is limited to {drug.max_single_mg} mg at a time"])


def daily_total(dose_mg: float, doses_per_day: int) -> float:
    return dose_mg * doses_per_day


def course_total(prescription: Prescription) -> float:
    return daily_total(prescription.dose_mg, prescription.doses_per_day) * prescription.days


def tablets_per_dose(drug: Drug, dose_mg: float) -> float:
    if drug.form != "tablet":
        raise ValidationFailed([f"{drug.name} is not a tablet"])
    return dose_mg / drug.strength_mg


def paediatric_dose(drug_code: str, weight_kg: float) -> float:
    """The amount of a drug for one dose, worked out from a child's weight."""
    drug = get_drug(drug_code)
    if weight_kg <= 0:
        raise ValidationFailed(["a weight must be more than zero"])
    amount = min(drug.mg_per_kg * weight_kg, drug.max_single_mg)
    return math.floor(amount / drug.step_mg) * drug.step_mg

def set_refills(store, prescription_id: str, count: int) -> Prescription:
    if not 0 <= count <= MAX_REFILLS:
        raise ValidationFailed([f"refills must be between 0 and {MAX_REFILLS}"])
    prescription = get_prescription(store, prescription_id)
    prescription.refills_left = count
    return save_prescription(store, prescription)


def use_refill(store, prescription_id: str) -> Prescription:
    prescription = get_prescription(store, prescription_id)
    if not prescription.active:
        raise ValidationFailed([f"prescription {prescription_id} has ended"])
    if prescription.refills_left <= 0:
        raise RefillsExhausted(prescription_id)
    prescription.refills_left -= 1
    return save_prescription(store, prescription)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """A percentage of an amount, half a cent going up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Pairs of ingredients that must not be taken together."""

from typing import FrozenSet, Iterable, List, Optional

BAD_PAIRS: List[FrozenSet[str]] = [
    frozenset(("merixin", "varenol")),
    frozenset(("bezorin", "coldaxine")),
]


def clash(first: str, second: str) -> bool:
    return frozenset((first.lower(), second.lower())) in BAD_PAIRS


def first_clash(ingredient: str, others: Iterable[str]) -> Optional[str]:
    for other in others:
        if clash(ingredient, other):
            return other
    return None

"""A week's visits by department."""

from collections import Counter
from typing import Dict, List

from ..clock import day_of
from ..config import SECONDS_PER_DAY
from ..models import APPOINTMENT_COMPLETED
from ..store.repositories import get_doctor


def days_of_week(first_moment: int) -> List[str]:
    return [day_of(first_moment + offset * SECONDS_PER_DAY) for offset in range(7)]


def visits_by_department(store, first_moment: int) -> Dict[str, int]:
    days = set(days_of_week(first_moment))
    counted: Counter = Counter()
    for appointment in store.appointments.values():
        if appointment.status == APPOINTMENT_COMPLETED and day_of(appointment.start_at) in days:
            counted[get_doctor(store, appointment.doctor_id).department] += 1
    return dict(counted)


def render_weekly(counts: Dict[str, int]) -> str:
    rows = ["Visits by department"]
    rows += [f"  {department}: {counts[department]}" for department in sorted(counts)]
    rows.append(f"Total: {sum(counts.values())}")
    return "\n".join(rows)

"""What is printed on a bottle."""

from ..errors import ValidationFailed
from ..models import Patient, Prescription
from .formulary import Drug


def volume_ml(prescription: Prescription, drug: Drug) -> float:
    """How many millilitres one dose comes to."""
    if drug.mg_per_ml <= 0:
        raise ValidationFailed([f"{drug.name} is not a liquid"])
    return round(prescription.dose_mg / drug.mg_per_ml) * 0.5


def label_text(prescription: Prescription, drug: Drug, patient: Patient) -> str:
    volume = volume_ml(prescription, drug)
    return "\n".join([
        patient.name,
        drug.name,
        f"Take {volume:.1f} ml, {prescription.doses_per_day} time(s) a day for {prescription.days} days",
    ])
