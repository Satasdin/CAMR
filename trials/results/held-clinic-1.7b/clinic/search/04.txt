1: """Deciding who is reminded about an appointment, and how."""
2: 
3: from typing import List
4: 
5: from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
6: from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
7: from ..store.repositories import get_doctor, get_patient, save_appointment
8: from .templates import email_body, email_subject, sms_body
9: 
10: 
11: def due_for_reminder(appointment: Appointment, now: int) -> bool:
12:     if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
13:         return False
14:     return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR
15: 
16: 
17: def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
18:     out = []
19:     if patient.wants_email and patient.email:
20:         out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
21:                            body=email_body(patient, doctor, appointment), sent_at=now))
22:     if patient.wants_sms and patient.phone:
23:         out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, appointment), sent_at=now))
24:     return out
25: 
26: 
27: def send_reminders(store, sender, clock) -> int:
28:     """Sends reminders for every appointment coming up and returns how many messages went out."""
29:     now = clock.now()
30:     count = 0
31:     for appointment in list(store.appointments.values()):
32:         if not due_for_reminder(appointment, now):
33:             continue
34:         patient = get_patient(store, appointment.patient_id)
35:         doctor = get_doctor(store, appointment.doctor_id)
36:         count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
37:         appointment.reminded = True
38:         save_appointment(store, appointment)
39:     return count

1: # Clinic
2: 
3: A small outpatient clinic back end. It registers patients, books appointments against
4: doctors' schedules, triages walk-ins, writes prescriptions, bills visits and reminds
5: patients what is coming up.
6: 
7: ## Layout
8: 
9: - `clinic/patients`: the patient register, allergies and contact preferences.
10: - `clinic/scheduling`: rosters, open slots, appointments, cancellations and the waitlist.

12: - `clinic/prescriptions`: the formulary, doses, prescribing, refills, interactions and labels.
13: - `clinic/billing`: tariffs, invoices, payments, insurance claims and statements.
14: - `clinic/reminders`: message templates, delivery and who gets reminded about what.
15: - `clinic/api`: request validation and the handlers the web layer calls.
16: - `clinic/reports`: daily and weekly figures for management.
17: - `clinic/store`: an in-memory store with small repository helpers.
18: 
19: Money is held as whole cents. Times are whole seconds since the epoch, taken from
20: `clinic.clock` so tests can control them.

"""Deciding who is reminded about an appointment, and how."""

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject, sms_body


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    if patient.wants_sms and patient.phone:
        out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count

from typing import List, Optional

from ..config import MIN_NAME_LENGTH
from ..errors import ValidationFailed
from ..models import Patient
from ..store.repositories import get_patient, save_patient


def register_patient(store, name: str, phone: str, email: str, date_of_birth: str, clock) -> Patient:
    if len(name.strip()) < MIN_NAME_LENGTH:
        raise ValidationFailed(["a patient needs a name"])
    if phone and find_by_phone(store, phone) is not None:
        raise ValidationFailed([f"{phone} is already registered"])
    patient = Patient(id=store.next_id("PAT"), name=name.strip(), phone=phone.strip(), email=email.strip(),
                      date_of_birth=date_of_birth, created_at=clock.now())
    return save_patient(store, patient)


def find_by_phone(store, phone: str) -> Optional[Patient]:
    for patient in store.patients.values():
        if patient.phone == phone:
            return patient
    return None


def search_by_name(store, fragment: str) -> List[Patient]:
    needle = fragment.strip().lower()
    return sorted((p for p in store.patients.values() if needle in p.name.lower()), key=lambda p: p.name)


def deactivate_patient(store, patient_id: str) -> Patient:
    patient = get_patient(store, patient_id)
    patient.active = False
    return save_patient(store, patient)

from ..models import (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED, APPOINTMENT_NO_SHOW,
                      Appointment)
from ..store.repositories import appointments_of_patient, get_appointment, get_doctor, get_patient, save_appointment
from .rosters import is_on_shift
from .slots import is_blocked, is_on_grid, taken_starts


def check_bookable(store, doctor_id: str, start_at: int, now: int) -> None:
    if start_at <= now:
        raise ValidationFailed(["an appointment must be in the future"])
    if start_at > now + MAX_ADVANCE_BOOKING_DAYS * SECONDS_PER_DAY:
        raise ValidationFailed([f"appointments can be booked up to {MAX_ADVANCE_BOOKING_DAYS} days ahead"])
    if not is_on_grid(start_at):
        raise ValidationFailed([f"appointments start on {SLOT_MINUTES} minute slots"])
    if is_blocked(start_at):
        raise ValidationFailed(["no appointments are offered at that time"])
    if not is_on_shift(store, doctor_id, start_at):
        raise OutsideRoster(doctor_id, start_at)
    if start_at in taken_starts(store, doctor_id):
        raise SlotTaken(doctor_id, start_at)


def book_appointment(store, patient_id: str, doctor_id: str, start_at: int, reason: str, clock) -> Appointment:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    now = clock.now()
    check_bookable(store, doctor_id, start_at, now)
    appointment = Appointment(id=store.next_id("APT"), patient_id=patient_id, doctor_id=doctor_id,
                              start_at=start_at, reason=reason, minutes=SLOT_MINUTES, created_at=now)
    return save_appointment(store, appointment)


def _move(store, appointment_id: str, expected: str, wanted: str) -> Appointment:
    appointment = get_appointment(store, appointment_id)
    if appointment.status != expected:
        raise InvalidTransition(appointment.status, wanted)
    appointment.status = wanted
    return save_appointment(store, appointment)


def check_in(store, appointment_id: str, clock) -> Appointment:
    appointment = _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    appointment.checked_in_at = clock.now()
    return save_appointment(store, appointment)


def complete_visit(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED)


def mark_no_show(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_NO_SHOW)


def upcoming_for(store, patient_id: str, now: int) -> List[Appointment]:
    found = [a for a in appointments_of_patient(store, patient_id) if a.status == APPOINTMENT_BOOKED and a.start_at > now]
    return sorted(found, key=lambda a: a.start_at)

"""Where the service gets the time, so tests can control it."""

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600

"""Deciding who is reminded about an appointment, and how."""

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject, sms_body


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    if patient.wants_sms and patient.phone:
        out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count

from typing import List

from ..config import MAX_ADVANCE_BOOKING_DAYS, SECONDS_PER_DAY, SLOT_MINUTES
from ..errors import InvalidTransition, OutsideRoster, SlotTaken, ValidationFailed
from ..models import (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED, APPOINTMENT_NO_SHOW,
                      Appointment)
from ..store.repositories import appointments_of_patient, get_appointment, get_doctor, get_patient, save_appointment
from .rosters import is_on_shift
from .slots import is_blocked, is_on_grid, taken_starts


def check_bookable(store, doctor_id: str, start_at: int, now: int) -> None:
    if start_at <= now:
        raise ValidationFailed(["an appointment must be in the future"])
    if start_at > now + MAX_ADVANCE_BOOKING_DAYS * SECONDS_PER_DAY:
        raise ValidationFailed([f"appointments can be booked up to {MAX_ADVANCE_BOOKING_DAYS} days ahead"])
    if not is_on_grid(start_at):
        raise ValidationFailed([f"appointments start on {SLOT_MINUTES} minute slots"])
    if is_blocked(start_at):
        raise ValidationFailed(["no appointments are offered at that time"])
    if not is_on_shift(store, doctor_id, start_at):
        raise OutsideRoster(doctor_id, start_at)
    if start_at in taken_starts(store, doctor_id):
        raise SlotTaken(doctor_id, start_at)


def book_appointment(store, patient_id: str, doctor_id: str, start_at: int, reason: str, clock) -> Appointment:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    now = clock.now()
    check_bookable(store, doctor_id, start_at, now)
    appointment = Appointment(id=store.next_id("APT"), patient_id=patient_id, doctor_id=doctor_id,
                              start_at=start_at, reason=reason, minutes=SLOT_MINUTES, created_at=now)
    return save_appointment(store, appointment)


def _move(store, appointment_id: str, expected: str, wanted: str) -> Appointment:
    appointment = get_appointment(store, appointment_id)
    if appointment.status != expected:
        raise InvalidTransition(appointment.status, wanted)
    appointment.status = wanted
    return save_appointment(store, appointment)


def check_in(store, appointment_id: str, clock) -> Appointment:
    appointment = _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    appointment.checked_in_at = clock.now()
    return save_appointment(store, appointment)


def complete_visit(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED)


def mark_no_show(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_NO_SHOW)


def upcoming_for(store, patient_id: str, now: int) -> List[Appointment]:
    found = [a for a in appointments_of_patient(store, patient_id) if a.status == APPOINTMENT_BOOKED and a.start_at > now]
    return sorted(found, key=lambda a: a.start_at)

"""Where the service gets the time, so tests can control it."""

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600

12:     if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
13:         return False
14:     return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR

20:         out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),

22:     if patient.wants_sms and patient.phone:

24:     return out

28:     """Sends reminders for every appointment coming up and returns how many messages went out."""

"""Deciding who is reminded about an appointment, and how."""

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject, sms_body


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    if patient.wants_sms and patient.phone:
        out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count

"""Registering patients and finding them again."""

from typing import List, Optional

from ..config import MIN_NAME_LENGTH
from ..errors import ValidationFailed
from ..models import Patient
from ..store.repositories import get_patient, save_patient


def register_patient(store, name: str, phone: str, email: str, date_of_birth: str, clock) -> Patient:
    if len(name.strip()) < MIN_NAME_LENGTH:
        raise ValidationFailed(["a patient needs a name"])
    if phone and find_by_phone(store, phone) is not None:
        raise ValidationFailed([f"{phone} is already registered"])
    patient = Patient(id=store.next_id("PAT"), name=name.strip(), phone=phone.strip(), email=email.strip(),
                      date_of_birth=date_of_birth, created_at=clock.now())
    return save_patient(store, patient)


def find_by_phone(store, phone: str) -> Optional[Patient]:
    for patient in store.patients.values():
        if patient.phone == phone:
            return patient
    return None


def search_by_name(store, fragment: str) -> List[Patient]:
    needle = fragment.strip().lower()
    return sorted((p for p in store.patients.values() if needle in p.name.lower()), key=lambda p: p.name)


def deactivate_patient(store, patient_id: str) -> Patient:
    patient = get_patient(store, patient_id)
    patient.active = False
    return save_patient(store, patient)

"""Booking appointments and moving them through a visit."""

from typing import List

from ..config import MAX_ADVANCE_BOOKING_DAYS, SECONDS_PER_DAY, SLOT_MINUTES
from ..errors import InvalidTransition, OutsideRoster, SlotTaken, ValidationFailed
from ..models import (APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED, APPOINTMENT_NO_SHOW,
                      Appointment)
from ..store.repositories import appointments_of_patient, get_appointment, get_doctor, get_patient, save_appointment
from .rosters import is_on_shift
from .slots import is_blocked, is_on_grid, taken_starts


def check_bookable(store, doctor_id: str, start_at: int, now: int) -> None:
    if start_at <= now:
        raise ValidationFailed(["an appointment must be in the future"])
    if start_at > now + MAX_ADVANCE_BOOKING_DAYS * SECONDS_PER_DAY:
        raise ValidationFailed([f"appointments can be booked up to {MAX_ADVANCE_BOOKING_DAYS} days ahead"])
    if not is_on_grid(start_at):
        raise ValidationFailed([f"appointments start on {SLOT_MINUTES} minute slots"])
    if is_blocked(start_at):
        raise ValidationFailed(["no appointments are offered at that time"])
    if not is_on_shift(store, doctor_id, start_at):
        raise OutsideRoster(doctor_id, start_at)
    if start_at in taken_starts(store, doctor_id):
        raise SlotTaken(doctor_id, start_at)


def book_appointment(store, patient_id: str, doctor_id: str, start_at: int, reason: str, clock) -> Appointment:
    get_patient(store, patient_id)
    get_doctor(store, doctor_id)
    now = clock.now()
    check_bookable(store, doctor_id, start_at, now)
    appointment = Appointment(id=store.next_id("APT"), patient_id=patient_id, doctor_id=doctor_id,
                              start_at=start_at, reason=reason, minutes=SLOT_MINUTES, created_at=now)
    return save_appointment(store, appointment)


def _move(store, appointment_id: str, expected: str, wanted: str) -> Appointment:
    appointment = get_appointment(store, appointment_id)
    if appointment.status != expected:
        raise InvalidTransition(appointment.status, wanted)
    appointment.status = wanted
    return save_appointment(store, appointment)


def check_in(store, appointment_id: str, clock) -> Appointment:
    appointment = _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_CHECKED_IN)
    appointment.checked_in_at = clock.now()
    return save_appointment(store, appointment)


def complete_visit(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_CHECKED_IN, APPOINTMENT_COMPLETED)


def mark_no_show(store, appointment_id: str) -> Appointment:
    return _move(store, appointment_id, APPOINTMENT_BOOKED, APPOINTMENT_NO_SHOW)


def upcoming_for(store, patient_id: str, now: int) -> List[Appointment]:
    found = [a for a in appointments_of_patient(store, patient_id) if a.status == APPOINTMENT_BOOKED and a.start_at > now]
    return sorted(found, key=lambda a: a.start_at)

"""Where the service gets the time, so tests can control it."""

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600
