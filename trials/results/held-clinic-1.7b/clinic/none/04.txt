"""The wording of reminders."""

from ..clock import day_of, time_of
from ..config import DEFAULT_SETTINGS, Settings
from ..models import Appointment, Doctor, Patient


def location_line(doctor: Doctor) -> str:
    return f"{doctor.name}, {doctor.department}, room {doctor.room}"


def email_subject(appointment: Appointment, settings: Settings = DEFAULT_SETTINGS) -> str:
    return f"{settings.clinic_name}: your appointment on {day_of(appointment.start_at)}"


def email_body(patient: Patient, doctor: Doctor, appointment: Appointment, settings: Settings = DEFAULT_SETTINGS) -> str:
    return "\n".join([
        f"Dear {patient.first_name},",
        f"You have an appointment on {day_of(appointment.start_at)} at {time_of(appointment.start_at)}.",
        f"You will be seen by: {location_line(doctor)}.",
        f"Reason for visit: {appointment.reason}",
        f"Booking reference: {appointment.id}",
        f"To change it, call {settings.front_desk_phone}.",
    ])


def sms_body(patient: Patient, appointment: Appointment, settings: Settings = DEFAULT_SETTINGS) -> str:
    return (f"{patient.first_name}, you have an appointment at {settings.clinic_name} on "
            f"{day_of(appointment.start_at)} at {time_of(appointment.start_at)}. "
            f"Booking reference {appointment.id}. To change it, call {settings.front_desk_phone}.")

"""Registering patients and finding them again."""

from typing import List, Optional

from ..config import MIN_NAME_LENGTH
from ..errors import ValidationFailed
from ..models import Patient
from ..store.repositories import get_patient, save_patient


def register_patient(store, name: str, phone: str, email: str, date_of_birth: str, clock) -> Patient:
    if len(name.strip()) < MIN_NAME_LENGTH:
        raise ValidationFailed(["a patient needs a name"])
    if phone and find_by_phone(store, phone) is not None:
        raise ValidationFailed([f"{phone} is already registered"])
    patient = Patient(id=store.next_id("PAT"), name=name.strip(), phone=phone.strip(), email=email.strip(),
                      date_of_birth=date_of_birth, created_at=clock.now())
    return save_patient(store, patient)


def find_by_phone(store, phone: str) -> Optional[Patient]:
    for patient in store.patients.values():
        if patient.phone == phone:
            return patient
    return None


def search_by_name(store, fragment: str) -> List[Patient]:
    needle = fragment.strip().lower()
    return sorted((p for p in store.patients.values() if needle in p.name.lower()), key=lambda p: p.name)


def deactivate_patient(store, patient_id: str) -> Patient:
    patient = get_patient(store, patient_id)
    patient.active = False
    return save_patient(store, patient)

from typing import List

from ..models import Patient
from ..store.repositories import get_patient, save_patient


def normalise(ingredient: str) -> str:
    return ingredient.strip().lower()


def add_allergy(store, patient_id: str, ingredient: str) -> Patient:
    patient = get_patient(store, patient_id)
    wanted = normalise(ingredient)
    if wanted not in [normalise(known) for known in patient.allergies]:
        patient.allergies.append(wanted)
    return save_patient(store, patient)


def remove_allergy(store, patient_id: str, ingredient: str) -> Patient:
    patient = get_patient(store, patient_id)
    unwanted = normalise(ingredient)
    patient.allergies = [known for known in patient.allergies if normalise(known) != unwanted]
    return save_patient(store, patient)


def is_allergic(patient: Patient, ingredient: str) -> bool:
    return normalise(ingredient) in [normalise(known) for known in patient.allergies]


def allergy_list(patient: Patient) -> List[str]:
    return sorted(normalise(known) for known in patient.allergies)

"""Registering patients and finding them again."""

from typing import List, Optional

from ..config import MIN_NAME_LENGTH
from ..errors import ValidationFailed
from ..models import Patient
from ..store.repositories import get_patient, save_patient


def register_patient(store, name: str, phone: str, email: str, date_of_birth: str, clock) -> Patient:
    if len(name.strip()) < MIN_NAME_LENGTH:
        raise ValidationFailed(["a patient needs a name"])
    if phone and find_by_phone(store, phone) is not None:
        raise ValidationFailed([f"{phone} is already registered"])
    patient = Patient(id=store.next_id("PAT"), name=name.strip(), phone=phone.strip(), email=email.strip(),
                      date_of_birth=date_of_birth, created_at=clock.now())
    return save_patient(store, patient)


def find_by_phone(store, phone: str) -> Optional[Patient]:
    for patient in store.patients.values():
        if patient.phone == phone:
            return patient
    return None


def search_by_name(store, fragment: str) -> List[Patient]:
    needle = fragment.strip().lower()
    return sorted((p for p in store.patients.values() if needle in p.name.lower()), key=lambda p: p.name)


def deactivate_patient(store, patient_id: str) -> Patient:
    patient = get_patient(store, patient_id)
    patient.active = False
    return save_patient(store, patient)
