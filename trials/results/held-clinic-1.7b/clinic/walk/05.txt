20:     sign = "-" if amount_cents < 0 else ""
21:     whole, part = divmod(abs(amount_cents), 100)
22:     return f"{sign}{currency} {whole:,}.{part:02d}"
23: 
24: 
25: def parse_amount(text: str) -> int:
26:     """Reads an amount typed by staff, such as '1,250.50', into cents."""
27:     cleaned = text.replace(",", "").strip()
28:     if not cleaned:
29:         raise ValueError("empty amount")
30:     negative = cleaned.startswith("-")
31:     if negative:
32:         cleaned = cleaned[1:]
33:     whole, _, part = cleaned.partition(".")
34:     if not whole.isdigit() and whole != "":
35:         raise ValueError(f"not an amount: {text}")
36:     if part and (not part.isdigit() or len(part) > 2):
37:         raise ValueError(f"not an amount: {text}")
38:     cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
39:     return -cents if negative else cents
40: 
41: 
42: def split_evenly(amount_cents: int, parts: int) -> list:
43:     """Splits an amount so the parts always add back to the amount."""
44:     if parts <= 0:
45:         raise ValueError("parts must be positive")
46:     share, remainder = divmod(amount_cents, parts)
47:     return [share + 1 if index < remainder else share for index in range(parts)]

"""Checking how much of a drug is prescribed."""

import math

from ..errors import ValidationFailed
from ..models import Prescription
from .formulary import Drug, get_drug


def check_single_dose(drug: Drug, dose_mg: float) -> None:
    if dose_mg <= 0:
        raise ValidationFailed(["a dose must be more than zero"])
    if drug.max_single_mg and dose_mg > drug.max_single_mg:
        raise ValidationFailed([f"{drug.name} is limited to {drug.max_single_mg} mg at a time"])


def daily_total(dose_mg: float, doses_per_day: int) -> float:
    return dose_mg * doses_per_day


def course_total(prescription: Prescription) -> float:
    return daily_total(prescription.dose_mg, prescription.doses_per_day) * prescription.days


def tablets_per_dose(drug: Drug, dose_mg: float) -> float:
    if drug.form != "tablet":
        raise ValidationFailed([f"{drug.name} is not a tablet"])
    return dose_mg / drug.strength_mg


def paediatric_dose(drug_code: str, weight_kg: float) -> float:
    """The amount of a drug for one dose, worked out from a child's weight."""
    drug = get_drug(drug_code)
    if weight_kg <= 0:
        raise ValidationFailed(["a weight must be more than zero"])
    amount = min(drug.mg_per_kg * weight_kg, drug.max_single_mg)
    return math.floor(amount / drug.step_mg) * drug.step_mg

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600

"""Delivering messages to a channel."""

from typing import Callable, Dict, List

from ..errors import DeliveryFailed
from ..models import Message


class Sender:
    """Hands each message to the function registered for its channel."""

    def __init__(self):
        self.channels: Dict[str, Callable[[Message], None]] = {}
        self.failed: List[Message] = []

    def register(self, channel: str, deliver: Callable[[Message], None]) -> None:
        self.channels[channel] = deliver

    def send(self, message: Message) -> bool:
        """True if the channel took the message."""
        deliver = self.channels.get(message.channel)
        if deliver is None:
            self.failed.append(message)
            return False
        try:
            deliver(message)
        except DeliveryFailed:
            self.failed.append(message)
            return False
        return True


def outbox_sender(store, channels=("sms", "email")) -> Sender:
    """A sender that keeps every message in the store's outbox."""
    sender = Sender()
    for channel in channels:
        sender.register(channel, store.outbox.append)
    return sender

def check_single_dose(drug: Drug, dose_mg: float) -> None:
    if dose_mg <= 0:
        raise ValidationFailed(["a dose must be more than zero"])
    if drug.max_single_mg and dose_mg > drug.max_single_mg:
        raise ValidationFailed([f"{drug.name} is limited to {drug.max_single_mg} mg at a time"])


def daily_total(dose_mg: float, doses_per_day: int) -> float:
    return dose_mg * doses_per_day


def course_total(prescription: Prescription) -> float:
    return daily_total(prescription.dose_mg, prescription.doses_per_day) * prescription.days


def tablets_per_dose(drug: Drug, dose_mg: float) -> float:
    if drug.form != "tablet":
        raise ValidationFailed([f"{drug.name} is not a tablet"])
    return dose_mg / drug.strength_mg


def paediatric_dose(drug_code: str, weight_kg: float) -> float:
    """The amount of a drug for one dose, worked out from a child's weight."""
    drug = get_drug(drug_code)
    if weight_kg <= 0:
        raise ValidationFailed(["a weight must be more than zero"])
    amount = min(drug.mg_per_kg * weight_kg, drug.max_single_mg)
    return math.floor(amount / drug.step_mg) * drug.step_mg

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600

BAD_PAIRS: List[FrozenSet[str]] = [
    frozenset(("merixin", "varenol")),

"""Delivering messages to a channel."""

from typing import Callable, Dict, List

from ..errors import DeliveryFailed
from ..models import Message


class Sender:
    """Hands each message to the function registered for its channel."""

    def __init__(self):
        self.channels: Dict[str, Callable[[Message], None]] = {}
        self.failed: List[Message] = []

    def register(self, channel: str, deliver: Callable[[Message], None]) -> None:
        self.channels[channel] = deliver

    def send(self, message: Message) -> bool:
        """True if the channel took the message."""
        deliver = self.channels.get(message.channel)
        if deliver is None:
            self.failed.append(message)
            return False
        try:
            deliver(message)
        except DeliveryFailed:
            self.failed.append(message)
            return False
        return True


def outbox_sender(store, channels=("sms", "email")) -> Sender:
    """A sender that keeps every message in the store's outbox."""
    sender = Sender()
    for channel in channels:
        sender.register(channel, store.outbox.append)
    return sender

"""Where the service gets the time, so tests can control it."""

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600

from typing import FrozenSet, Iterable, List, Optional

BAD_PAIRS: List[FrozenSet[str]] = [
    frozenset(("merixin", "varenol")),
    frozenset(("bezorin", "coldaxine")),
]


def clash(first: str, second: str) -> bool:
    return frozenset((first.lower(), second.lower())) in BAD_PAIRS


def first_clash(ingredient: str, others: Iterable[str]) -> Optional[str]:
    for other in others:
        if clash(ingredient, other):
            return other
    return None

def set_refills(store, prescription_id: str, count: int) -> Prescription:

        raise ValidationFailed([f"refills must be between 0 and {MAX_REFILLS}"])
    prescription = get_prescription(store, prescription_id)
    prescription.refills_left = count

def use_refill(store, prescription_id: str) -> Prescription:
    prescription = get_prescription(store, prescription_id)
    if not prescription.active:
        raise ValidationFailed([f"prescription {prescription_id} has ended"])
    if prescription.refills_left <= 0:
        raise RefillsExhausted(prescription_id)
    prescription.refills_left -= 1
    return save_prescription(store, prescription)

"""Delivering messages to a channel."""

from typing import Callable, Dict, List

from ..errors import DeliveryFailed
from ..models import Message


class Sender:
    """Hands each message to the function registered for its channel."""

    def __init__(self):
        self.channels: Dict[str, Callable[[Message], None]] = {}
        self.failed: List[Message] = []

    def register(self, channel: str, deliver: Callable[[Message], None]) -> None:
        self.channels[channel] = deliver

    def send(self, message: Message) -> bool:
        """True if the channel took the message."""
        deliver = self.channels.get(message.channel)
        if deliver is None:
            self.failed.append(message)
            return False
        try:
            deliver(message)
        except DeliveryFailed:
            self.failed.append(message)
            return False
        return True


def outbox_sender(store, channels=("sms", "email")) -> Sender:
    """A sender that keeps every message in the store's outbox."""
    sender = Sender()
    for channel in channels:
        sender.register(channel, store.outbox.append)
    return sender
