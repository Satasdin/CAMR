"""Turning vital signs into a triage score."""

from dataclasses import dataclass
from typing import List, Tuple

from ..models import Vitals


@dataclass
class TriageScore:
    lines: List[Tuple[str, int]]

    @property
    def total(self) -> int:
        return sum(points for _, points in self.lines)

    def points(self, name: str) -> int:
        return dict(self.lines).get(name, 0)


def temperature_points(temperature_c: float) -> int:
    if temperature_c >= 39.5 or temperature_c < 35.0:
        return 2
    if temperature_c >= 38.0:
        return 1
    return 0


def heart_rate_points(heart_rate: int) -> int:
    if heart_rate >= 130 or heart_rate < 40:
        return 3
    if heart_rate >= 110:
        return 2
    if heart_rate >= 100 or heart_rate < 50:
        return 1
    return 0


def breathing_points(breaths_per_minute: int) -> int:
    if breaths_per_minute >= 30 or breaths_per_minute < 8:
        return 3
    if breaths_per_minute >= 24:
        return 2
    return 0


def pressure_points(systolic: int) -> int:
    if systolic < 90:
        return 3
    if systolic < 100 or systolic >= 200:
        return 2
    return 0


def score_vitals(vitals: Vitals) -> TriageScore:
    """Every vital sign that scores, as (name, points) pairs."""
    candidates = [
        ("temperature", temperature_points(vitals.temperature_c)),
        ("heart_rate", heart_rate_points(vitals.heart_rate)),
        ("breathing", breathing_points(vitals.breaths_per_minute)),
        ("pressure", pressure_points(vitals.systolic)),
    ]
    return TriageScore([(name, points) for name, points in candidates if points])

"""The queue of walk-in patients waiting to be seen."""

from typing import List, Optional

from ..config import SECONDS_PER_MINUTE
from ..errors import InvalidTransition
from ..models import TRIAGE_SEEN, TRIAGE_WAITING, TriageCase, Vitals
from ..store.repositories import get_patient, get_triage_case, save_triage_case
from .levels import LEVEL_ORDER, level_for, target_minutes
from .scoring import score_vitals


def admit(store, patient_id: str, vitals: Vitals, clock) -> TriageCase:
    get_patient(store, patient_id)
    total = score_vitals(vitals).total
    case = TriageCase(id=store.next_id("TRG"), patient_id=patient_id, vitals=vitals, score=total,
                      level=level_for(total), arrived_at=clock.now())
    return save_triage_case(store, case)


def waiting(store) -> List[TriageCase]:
    found = [c for c in store.triage_cases.values() if c.status == TRIAGE_WAITING]
    return sorted(found, key=lambda c: (LEVEL_ORDER[c.level], c.arrived_at))


def next_case(store) -> Optional[TriageCase]:
    queue = waiting(store)
    return queue[0] if queue else None


def mark_seen(store, case_id: str, clock) -> TriageCase:
    case = get_triage_case(store, case_id)
    if case.status != TRIAGE_WAITING:
        raise InvalidTransition(case.status, TRIAGE_SEEN)
    case.status = TRIAGE_SEEN
    case.seen_at = clock.now()
    return save_triage_case(store, case)


def overdue(store, now: int) -> List[TriageCase]:
    return [c for c in waiting(store) if now - c.arrived_at > target_minutes(c.level) * SECONDS_PER_MINUTE]

class IdGenerator:
    """Hands out identifiers such as PAT-00001, one sequence per prefix."""

    def __init__(self):

    def next(self, prefix: str) -> str:

from typing import FrozenSet, Iterable, List, Optional

BAD_PAIRS: List[FrozenSet[str]] = [
    frozenset(("merixin", "varenol")),
    frozenset(("bezorin", "coldaxine")),
]


def clash(first: str, second: str) -> bool:
    return frozenset((first.lower(), second.lower())) in BAD_PAIRS


def first_clash(ingredient: str, others: Iterable[str]) -> Optional[str]:
    for other in others:
        if clash(ingredient, other):
            return other
    return None
