20:     sign = "-" if amount_cents < 0 else ""

22:     return f"{sign}{currency} {whole:,}.{part:02d}"

25: def parse_amount(text: str) -> int:
26:     """Reads an amount typed by staff, such as '1,250.50', into cents."""
27:     cleaned = text.replace(",", "").strip()
28:     if not cleaned:
29:         raise ValueError("empty amount")
30:     negative = cleaned.startswith("-")
31:     if negative:
32:         cleaned = cleaned[1:]
33:     whole, _, part = cleaned.partition(".")
34:     if not whole.isdigit() and whole != "":
35:         raise ValueError(f"not an amount: {text}")
36:     if part and (not part.isdigit() or len(part) > 2):
37:         raise ValueError(f"not an amount: {text}")
38:     cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
39:     return -cents if negative else cents
40:

43:     """Splits an amount so the parts always add back to the amount."""
44:     if parts <= 0:

46:     share, remainder = divmod(amount_cents, parts)

        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Where the service gets the time, so tests can control it."""

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600

"""Deciding who is reminded about an appointment, and how."""

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject, sms_body


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    if patient.wants_sms and patient.phone:
        out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, doctor, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """A percentage of an amount, half a cent going up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Where the service gets the time, so tests can control it."""

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600

"""Deciding who is reminded about an appointment, and how."""

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject, sms_body


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    if patient.wants_sms and patient.phone:
        out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, doctor, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """A percentage of an amount, half a cent going up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Where the service gets the time, so tests can control it."""

import time
from datetime import datetime, timezone

LOCAL_OFFSET_HOURS = 3


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def _local(moment: int) -> datetime:
    return datetime.fromtimestamp(moment + LOCAL_OFFSET_HOURS * 3600, tz=timezone.utc)


def hour_of(moment: int) -> int:
    """The hour of the day at a moment, in the clinic's local time."""
    return _local(moment).hour


def day_of(moment: int) -> str:
    return _local(moment).strftime("%Y-%m-%d")


def time_of(moment: int) -> str:
    return _local(moment).strftime("%H:%M")


def weekday_of(moment: int) -> int:
    """Monday is 0 and Sunday is 6."""
    return _local(moment).weekday()


def moment_at(day: str, hour: int, minute: int = 0) -> int:
    """The moment of a local date and time, such as '2024-03-05' at 10:40."""
    local = datetime.strptime(day, "%Y-%m-%d").replace(hour=hour, minute=minute, tzinfo=timezone.utc)
    return int(local.timestamp()) - LOCAL_OFFSET_HOURS * 3600

"""Deciding who is reminded about an appointment, and how."""

from typing import List

from ..config import REMINDER_WINDOW_HOURS, SECONDS_PER_HOUR
from ..models import APPOINTMENT_BOOKED, Appointment, Doctor, Message, Patient
from ..store.repositories import get_doctor, get_patient, save_appointment
from .templates import email_body, email_subject, sms_body


def due_for_reminder(appointment: Appointment, now: int) -> bool:
    if appointment.status != APPOINTMENT_BOOKED or appointment.reminded:
        return False
    return now < appointment.start_at <= now + REMINDER_WINDOW_HOURS * SECONDS_PER_HOUR


def messages_for(patient: Patient, doctor: Doctor, appointment: Appointment, now: int) -> List[Message]:
    out = []
    if patient.wants_email and patient.email:
        out.append(Message(channel="email", to=patient.email, subject=email_subject(appointment),
                           body=email_body(patient, doctor, appointment), sent_at=now))
    if patient.wants_sms and patient.phone:
        out.append(Message(channel="sms", to=patient.phone, body=sms_body(patient, doctor, appointment), sent_at=now))
    return out


def send_reminders(store, sender, clock) -> int:
    """Sends reminders for every appointment coming up and returns how many messages went out."""
    now = clock.now()
    count = 0
    for appointment in list(store.appointments.values()):
        if not due_for_reminder(appointment, now):
            continue
        patient = get_patient(store, appointment.patient_id)
        doctor = get_doctor(store, appointment.doctor_id)
        count += sum(1 for message in messages_for(patient, doctor, appointment, now) if sender.send(message))
        appointment.reminded = True
        save_appointment(store, appointment)
    return count
