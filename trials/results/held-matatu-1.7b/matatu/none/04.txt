"""Applying the provider's statement to the tickets it pays for."""

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:
            _mark_paid(store, ticket, line)
            result.matched += 1
    return result

"""Selling a ticket for a seat on a trip."""

from ..config import DEFAULT_SETTINGS, MAX_TICKETS_PER_PHONE, RIDER_ADULT, Settings
from ..errors import NoSeatsLeft, TooManyTickets, TripNotOpen
from ..fares.quote import quote_fare
from ..models import METHOD_CASH, METHOD_MOBILE, TICKET_CANCELLED, TICKET_PAID, TRIP_SCHEDULED, Ticket
from ..routes.network import stages_between
from ..store.repositories import get_route, get_trip, get_vehicle, save_ticket, tickets_on_trip


def _live(tickets) -> list:
    return [ticket for ticket in tickets if ticket.status != TICKET_CANCELLED]


def issue_ticket(store, trip_id: str, from_stage: str, to_stage: str, phone: str, clock, method: str = METHOD_MOBILE,
                 rider: str = RIDER_ADULT, bags: int = 0, raining: bool = False,
                 settings: Settings = DEFAULT_SETTINGS) -> Ticket:
    """Sells one seat. Cash tickets are paid at once; mobile tickets wait for the statement."""
    trip = get_trip(store, trip_id)
    if trip.status != TRIP_SCHEDULED:
        raise TripNotOpen(trip_id, trip.status)
    sold = _live(tickets_on_trip(store, trip_id))
    if len(sold) >= get_vehicle(store, trip.vehicle_id).seats:
        raise NoSeatsLeft(trip_id)
    if sum(1 for ticket in sold if ticket.phone == phone) >= MAX_TICKETS_PER_PHONE:
        raise TooManyTickets(phone, MAX_TICKETS_PER_PHONE)
    route = get_route(store, trip.route_id)
    now = clock.now()
    quote = quote_fare(stages_between(route, from_stage, to_stage), now, rider, bags, raining, settings)
    ticket = Ticket(id=store.next_id("TKT"), trip_id=trip_id, route_id=route.id, vehicle_id=trip.vehicle_id,
                    from_stage=from_stage, to_stage=to_stage, phone=phone, fare_kes=quote.total_kes, method=method,
                    issued_at=now)
    if method == METHOD_CASH:
        ticket.status = TICKET_PAID
        ticket.paid_at = now
    return save_ticket(store, ticket)


def cancel_ticket(store, ticket: Ticket) -> Ticket:
    ticket.status = TICKET_CANCELLED
    return save_ticket(store, ticket)

"""Giving money back for tickets that were not used."""

from ..config import REFUND_WINDOW_SECONDS
from ..errors import RefundRefused
from ..models import METHOD_MOBILE, TICKET_PAID, TICKET_REFUNDED, Refund, Ticket
from ..store.repositories import get_ticket, save_refund, save_ticket


def refund_ticket(store, ticket_id: str, provider, clock, reason: str = "") -> Ticket:
    ticket = get_ticket(store, ticket_id)
    if ticket.status != TICKET_PAID:
        raise RefundRefused(ticket_id, f"status is {ticket.status}")
    if ticket.boarded_at is not None:
        raise RefundRefused(ticket_id, "already boarded")
    now = clock.now()
    if now - (ticket.paid_at or 0) > REFUND_WINDOW_SECONDS:
        raise RefundRefused(ticket_id, "too late")
    if ticket.method == METHOD_MOBILE:
        provider.send_back(ticket.phone, ticket.fare_kes, ticket.id)
    ticket.status = TICKET_REFUNDED
    save_refund(store, Refund(id=store.next_id("RFD"), ticket_id=ticket_id, amount_kes=ticket.fare_kes,
                              reason=reason, created_at=now))
    return save_ticket(store, ticket)

from ..errors import ProviderTimeout


class MoneyProvider:
    """What the back office needs from a mobile-money provider."""

    def send_back(self, phone: str, amount_kes: int, reference: str) -> str:
        """Returns the provider's transaction code. Raises ProviderTimeout."""
        raise NotImplementedError


class SandboxProvider(MoneyProvider):
    """Keeps transfers in memory."""

    def __init__(self):
        self.sent: List[tuple] = []
        self.fail_next: Optional[Exception] = None

    def send_back(self, phone: str, amount_kes: int, reference: str) -> str:
        if self.fail_next is not None:
            problem, self.fail_next = self.fail_next, None
            raise problem
        self.sent.append((phone, amount_kes, reference))
        return f"SBX{len(self.sent):06d}"

    def simulate_timeout(self) -> None:
        self.fail_next = ProviderTimeout("provider did not answer")

"""Reading the statement file the provider sends each day."""

from typing import List

from ..errors import BadStatement
from ..models import StatementLine
from ..money import parse_amount


def parse_statement(text: str) -> List[StatementLine]:
    """Each line is: code, phone, amount, account reference, seconds since the epoch."""
    lines = []
    for number, raw in enumerate(text.splitlines(), start=1):
        if not raw.strip() or raw.lstrip().startswith("#"):
            continue
        parts = [part.strip() for part in raw.split(",")]
        if len(parts) != 5:
            raise BadStatement(number, "expected five fields")
        code, phone, amount, reference, at = parts
        try:
            lines.append(StatementLine(code=code, phone=phone, amount_kes=parse_amount(amount), account_ref=reference,
                                       at=int(at)))
        except ValueError as problem:
            raise BadStatement(number, str(problem)) from None
    return lines

"""Handing messages to a channel."""

from ..errors import DeliveryFailed
from ..models import Message


class Channel:
    name = "channel"

    def send(self, message: Message) -> None:
        raise NotImplementedError


class OutboxChannel(Channel):
    """Keeps messages in the store's outbox, where a worker or a test can read them."""

    def __init__(self, store, name: str):
        self.store = store
        self.name = name

    def send(self, message: Message) -> None:
        if not message.to:
            raise DeliveryFailed(f"no address for {self.name}")
        self.store.outbox.append(message)


class Sender:
    """Sends through named channels."""

    def __init__(self, channels):
        self.channels = {channel.name: channel for channel in channels}

    def send(self, message: Message) -> bool:
        """Returns False if there is no such channel or the channel refused the message."""
        channel = self.channels.get(message.channel)
        if channel is None:
            return False
        try:
            channel.send(message)
        except DeliveryFailed:
            return False
        return True


def outbox_sender(store) -> Sender:
    return Sender([OutboxChannel(store, "sms")])
