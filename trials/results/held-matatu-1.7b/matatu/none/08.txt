@dataclass
class Context:
    store: object
    provider: object
    clock: object
    sender: Sender


def handled(function):
    """Turns expected errors into a response the web layer can send."""

    @wraps(function)
    def wrapper(request, context):
        try:
            return function(request, context)
        except MatatuError as problem:
            return {"error": problem.code, "message": str(problem)}

    return wrapper


@handled
def quote(request, context):
    require(request, "route_id", "from_stage", "to_stage")
    route = get_route(context.store, request["route_id"])
    stages = stages_between(route, request["from_stage"], request["to_stage"])
    result = quote_fare(stages, context.clock.now(), request.get("rider", "adult"), int(request.get("bags", 0)),
                        bool(request.get("raining")))
    return {"total_kes": result.total_kes, "band": result.band, "peak": result.peak, "lines": result.lines}


@handled
def sell_ticket(request, context):
    check_sale(request)
    ticket = issue_ticket(context.store, request["trip_id"], request["from_stage"], request["to_stage"],
                          request["phone"], context.clock, request.get("method", "mobile"),
                          request.get("rider", "adult"), int(request.get("bags", 0)), bool(request.get("raining")))
    sent = context.sender.send(Message(channel="sms", to=ticket.phone, body=ticket_text(ticket),
                                       sent_at=context.clock.now()))
    return {"ticket_id": ticket.id, "fare_kes": ticket.fare_kes, "status": ticket.status, "message_sent": sent}


@handled
def board(request, context):
    require(request, "ticket_id", "trip_id")
    ticket = board_ticket(context.store, request["ticket_id"], request["trip_id"], context.clock)
    return {"ticket_id": ticket.id, "boarded_at": ticket.boarded_at}


@handled
def ticket_status(request, context):
    require(request, "ticket_id")
    ticket = get_ticket(context.store, request["ticket_id"])
    return {"ticket_id": ticket.id, "status": ticket.status, "fare_kes": ticket.fare_kes}


@handled

"""Checking requests before they reach the back office."""

from typing import Dict, List

from ..config import RIDER_TYPES
from ..errors import ValidationFailed

REQUIRED_SALE_FIELDS = ("trip_id", "from_stage", "to_stage", "phone")


def missing_fields(request: Dict, fields) -> List[str]:
    return [f"{name} is required" for name in fields if request.get(name) in (None, "")]


def phone_problems(phone: str) -> List[str]:
    if not phone.startswith("+254") or not phone[1:].isdigit() or len(phone) != 13:
        return ["phone must look like +254700000000"]
    return []


def sale_problems(request: Dict) -> List[str]:
    problems = missing_fields(request, REQUIRED_SALE_FIELDS)
    if problems:
        return problems
    problems.extend(phone_problems(request["phone"]))
    if request.get("rider", "adult") not in RIDER_TYPES:
        problems.append("rider is not one we know")
    bags = request.get("bags", 0)
    if not isinstance(bags, int) or isinstance(bags, bool) or bags < 0:
        problems.append("bags must be a whole number")
    return problems


def require(request: Dict, *fields) -> None:
    problems = missing_fields(request, fields)
    if problems:
        raise ValidationFailed(problems)


def check_sale(request: Dict) -> None:
    problems = sale_problems(request)
    if problems:
        raise ValidationFailed(problems)

import calendar
import time
from datetime import datetime, timezone

from .config import LOCAL_OFFSET_HOURS, SECONDS_PER_HOUR


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def hour_of(moment: int, offset_hours: int = LOCAL_OFFSET_HOURS) -> int:
    """The hour of the day at a moment, in local time."""
    return datetime.fromtimestamp(moment + offset_hours * SECONDS_PER_HOUR, tz=timezone.utc).hour


def weekday_of(moment: int, offset_hours: int = LOCAL_OFFSET_HOURS) -> int:
    """Monday is 0."""
    return datetime.fromtimestamp(moment + offset_hours * SECONDS_PER_HOUR, tz=timezone.utc).weekday()


def day_of(moment: int, offset_hours: int = LOCAL_OFFSET_HOURS) -> str:
    return datetime.fromtimestamp(moment + offset_hours * SECONDS_PER_HOUR, tz=timezone.utc).strftime("%Y-%m-%d")


OPERATING_DAY_START_HOUR = 3


def operating_day(moment: int, offset_hours: int = LOCAL_OFFSET_HOURS) -> str:
    """The operating day a moment belongs to, written YYYY-MM-DD."""
    return day_of(moment - OPERATING_DAY_START_HOUR * SECONDS_PER_HOUR, offset_hours)


def moment_of(day: str, hour: int, minute: int = 0, offset_hours: int = LOCAL_OFFSET_HOURS) -> int:
    """The moment of a local time on a day written YYYY-MM-DD."""
    parsed = datetime.strptime(day, "%Y-%m-%d")
    return calendar.timegm(parsed.timetuple()) + (hour - offset_hours) * SECONDS_PER_HOUR + minute * 60

from ..config import COMMISSION_EXCLUDED_ROUTES, CONDUCTOR_COMMISSION_PERCENT, DRIVER_COMMISSION_PERCENT, ROLE_DRIVER
from ..models import TICKET_PAID, CrewMember, Shift, Ticket

def rate_for(member: CrewMember) -> int:
    return DRIVER_COMMISSION_PERCENT if member.role == ROLE_DRIVER else CONDUCTOR_COMMISSION_PERCENT


def commission_on(ticket: Ticket, member: CrewMember) -> int:
    if ticket.route_id in COMMISSION_EXCLUDED_ROUTES:
        return 0
    return percent_of(ticket.fare_kes, rate_for(member))


def _in_shift(ticket: Ticket, shift: Shift) -> bool:
    if ticket.vehicle_id != shift.vehicle_id or ticket.status != TICKET_PAID:
        return False
    return ticket.issued_at >= shift.started_at and (shift.ended_at is None or ticket.issued_at <= shift.ended_at)


def commission_for_shift(store, shift: Shift) -> int:
    member = get_crew_member(store, shift.crew_id)
    return sum(commission_on(ticket, member) for ticket in store.tickets.values() if _in_shift(ticket, shift))

"""Identifiers people can read out over the phone."""

ALPHABET = "23456789ABCDEFGHJKLMNPQRSTUVWXYZ"


def encode(number: int, width: int = 6) -> str:
    if number < 0:
        raise ValueError("number must not be negative")
    digits = []
    while number:
        number, remainder = divmod(number, len(ALPHABET))
        digits.append(ALPHABET[remainder])
    text = "".join(reversed(digits)) or ALPHABET[0]
    return text.rjust(width, ALPHABET[0])


class IdGenerator:
    """Hands out identifiers in order, with a prefix per record type."""

    def __init__(self):
        self.counters = {}

    def next(self, prefix: str) -> str:
        value = self.counters.get(prefix, 0) + 1
        self.counters[prefix] = value
        return f"{prefix}-{encode(value)}"

ALPHABET = "23456789ABCDEFGHJKLMNPQRSTUVWXYZ"


def encode(number: int, width: int = 6) -> str:
    if number < 0:
        raise ValueError("number must not be negative")
    digits = []
    while number:
        number, remainder = divmod(number, len(ALPHABET))
        digits.append(ALPHABET[remainder])
    text = "".join(reversed(digits)) or ALPHABET[0]
    return text.rjust(width, ALPHABET[0])


class IdGenerator:
    """Hands out identifiers in order, with a prefix per record type."""

    def __init__(self):
        self.counters = {}

    def next(self, prefix: str) -> str:
        value = self.counters.get(prefix, 0) + 1
        self.counters[prefix] = value
        return f"{prefix}-{encode(value)}"

ALPHABET = "23456789ABCDEFGHJKLMNPQRSTUVWXYZ"


def encode(number: int, width: int = 6) -> str:
    if number < 0:
        raise ValueError("number must not be negative")
    digits = []
    while number:
        number, remainder = divmod(number, len(ALPHABET))
        digits.append(ALPHABET[remainder])
    text = "".join(reversed(digits)) or ALPHABET[0]
    return text.rjust(width, ALPHABET[0])


class IdGenerator:
    """Hands out identifiers in order, with a prefix per record type."""

    def __init__(self):
        self.counters = {}

    def next(self, prefix: str) -> str:
        value = self.counters.get(prefix, 0) + 1
        self.counters[prefix] = value
        return f"{prefix}-{encode(value)}"
