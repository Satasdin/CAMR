"""Applying the provider's statement to the tickets it pays for."""

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:

from typing import List

from ..errors import BadStatement
from ..models import StatementLine
from ..money import parse_amount


def parse_statement(text: str) -> List[StatementLine]:
    """Each line is: code, phone, amount, account reference, seconds since the epoch."""
    lines = []
    for number, raw in enumerate(text.splitlines(), start=1):
        if not raw.strip() or raw.lstrip().startswith("#"):
            continue
        parts = [part.strip() for part in raw.split(",")]
        if len(parts) != 5:
            raise BadStatement(number, "expected five fields")
        code, phone, amount, reference, at = parts
        try:
            lines.append(StatementLine(code=code, phone=phone, amount_kes=parse_amount(amount), account_ref=reference,
                                       at=int(at)))
        except ValueError as problem:
            raise BadStatement(number, str(problem)) from None
    return lines

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:
            _mark_paid(store, ticket, line)
            result.matched += 1
    return result

"""Reading the statement file the provider sends each day."""

from typing import List

from ..errors import BadStatement
from ..models import StatementLine
from ..money import parse_amount


def parse_statement(text: str) -> List[StatementLine]:
    """Each line is: code, phone, amount, account reference, seconds since the epoch."""
    lines = []
    for number, raw in enumerate(text.splitlines(), start=1):
        if not raw.strip() or raw.lstrip().startswith("#"):
            continue
        parts = [part.strip() for part in raw.split(",")]
        if len(parts) != 5:
            raise BadStatement(number, "expected five fields")
        code, phone, amount, reference, at = parts
        try:
            lines.append(StatementLine(code=code, phone=phone, amount_kes=parse_amount(amount), account_ref=reference,
                                       at=int(at)))
        except ValueError as problem:
            raise BadStatement(number, str(problem)) from None
    return lines

@dataclass
class DailyFigures:
    day: str
    tickets_paid: int
    cash_kes: int
    mobile_kes: int
    refunded_kes: int
    methods: Dict[str, int]

    def net_kes(self) -> int:
        return self.cash_kes + self.mobile_kes - self.refunded_kes


def figures_for(store, day: str) -> DailyFigures:
    paid = [t for t in store.tickets.values()
            if t.status in (TICKET_PAID, TICKET_REFUNDED) and t.paid_at is not None and day_of(t.paid_at) == day]
    counted = Counter(t.method for t in paid)
    return DailyFigures(day=day, tickets_paid=len(paid),
                        cash_kes=sum(t.fare_kes for t in paid if t.method == METHOD_CASH),
                        mobile_kes=sum(t.fare_kes for t in paid if t.method == METHOD_MOBILE),
                        refunded_kes=sum(t.fare_kes for t in paid if t.status == TICKET_REFUNDED),
                        methods=dict(counted))


def render_daily(figures: DailyFigures) -> str:
    return "\n".join([
        f"Daily figures for {figures.day}",
        f"Tickets paid: {figures.tickets_paid}",
        f"Cash: {format_kes(figures.cash_kes)}",
        f"Mobile money: {format_kes(figures.mobile_kes)}",
        f"Refunded: {format_kes(figures.refunded_kes)}",
        f"Net: {format_kes(figures.net_kes())}",

"""Applying the provider's statement to the tickets it pays for."""

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:
            _mark_paid(store, ticket, line)
            result.matched += 1
    return result

def parse_statement(text: str) -> List[StatementLine]:
    """Each line is: code, phone, amount, account reference, seconds since the epoch."""
    lines = []
    for number, raw in enumerate(text.splitlines(), start=1):
        if not raw.strip() or raw.lstrip().startswith("#"):
            continue
        parts = [part.strip() for part in raw.split(",")]
        if len(parts) != 5:
            raise BadStatement(number, "expected five fields")
        code, phone, amount, reference, at = parts
        try:
            lines.append(StatementLine(code=code, phone=phone, amount_kes=parse_amount(amount), account_ref=reference,
                                       at=int(at)))
        except ValueError as problem:
            raise BadStatement(number, str(problem)) from None
    return lines

from collections import Counter
from dataclasses import dataclass
from typing import Dict

from ..clock import day_of
from ..models import METHOD_CASH, METHOD_MOBILE, TICKET_PAID, TICKET_REFUNDED
from ..money import format_kes


@dataclass
class DailyFigures:
    day: str
    tickets_paid: int
    cash_kes: int
    mobile_kes: int
    refunded_kes: int
    methods: Dict[str, int]

    def net_kes(self) -> int:
        return self.cash_kes + self.mobile_kes - self.refunded_kes


def figures_for(store, day: str) -> DailyFigures:
    paid = [t for t in store.tickets.values()
            if t.status in (TICKET_PAID, TICKET_REFUNDED) and t.paid_at is not None and day_of(t.paid_at) == day]
    counted = Counter(t.method for t in paid)
    return DailyFigures(day=day, tickets_paid=len(paid),
                        cash_kes=sum(t.fare_kes for t in paid if t.method == METHOD_CASH),
                        mobile_kes=sum(t.fare_kes for t in paid if t.method == METHOD_MOBILE),
                        refunded_kes=sum(t.fare_kes for t in paid if t.status == TICKET_REFUNDED),
                        methods=dict(counted))


def render_daily(figures: DailyFigures) -> str:
    return "\n".join([
        f"Daily figures for {figures.day}",
        f"Tickets paid: {figures.tickets_paid}",
        f"Cash: {format_kes(figures.cash_kes)}",
        f"Mobile money: {format_kes(figures.mobile_kes)}",
        f"Refunded: {format_kes(figures.refunded_kes)}",
        f"Net: {format_kes(figures.net_kes())}",

The task is to modify the reconciliation process so that statement lines with matching phone numbers and amounts are correctly matched to tickets, reducing the number of lines stuck in suspense with the reason "unknown reference".

The current implementation in `reconcile.py` handles matching by checking if a ticket exists for a given line, but it does not account for the account reference, which is often different from the ticket's reference. This is why some lines are added to suspense with the reason "unknown reference".

To fix this, we need to ensure that lines with matching phone numbers and amounts are matched to tickets, even if the account reference is different. This can be done by modifying the reconciliation logic to prioritize matching on phone number and amount, and then handle the account reference as a secondary check.

"""
Applying the provider's statement to the tickets it pays for."""
"""

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:
            _mark_paid(store, ticket, line)
            result.matched += 1
    return result
