17: 
18: 
19: def round_up_to(amount_kes: int, step: int) -> int:
20:     if step <= 0:
21:         raise ValueError("step must be positive")
22:     return -(-amount_kes // step) * step
23: 
24: 
25: def format_kes(amount_kes: int, currency: str = CURRENCY) -> str:
26:     sign = "-" if amount_kes < 0 else ""
27:     return f"{sign}{currency} {abs(amount_kes):,}"
28: 
29: 
30: def parse_amount(text: str) -> int:
31:     """Reads an amount from a statement or typed by staff, such as '1,250' or '1,250.00'."""
32:     cleaned = text.replace(",", "").strip()
33:     if not cleaned:
34:         raise ValueError("empty amount")
35:     whole, _, part = cleaned.partition(".")
36:     if not whole.isdigit():
37:         raise ValueError(f"not an amount: {text}")
38:     if part.strip("0"):
39:         raise ValueError(f"not a whole number of shillings: {text}")
40:     return int(whole)
41: 
42: 
43: def split_evenly(amount_kes: int, parts: int) -> list:
44:     """Splits an amount so the parts always add back to the amount."""
45:     if parts <= 0:
46:         raise ValueError("parts must be positive")
47:     share, remainder = divmod(amount_kes, parts)
48:     return [share + 1 if index < remainder else share for index in range(parts)]

8: def percent_of(amount_kes: int, percent: int) -> int:
9:     """Rounds half up."""
10:     if amount_kes < 0:
11:         raise ValueError("amount must not be negative")
12:     return (amount_kes * percent + 50) // 100
13: 
14: 
15: def add(amounts: Iterable[int]) -> int:
16:     return sum(amounts)
17: 
18: 
19: def round_up_to(amount_kes: int, step: int) -> int:
20:     if step <= 0:
21:         raise ValueError("step must be positive")
22:     return -(-amount_kes // step) * step
23: 
24: 
25: def format_kes(amount_kes: int, currency: str = CURRENCY) -> str:
26:     sign = "-" if amount_kes < 0 else ""
27:     return f"{sign}{currency} {abs(amount_kes):,}"
28: 
29: 
30: def parse_amount(text: str) -> int:
31:     """Reads an amount from a statement or typed by staff, such as '1,250' or '1,250.00'."""
32:     cleaned = text.replace(",", "").strip()
33:     if not cleaned:
34:         raise ValueError("empty amount")
35:     whole, _, part = cleaned.partition(".")
36:     if not whole.isdigit():
37:         raise ValueError(f"not an amount: {text}")
38:     if part.strip("0"):
39:         raise ValueError(f"not a whole number of shillings: {text}")
40:     return int(whole)
41: 
42: 
43: def split_evenly(amount_kes: int, parts: int) -> list:
44:     """Splits an amount so the parts always add back to the amount."""
45:     if parts <= 0:
46:         raise ValueError("parts must be positive")
47:     share, remainder = divmod(amount_kes, parts)
48:     return [share + 1 if index < remainder else share for index in range(parts)]

# Owner payouts

Vehicle owners are paid per date. Add `owner_payout(store, owner_id, date)` to `matatu/fleet/owners.py`, where `date` is a string like `2023-11-15`. It returns a dictionary with these keys:

- `owner_id` and `date`: as given,

"""The figures sent to finance each morning."""

from collections import Counter
from dataclasses import dataclass
from typing import Dict

from ..clock import day_of
from ..models import METHOD_CASH, METHOD_MOBILE, TICKET_PAID, TICKET_REFUNDED
from ..money import format_kes


@dataclass
class DailyFigures:
    day: str
    tickets_paid: int
    cash_kes: int
    mobile_kes: int
    refunded_kes: int
    methods: Dict[str, int]

    def net_kes(self) -> int:
        return self.cash_kes + self.mobile_kes - self.refunded_kes


def figures_for(store, day: str) -> DailyFigures:
    paid = [t for t in store.tickets.values()
            if t.status in (TICKET_PAID, TICKET_REFUNDED) and t.paid_at is not None and day_of(t.paid_at) == day]
    counted = Counter(t.method for t in paid)
    return DailyFigures(day=day, tickets_paid=len(paid),
                        cash_kes=sum(t.fare_kes for t in paid if t.method == METHOD_CASH),
                        mobile_kes=sum(t.fare_kes for t in paid if t.method == METHOD_MOBILE),
                        refunded_kes=sum(t.fare_kes for t in paid if t.status == TICKET_REFUNDED),
                        methods=dict(counted))


def render_daily(figures: DailyFigures) -> str:
    return "\n".join([
        f"Daily figures for {figures.day}",
        f"Tickets paid: {figures.tickets_paid}",
        f"Cash: {format_kes(figures.cash_kes)}",
        f"Mobile money: {format_kes(figures.mobile_kes)}",
        f"Refunded: {format_kes(figures.refunded_kes)}",
        f"Net: {format_kes(figures.net_kes())}",
    ])

"""Helpers for working with whole shillings."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_kes: int, percent: int) -> int:
    """Rounds half up."""
    if amount_kes < 0:
        raise ValueError("amount must not be negative")
    return (amount_kes * percent + 50) // 100


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def round_up_to(amount_kes: int, step: int) -> int:
    if step <= 0:
        raise ValueError("step must be positive")
    return -(-amount_kes // step) * step


def format_kes(amount_kes: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_kes < 0 else ""
    return f"{sign}{currency} {abs(amount_kes):,}"


def parse_amount(text: str) -> int:
    """Reads an amount from a statement or typed by staff, such as '1,250' or '1,250.00'."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit():
        raise ValueError(f"not an amount: {text}")
    if part.strip("0"):
        raise ValueError(f"not a whole number of shillings: {text}")
    return int(whole)


def split_evenly(amount_kes: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_kes, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Where the back office gets the time, so tests can control it."""

import calendar
import time
from datetime import datetime, timezone

from .config import LOCAL_OFFSET_HOURS, SECONDS_PER_HOUR


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def hour_of(moment: int, offset_hours: int = LOCAL_OFFSET_HOURS) -> int:
    """The hour of the day at a moment, in local time."""
    return datetime.fromtimestamp(moment + offset_hours * SECONDS_PER_HOUR, tz=timezone.utc).hour


def weekday_of(moment: int, offset_hours: int = LOCAL_OFFSET_HOURS) -> int:
    """Monday is 0."""
    return datetime.fromtimestamp(moment + offset_hours * SECONDS_PER_HOUR, tz=timezone.utc).weekday()


def day_of(moment: int, offset_hours: int = LOCAL_OFFSET_HOURS) -> str:
    return datetime.fromtimestamp(moment + offset_hours * SECONDS_PER_HOUR, tz=timezone.utc).strftime("%Y-%m-%d")


OPERATING_DAY_START_HOUR = 3


def operating_day(moment: int, offset_hours: int = LOCAL_OFFSET_HOURS) -> str:
    """The operating day a moment belongs to, written YYYY-MM-DD."""
    return day_of(moment - OPERATING_DAY_START_HOUR * SECONDS_PER_HOUR, offset_hours)


def moment_of(day: str, hour: int, minute: int = 0, offset_hours: int = LOCAL_OFFSET_HOURS) -> int:
    """The moment of a local time on a day written YYYY-MM-DD."""
    parsed = datetime.strptime(day, "%Y-%m-%d")
    return calendar.timegm(parsed.timetuple()) + (hour - offset_hours) * SECONDS_PER_HOUR + minute * 60

from typing import Dict, List, Tuple

from ..clock import day_of
from ..models import TICKET_PAID


def fares_by_route(store, day: str) -> Dict[str, int]:
    totals: Dict[str, int] = {}
    for ticket in store.tickets.values():
        if ticket.status == TICKET_PAID and day_of(ticket.issued_at) == day:
            totals[ticket.route_id] = totals.get(ticket.route_id, 0) + ticket.fare_kes
    return totals


def league_table(store, day: str) -> List[Tuple[str, int]]:
    """Routes from highest fares to lowest; ties in route order."""
    return sorted(fares_by_route(store, day).items(), key=lambda pair: (-pair[1], pair[0]))

        return 0
    return percent_of(ticket.fare_kes, rate_for(member))


def _in_shift(ticket: Ticket, shift: Shift) -> bool:
    if ticket.vehicle_id != shift.vehicle_id or ticket.status != TICKET_PAID:
        return False
    return ticket.issued_at >= shift.started_at and (shift.ended_at is None or ticket.issued_at <= shift.ended_at)


def commission_for_shift(store, shift: Shift) -> int:
    member = get_crew_member(store, shift.crew_id)
    return sum(commission_on(ticket, member) for ticket in store.tickets.values() if _in_shift(ticket, shift))
