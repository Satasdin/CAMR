3: from typing import List
4: 
5: from ..errors import BadStatement
6: from ..models import StatementLine
7: from ..money import parse_amount
8: 
9: 
10: def parse_statement(text: str) -> List[StatementLine]:
11:     """Each line is: code, phone, amount, account reference, seconds since the epoch."""
12:     lines = []
13:     for number, raw in enumerate(text.splitlines(), start=1):
14:         if not raw.strip() or raw.lstrip().startswith("#"):
15:             continue
16:         parts = [part.strip() for part in raw.split(",")]
17:         if len(parts) != 5:
18:             raise BadStatement(number, "expected five fields")
19:         code, phone, amount, reference, at = parts
20:         try:
21:             lines.append(StatementLine(code=code, phone=phone, amount_kes=parse_amount(amount), account_ref=reference,
22:                                        at=int(at)))
23:         except ValueError as problem:
24:             raise BadStatement(number, str(problem)) from None
25:     return lines
</excerpt>

<excerpt file="README.md" kind="code">
1: # Matatu
2: 
3: The back office of a Nairobi minibus operator. It keeps the routes and stages, works
4: out fares, sells tickets, tracks vehicles and crews, matches the mobile-money statement
5: to tickets and prepares the daily figures.
6: 
7: ## Layout
8: 
9: - `matatu/routes`: stages and the order they come in along a route.
10: - `matatu/fares`: distance bands, peak hours, extras and rider discounts, combined into a quote.
11: - `matatu/tickets`: selling tickets, boarding, refunds and day passes.
12: - `matatu/fleet`: vehicles, inspections, fuel and vehicle owners.
13: - `matatu/crews`: drivers and conductors, shifts and commission.
14: - `matatu/payments`: the mobile-money statement, matching lines to tickets, suspense and the provider.
15: - `matatu/reports`: figures for finance and the routes table.
16: - `matatu/notify`: message templates and delivery channels.
17: - `matatu/api`: request validation and the handlers the web layer calls.
18: - `matatu/store`: an in-memory store with small repository helpers.
19: 
20: Money is held as whole Kenya shillings. Times are whole seconds since the epoch, taken from
21: `matatu.clock` so tests can control them.
22: 
23: ## Running the tests
24: 
25:     python3 -m unittest discover -s tests
</excerpt>

13: 
14: 
15: def issue_ticket(store, trip_id: str, from_stage: str, to_stage: str, phone: str, clock, method: str = METHOD_MOBILE,
16:                  rider: str = RIDER_ADULT, bags: int = 0, raining: bool = False,

18:     """Sells one seat. Cash tickets are paid at once; mobile tickets wait for the statement."""

20:     if trip.status != TRIP_SCHEDULED:
21:         raise TripNotOpen(trip_id, trip.status)
22:     sold = _live(tickets_on_trip(store, trip_id))
23:     if len(sold) >= get_vehicle(store, trip.vehicle_id).seats:
24:         raise NoSeatsLeft(trip_id)

27:     route = get_route(store, trip.route_id)
28:     now = clock.now()

32:                     issued_at=now)

1: # Matatu
2: 
3: The back office of a Nairobi minibus operator. It keeps the routes and stages, works
4: out fares, sells tickets, tracks vehicles and crews, matches the mobile-money statement
5: to tickets and prepares the daily figures.
6: 
7: ## Layout
8: 
9: - `matatu/routes`: stages and the order they come in along a route.
10: - `matatu/fares`: distance bands, peak hours, extras and rider discounts, combined into a quote.
11: - `matatu/tickets`: selling tickets, boarding, refunds and day passes.
12: - `matatu/fleet`: vehicles, inspections, fuel and vehicle owners.
13: - `matatu/crews`: drivers and conductors, shifts and commission.
14: - `matatu/payments`: the mobile-money statement, matching lines to tickets, suspense and the provider.
15: - `matatu/reports`: figures for finance and the routes table.
16: - `matatu/notify`: message templates and delivery channels.
17: - `matatu/api`: request validation and the handlers the web layer calls.
18: - `matatu/store`: an in-memory store with small repository helpers.
19: 
20: Money is held as whole Kenya shillings. Times are whole seconds since the epoch, taken from
21: `matatu.clock` so tests can control them.

"""Applying the provider's statement to the tickets it pays for."""

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:
            _mark_paid(store, ticket, line)
            result.matched += 1
    return result

"""Reading the statement file the provider sends each day."""

from typing import List

from ..errors import BadStatement
from ..models import StatementLine
from ..money import parse_amount


def parse_statement(text: str) -> List[StatementLine]:
    """Each line is: code, phone, amount, account reference, seconds since the epoch."""
    lines = []
    for number, raw in enumerate(text.splitlines(), start=1):
        if not raw.strip() or raw.lstrip().startswith("#"):
            continue
        parts = [part.strip() for part in raw.split(",")]
        if len(parts) != 5:
            raise BadStatement(number, "expected five fields")
        code, phone, amount, reference, at = parts
        try:
            lines.append(StatementLine(code=code, phone=phone, amount_kes=parse_amount(amount), account_ref=reference,
                                       at=int(at)))
        except ValueError as problem:
            raise BadStatement(number, str(problem)) from None
    return lines

from collections import Counter
from dataclasses import dataclass
from typing import Dict

from ..clock import day_of
from ..models import METHOD_CASH, METHOD_MOBILE, TICKET_PAID, TICKET_REFUNDED
from ..money import format_kes


@dataclass
class DailyFigures:
    day: str
    tickets_paid: int
    cash_kes: int
    mobile_kes: int
    refunded_kes: int
    methods: Dict[str, int]

    def net_kes(self) -> int:
        return self.cash_kes + self.mobile_kes - self.refunded_kes


def figures_for(store, day: str) -> DailyFigures:
    paid = [t for t in store.tickets.values()
            if t.status in (TICKET_PAID, TICKET_REFUNDED) and t.paid_at is not None and day_of(t.paid_at) == day]
    counted = Counter(t.method for t in paid)
    return DailyFigures(day=day, tickets_paid=len(paid),
                        cash_kes=sum(t.fare_kes for t in paid if t.method == METHOD_CASH),
                        mobile_kes=sum(t.fare_kes for t in paid if t.method == METHOD_MOBILE),
                        refunded_kes=sum(t.fare_kes for t in paid if t.status == TICKET_REFUNDED),
                        methods=dict(counted))


def render_daily(figures: DailyFigures) -> str:
    return "\n".join([
        f"Daily figures for {figures.day}",
        f"Tickets paid: {figures.tickets_paid}",
        f"Cash: {format_kes(figures.cash_kes)}",
        f"Mobile money: {format_kes(figures.mobile_kes)}",
        f"Refunded: {format_kes(figures.refunded_kes)}",
        f"Net: {format_kes(figures.net_kes())}",
    ])

def fares_by_route(store, day: str) -> Dict[str, int]:
    totals: Dict[str, int] = {}
    for ticket in store.tickets.values():
        if ticket.status == TICKET_PAID and day_of(ticket.issued_at) == day:
            totals[ticket.route_id] = totals.get(ticket.route_id, 0) + ticket.fare_kes
    return totals


def league_table(store, day: str) -> List[Tuple[str, int]]:
    """Routes from highest fares to lowest; ties in route order."""
    return sorted(fares_by_route(store, day).items(), key=lambda pair: (-pair[1], pair[0]))

def position(route: Route, stage_id: str) -> int:
    try:
        return route.stage_ids.index(stage_id)
    except ValueError:
        raise StageNotOnRoute(stage_id, route.id) from None


def stages_between(route: Route, from_stage: str, to_stage: str) -> int:
    """How many stages a passenger travels, in either direction along the route."""
    return abs(position(route, to_stage) - position(route, from_stage))


def serves(route: Route, stage_id: str) -> bool:
    return stage_id in route.stage_ids


def routes_through(store, stage_id: str) -> List[Route]:
    return [route for route in store.routes.values() if route.active and serves(route, stage_id)]
