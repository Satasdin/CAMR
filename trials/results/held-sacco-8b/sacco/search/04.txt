Question: Task:
# Tellers need to correct a mistyped deposit

Tellers keep mistyping the amount when they post a deposit, and the ledger balance and the member's printed statement are then wrong until someone at head office fixes the ledger by hand. A deposit of KES 500 posted as KES 5,000 is the usual story.

Add `correct_deposit(store, entry_id, right_amount_cents, clock)` to `sacco/savings/deposits.py`. It fixes the posted deposit so that the ledger balance and the member's statement show the amount that was really paid, and returns the account's new balance in cents. Only a deposit can be corrected; a ledger entry of any other kind, or an amount that is not positive, raises `ValidationFailed`. An unknown entry raises `NotFound`.

def deposit(store, member_id: str, amount_cents: int, clock, source: str = "branch", reference: str = "") -> LedgerEntry:

def deposit(store, member_id: str, amount_cents: int, clock, source: str = "branch", reference: str = "") -> LedgerEntry:
    if amount_cents < MIN_DEPOSIT_CENTS:

from typing import List

from ..errors import NotFound
from ..models import LedgerEntry


def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)

# Tellers need to correct a mistyped deposit

Tellers keep mistyping the amount when they post a deposit, and the ledger balance and the member's printed statement are then wrong until someone at head office fixes the ledger by hand. A deposit of KES 500 posted as KES 5,000 is the usual story.

Add `correct_deposit(store, entry_id, right_amount_cents, clock)` to `sacco/savings/deposits.py`. It fixes the posted deposit so that the ledger balance and the member's statement show the amount that was really paid, and returns the account's new balance in cents. Only a deposit can be corrected; a ledger entry of any other kind, or an amount that is not positive, raises `ValidationFailed`. An unknown entry raises `NotFound`.

"""Every movement of money on a savings account."""

from typing import List

from ..errors import NotFound
from ..models import LedgerEntry


def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)
