12:     return (amount_kes * percent + 50) // 100
13:

15: def add(amounts: Iterable[int]) -> int:
16:     return sum(amounts)
17: 
18: 
19: def round_up_to(amount_kes: int, step: int) -> int:
20:     if step <= 0:
21:         raise ValueError("step must be positive")
22:     return -(-amount_kes // step) * step
23: 
24: 
25: def format_kes(amount_kes: int, currency: str = CURRENCY) -> str:
26:     sign = "-" if amount_kes < 0 else ""
27:     return f"{sign}{currency} {abs(amount_kes):,}"
28: 
29: 
30: def parse_amount(text: str) -> int:
31:     """Reads an amount from a statement or typed by staff, such as '1,250' or '1,250.00'."""
32:     cleaned = text.replace(",", "").strip()
33:     if not cleaned:
34:         raise ValueError("empty amount")
35:     whole, _, part = cleaned.partition(".")
36:     if not whole.isdigit():
37:         raise ValueError(f"not an amount: {text}")
38:     if part.strip("0"):
39:         raise ValueError(f"not a whole number of shillings: {text}")
40:     return int(whole)
41: 
42: 
43: def split_evenly(amount_kes: int, parts: int) -> list:
44:     """Splits an amount so the parts always add back to the amount."""
45:     if parts <= 0:
46:         raise ValueError("parts must be positive")
47:     share, remainder = divmod(amount_kes, parts)
48:     return [share + 1 if index < remainder else share for index in range(parts)]

"""Applying the provider's statement to the tickets it pays for."""

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:
            _mark_paid(store, ticket, line)
            result.matched += 1
    return result

"""Applying the provider's statement to the tickets it pays for."""

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:
            _mark_paid(store, ticket, line)
            result.matched += 1
    return result

"""Applying the provider's statement to the tickets it pays for."""

from dataclasses import dataclass
from typing import Iterable

from ..models import TICKET_ISSUED, TICKET_PAID, StatementLine, Ticket
from ..store.repositories import save_ticket
from .matcher import ticket_for_line
from .suspense import add_to_suspense


@dataclass
class ReconcileResult:
    matched: int = 0
    suspended: int = 0
    skipped: int = 0


def _mark_paid(store, ticket: Ticket, line: StatementLine) -> Ticket:
    ticket.status = TICKET_PAID
    ticket.paid_at = line.at
    ticket.payment_code = line.code
    return save_ticket(store, ticket)


def reconcile_statement(store, lines: Iterable[StatementLine], clock) -> ReconcileResult:
    """Goes through a statement line by line. A line already seen is skipped."""
    result = ReconcileResult()
    for line in lines:
        if line.code in store.seen_codes:
            result.skipped += 1
            continue
        store.seen_codes.add(line.code)
        ticket = ticket_for_line(store, line)
        if ticket is None:
            add_to_suspense(store, line, "unknown reference", clock)
            result.suspended += 1
        elif ticket.status != TICKET_ISSUED:
            add_to_suspense(store, line, "ticket not open", clock)
            result.suspended += 1
        elif line.amount_kes != ticket.fare_kes:
            add_to_suspense(store, line, "wrong amount", clock)
            result.suspended += 1
        else:
            _mark_paid(store, ticket, line)
            result.matched += 1
    return result
