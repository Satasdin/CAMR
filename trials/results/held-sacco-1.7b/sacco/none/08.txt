"""The member register."""

from ..config import DAYS_PER_MONTH, SECONDS_PER_DAY
from ..errors import ValidationFailed
from ..models import Member
from ..savings.accounts import open_account
from ..store.repositories import get_member, save_member
from .kyc import member_problems, normalise_phone


def register_member(store, name: str, phone: str, email: str, national_id: str, clock) -> Member:
    problems = member_problems(name, phone, national_id)
    if problems:
        raise ValidationFailed(problems)
    taken = [m for m in store.members.values() if m.national_id == national_id]
    if taken:
        raise ValidationFailed([f"national_id {national_id} is already registered"])
    member = Member(id=store.next_id("MBR"), name=name.strip(), phone=normalise_phone(phone), email=email,
                    national_id=national_id, joined_at=clock.now())
    save_member(store, member)
    open_account(store, member.id, clock)
    return member


def months_as_member(member: Member, now: int) -> int:
    return (now - member.joined_at) // (DAYS_PER_MONTH * SECONDS_PER_DAY)


def deactivate_member(store, member_id: str) -> Member:
    member = get_member(store, member_id)
    member.active = False
    return save_member(store, member)

"""Checking who a member is before they join."""

from typing import List


def normalise_phone(phone: str) -> str:
    """Turns local numbers such as 0712345678 into +254712345678."""
    cleaned = phone.replace(" ", "").replace("-", "")
    if cleaned.startswith("07") or cleaned.startswith("01"):
        return "+254" + cleaned[1:]
    if cleaned.startswith("254"):
        return "+" + cleaned
    return cleaned


def phone_problems(phone: str) -> List[str]:
    cleaned = normalise_phone(phone)
    if not cleaned.startswith("+254") or len(cleaned) != 13 or not cleaned[1:].isdigit():
        return [f"{phone} is not a Kenyan mobile number"]
    return []


def national_id_problems(national_id: str) -> List[str]:
    if not national_id.isdigit() or not 7 <= len(national_id) <= 8:
        return ["national_id must be 7 or 8 digits"]
    return []


def member_problems(name: str, phone: str, national_id: str) -> List[str]:
    problems = [] if name.strip() else ["name is required"]
    return problems + phone_problems(phone) + national_id_problems(national_id)

"""Applying for a loan and deciding on it."""

from ..errors import InvalidLoanState, NotSecured, ValidationFailed
from ..models import LOAN_APPLIED, LOAN_APPROVED, LOAN_REJECTED, Loan
from ..store.repositories import get_loan, get_member, save_loan
from .eligibility import application_problems
from .guarantors import security_gap
from .products import product


def apply_for_loan(store, member_id: str, product_code: str, amount_cents: int, term_months: int, clock) -> Loan:
    member = get_member(store, member_id)
    loan_product = product(product_code)
    problems = application_problems(store, member, loan_product, amount_cents, term_months, clock.now())
    if problems:
        raise ValidationFailed(problems)
    loan = Loan(id=store.next_id("LN"), member_id=member_id, product=loan_product.code, principal_cents=amount_cents,
                term_months=term_months, monthly_rate_bp=loan_product.monthly_rate_bp, applied_at=clock.now())
    return save_loan(store, loan)


def approve_loan(store, loan_id: str, clock) -> Loan:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPLIED:
        raise InvalidLoanState(loan_id, loan.status)
    short_by = security_gap(store, loan)
    if short_by:
        raise NotSecured(loan_id, short_by)
    loan.status = LOAN_APPROVED
    loan.approved_at = clock.now()
    return save_loan(store, loan)


def reject_loan(store, loan_id: str, reason: str) -> Loan:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPLIED:
        raise InvalidLoanState(loan_id, loan.status)
    loan.status = LOAN_REJECTED
    loan.rejection_reason = reason
    return save_loan(store, loan)

"""Sending an approved loan to the member."""

from ..errors import InvalidLoanState
from ..models import ENTRY_FEE, LOAN_ACTIVE, LOAN_APPROVED, Disbursement, Loan
from ..money import basis_points_of
from ..savings.ledger import post_entry
from ..store.repositories import account_of_member, get_loan, get_member, save_loan
from .fees import record_fee_income

PROCESSING_FEE_BASIS_POINTS = 150


def processing_fee(amount_cents: int) -> int:
    return basis_points_of(amount_cents, PROCESSING_FEE_BASIS_POINTS)


def _activate(store, loan: Loan, clock) -> Loan:
    loan.status = LOAN_ACTIVE
    loan.disbursed_at = clock.now()
    loan.outstanding_principal_cents = loan.principal_cents
    return save_loan(store, loan)


def disburse_loan(store, loan_id: str, gateway, clock) -> Disbursement:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPROVED:
        raise InvalidLoanState(loan_id, loan.status)
    member = get_member(store, loan.member_id)
    fee = processing_fee(loan.principal_cents)
    transfer = gateway.send(member.phone, loan.principal_cents, reference=loan.id)
    _activate(store, loan, clock)
    account = account_of_member(store, loan.member_id)
    post_entry(store, account.id, ENTRY_FEE, -fee, clock, reference=loan.id, note="processing fee")
    record_fee_income(store, loan.id, fee, clock)
    return Disbursement(loan.id, sent_cents=loan.principal_cents, fee_cents=fee, receipt_id=transfer.transfer_id)

"""Whether a member may borrow what they asked for."""

from typing import List

from ..config import MIN_MEMBERSHIP_MONTHS
from ..members.registry import months_as_member
from ..members.shares import has_minimum_shares
from ..models import Member
from ..money import format_cents
from ..savings.accounts import savings_balance
from .products import LoanProduct


def loan_limit(store, member: Member, loan_product: LoanProduct) -> int:
    return savings_balance(store, member.id) * loan_product.savings_multiple


def application_problems(store, member: Member, loan_product: LoanProduct, amount_cents: int, term_months: int, now: int) -> List[str]:
    problems = []
    if not member.active:
        problems.append("member is not active")
    if months_as_member(member, now) < MIN_MEMBERSHIP_MONTHS:
        problems.append(f"members can borrow after {MIN_MEMBERSHIP_MONTHS} months")
    if not has_minimum_shares(member):
        problems.append("member does not hold the minimum shares")
    if amount_cents <= 0:
        problems.append("amount must be positive")
    limit = loan_limit(store, member, loan_product)
    if amount_cents > limit:
        problems.append(f"amount is more than the limit of {format_cents(limit)}")
    if not 1 <= term_months <= loan_product.max_term_months:
        problems.append(f"term must be between 1 and {loan_product.max_term_months} months")
    return problems

"""Members who stand behind another member's loan."""

from typing import List

from ..config import MAX_GUARANTORS
from ..errors import ValidationFailed
from ..models import Guarantee, Loan
from ..savings.accounts import savings_balance
from ..store.repositories import get_loan, get_member


def guarantees_for(store, loan_id: str) -> List[Guarantee]:
    return [item for item in store.guarantees if item.loan_id == loan_id and not item.released]


def pledged_by(store, member_id: str) -> int:
    return sum(item.amount_cents for item in store.guarantees if item.member_id == member_id and not item.released)


def add_guarantee(store, loan_id: str, member_id: str, amount_cents: int, clock) -> Guarantee:
    loan = get_loan(store, loan_id)
    get_member(store, member_id)
    if member_id == loan.member_id:
        raise ValidationFailed(["a member cannot guarantee their own loan"])
    if len(guarantees_for(store, loan_id)) >= MAX_GUARANTORS:
        raise ValidationFailed([f"a loan can have at most {MAX_GUARANTORS} guarantors"])
    if amount_cents <= 0 or amount_cents > savings_balance(store, member_id) - pledged_by(store, member_id):
        raise ValidationFailed(["pledge is more than the guarantor has free"])
    guarantee = Guarantee(loan_id=loan_id, member_id=member_id, amount_cents=amount_cents, created_at=clock.now())
    store.guarantees.append(guarantee)
    return guarantee


def security_gap(store, loan: Loan) -> int:
    """How much of the loan is still not covered by savings and pledges."""
    pledged = sum(item.amount_cents for item in guarantees_for(store, loan.id))
    covered = savings_balance(store, loan.member_id) + pledged
    return max(0, loan.principal_cents - covered)


def release_guarantees(store, loan_id: str) -> int:
    released = guarantees_for(store, loan_id)
    for item in released:
        item.released = True
    return len(released)

from dataclasses import dataclass
from typing import List, Optional

from ..errors import TransferFailed


@dataclass
class Transfer:
    transfer_id: str
    phone: str
    amount_cents: int
    reference: str


class MobileMoneyGateway:
    """What the service needs from a mobile money provider."""

    def send(self, phone: str, amount_cents: int, reference: str) -> Transfer:
        """Sends money. Raises TransferFailed."""
        raise NotImplementedError


class SandboxMpesa(MobileMoneyGateway):
    """Keeps transfers in memory."""

    def __init__(self):
        self.transfers: List[Transfer] = []
        self.fail_next: Optional[Exception] = None

    def send(self, phone: str, amount_cents: int, reference: str) -> Transfer:
        if self.fail_next is not None:
            problem, self.fail_next = self.fail_next, None
            raise problem
        if amount_cents <= 0:
            raise TransferFailed("amount must be positive")
        transfer = Transfer(f"MP{len(self.transfers) + 1:06d}", phone, amount_cents, reference)
        self.transfers.append(transfer)
        return transfer

    def simulate_failure(self) -> None:
        self.fail_next = TransferFailed("provider did not accept the transfer")

class Allocation:

def allocate(loan: Loan, amount_cents: int) -> Allocation:
    remaining = amount_cents
    penalty = min(remaining, loan.penalty_cents)
    remaining -= penalty
    interest = min(remaining, loan.accrued_interest_cents)
