from dataclasses import dataclass
from functools import wraps

from ..errors import SaccoError
from ..loans.applications import apply_for_loan, approve_loan
from ..loans.disbursement import disburse_loan
from ..members.registry import register_member

from ..repayments.payments import apply_repayment
from ..savings.deposits import deposit

from ..savings.withdrawals import withdraw
from ..statements.loan import loan_statement
from ..statements.member import member_statement
from ..store.repositories import get_loan
from .validators import require, require_amount


@dataclass
class Context:
    store: object
    gateway: object
    clock: object
    sender: object


def handled(function):
    """Turns expected errors into a response the web layer can send."""

    @wraps(function)
    def wrapper(request, context):
        try:
            return function(request, context)
        except SaccoError as problem:
            return {"error": problem.code, "message": str(problem)}

    return wrapper


@handled
def join(request, context):
    require(request, "name", "phone", "national_id")
    member = register_member(context.store, request["name"], request["phone"], request.get("email", ""), request["national_id"], context.clock)
    return {"member_id": member.id}


@handled
def deposit_cash(request, context):
    require(request, "member_id")
    amount = require_amount(request)
    entry = deposit(context.store, request["member_id"], amount, context.clock, reference=request.get("reference", ""))
    notify_deposit(context.store, request["member_id"], amount, entry.id, context.sender, context.clock)
    return {"entry_id": entry.id}


@handled
def withdraw_cash(request, context):
    require(request, "member_id")
    amount = require_amount(request)
    entry = withdraw(context.store, request["member_id"], amount, context.clock)
    return {"entry_id": entry.id}


@handled
def request_loan(request, context):
    require(request, "member_id", "product", "term_months")
    amount = require_amount(request)
    loan = apply_for_loan(context.store, request["member_id"], request["product"], amount, request["term_months"], context.clock)
    return {"loan_id": loan.id, "status": loan.status}


@handled
def approve(request, context):
    require(request, "loan_id")
    loan = approve_loan(context.store, request["loan_id"], context.clock)
    return {"loan_id": loan.id, "status": loan.status}


@handled
def send_out(request, context):
    require(request, "loan_id")
    done = disburse_loan(context.store, request["loan_id"], context.gateway, context.clock)
    loan = get_loan(context.store, request["loan_id"])
    notify_loan_sent(context.store, loan.member_id, done.sent_cents, done.receipt_id, context.sender, context.clock)
    return {"loan_id": done.loan_id, "sent_cents": done.sent_cents, "fee_cents": done.fee_cents, "receipt_id": done.receipt_id}


@handled
def repay(request, context):
    require(request, "loan_id")
    amount = require_amount(request)
    repayment = apply_repayment(context.store, request["loan_id"], amount, context.clock, reference=request.get("reference", ""))
    loan = get_loan(context.store, request["loan_id"])
    notify_repayment(context.store, loan.member_id, amount, repayment.id, context.sender, context.clock)
    return {"repayment_id": repayment.id, "loan_status": loan.status}


@handled
def savings_statement(request, context):
    require(request, "member_id", "start", "end")
    return member_statement(context.store, request["member_id"], request["start"], request["end"])


@handled
def loan_position(request, context):
    require(request, "loan_id")
    return loan_statement(context.store, request["loan_id"])


@handled
def reverse_posting(request, context):
    require(request, "entry_id", "reason")
    balance = reverse_entry(context.store, request["entry_id"], request["reason"], context.clock)
    return {"entry_id": request["entry_id"], "balance_cents": balance}

"""Checking requests before they reach the service."""

from typing import Dict, List

from ..errors import ValidationFailed


def missing_fields(request: Dict, fields) -> List[str]:
    return [f"{name} is required" for name in fields if request.get(name) in (None, "")]


def require(request: Dict, *fields) -> None:
    problems = missing_fields(request, fields)
    if problems:
        raise ValidationFailed(problems)


def require_amount(request: Dict, field: str = "amount_cents") -> int:
    require(request, field)
    amount = request[field]
    if not isinstance(amount, int) or isinstance(amount, bool) or amount <= 0:
        raise ValidationFailed([f"{field} must be a positive whole number of cents"])
    return amount

import time
from datetime import datetime, timezone


class SystemClock:
    def now(self) -> int:
        return int(time.time())


class FixedClock:
    """A clock that only moves when told to."""

    def __init__(self, start: int = 1_700_000_000):
        self.current = start

    def now(self) -> int:
        return self.current

    def advance(self, seconds: int) -> int:
        self.current += seconds
        return self.current


def hour_of(moment: int, offset_hours: int = 3) -> int:
    """The hour of the day at a moment, in Nairobi time."""
    return datetime.fromtimestamp(moment + offset_hours * 3600, tz=timezone.utc).hour


def month_of(moment: int, offset_hours: int = 3) -> str:
    return datetime.fromtimestamp(moment + offset_hours * 3600, tz=timezone.utc).strftime("%Y-%m")


def day_of(moment: int, offset_hours: int = 3) -> str:
    return datetime.fromtimestamp(moment + offset_hours * 3600, tz=timezone.utc).strftime("%Y-%m-%d")

"""Identifiers people can read out over the phone."""

ALPHABET = "23456789ABCDEFGHJKLMNPQRSTUVWXYZ"


def encode(number: int, width: int = 6) -> str:
    if number < 0:
        raise ValueError("number must not be negative")
    digits = []
    while number:
        number, remainder = divmod(number, len(ALPHABET))
        digits.append(ALPHABET[remainder])
    text = "".join(reversed(digits)) or ALPHABET[0]
    return text.rjust(width, ALPHABET[0])


class IdGenerator:
    """Hands out identifiers in order, with a prefix per record type."""

    def __init__(self):
        self.counters = {}

    def next(self, prefix: str) -> str:
        value = self.counters.get(prefix, 0) + 1
        self.counters[prefix] = value
        return f"{prefix}-{encode(value)}"

"""Charging each active loan its month of interest."""

from typing import Dict

from ..models import LOAN_ACTIVE
from ..store.repositories import save_loan
from .rates import monthly_interest


def accrue_month(store) -> Dict[str, int]:
    """Adds one month of interest to every active loan and returns what was added."""
    charged = {}
    for loan in store.loans.values():
        if loan.status != LOAN_ACTIVE:
            continue
        amount = monthly_interest(loan.outstanding_principal_cents, loan.monthly_rate_bp)
        loan.accrued_interest_cents += amount
        save_loan(store, loan)
        charged[loan.id] = amount
    return charged

"""Charges for instalments paid late."""

from typing import Dict

from ..config import PENALTY_PERCENT
from ..models import LOAN_ACTIVE
from ..money import percent_of
from ..repayments.arrears import overdue_cents
from ..store.repositories import save_loan


def penalty_on(overdue: int) -> int:
    return percent_of(overdue, PENALTY_PERCENT)


def charge_penalties(store, now: int) -> Dict[str, int]:
    """Adds a penalty to every active loan with money overdue and returns what was added."""
    charged = {}
    for loan in store.loans.values():
        if loan.status != LOAN_ACTIVE:
            continue
        amount = penalty_on(overdue_cents(store, loan, now))
        if amount:
            loan.penalty_cents += amount
            save_loan(store, loan)
            charged[loan.id] = amount
    return charged
