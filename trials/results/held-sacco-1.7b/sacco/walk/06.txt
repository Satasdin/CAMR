<excerpt file="README.md" kind="code">
1: # Sacco
2: 
3: The back office of a small savings and credit co-operative in Kenya. It keeps members
4: and their shares, takes deposits, lends against savings, collects repayments, charges
5: interest and tells members what is happening.
6: 
7: ## Layout
8: 
9: - `sacco/members`: the member register, shares and identity checks.
10: - `sacco/savings`: savings accounts, the ledger, deposits, withdrawals and dividends.
11: - `sacco/loans`: products, eligibility, applications, guarantors, schedules, fees and sending money out.
12: - `sacco/interest`: monthly interest, accrual and late penalties.
13: - `sacco/repayments`: splitting a payment, applying it, arrears and recovery.
14: - `sacco/statements`: statements for savings and for loans.
15: - `sacco/notify`: message templates, delivery channels and who gets told what.
16: - `sacco/api`: request validation and the handlers the web layer calls.
17: - `sacco/reports`: portfolio and monthly figures for the board.
18: - `sacco/store`: an in-memory store with small repository helpers.
19: 
20: Money is held as whole cents. Times are whole seconds since the epoch, taken from
21: `sacco.clock` so tests can control them. Rates are basis points per month.
22: 
23: ## Running the tests
24: 
25:     python3 -m unittest discover -s tests

1: # Sacco
2: 
3: The back office of a small savings and credit co-operative in Kenya. It keeps members
4: and their shares, takes deposits, lends against savings, collects repayments, charges
5: interest and tells members what is happening.
6: 
7: ## Layout
8: 
9: - `sacco/members`: the member register, shares and identity checks.
10: - `sacco/savings`: savings accounts, the ledger, deposits, withdrawals and dividends.
11: - `sacco/loans`: products, eligibility, applications, guarantors, schedules, fees and sending money out.
12: - `sacco/interest`: monthly interest, accrual and late penalties.
13: - `sacco/repayments`: splitting a payment, applying it, arrears and recovery.
14: - `sacco/statements`: statements for savings and for loans.
15: - `sacco/notify`: message templates, delivery channels and who gets told what.
16: - `sacco/api`: request validation and the handlers the web layer calls.
17: - `sacco/reports`: portfolio and monthly figures for the board.
18: - `sacco/store`: an in-memory store with small repository helpers.
19: 
20: Money is held as whole cents. Times are whole seconds since the epoch, taken from
21: `sacco.clock` so tests can control them. Rates are basis points per month.

"""Sending an approved loan to the member."""

from ..errors import InvalidLoanState
from ..models import LOAN_ACTIVE, LOAN_APPROVED, Disbursement, Loan
from ..money import basis_points_of
from ..store.repositories import get_loan, get_member, save_loan
from .fees import record_fee_income

PROCESSING_FEE_BASIS_POINTS = 150


def processing_fee(amount_cents: int) -> int:
    return basis_points_of(amount_cents, PROCESSING_FEE_BASIS_POINTS)


def _activate(store, loan: Loan, clock) -> Loan:
    loan.status = LOAN_ACTIVE
    loan.disbursed_at = clock.now()
    loan.outstanding_principal_cents = loan.principal_cents
    return save_loan(store, loan)


def disburse_loan(store, loan_id: str, gateway, clock) -> Disbursement:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPROVED:
        raise InvalidLoanState(loan_id, loan.status)
    member = get_member(store, loan.member_id)
    fee = processing_fee(loan.principal_cents)
    sent = loan.principal_cents - fee
    transfer = gateway.send(member.phone, sent, reference=loan.id)
    _activate(store, loan, clock)
    record_fee_income(store, loan.id, fee, clock)
    return Disbursement(loan.id, sent_cents=sent, fee_cents=fee, receipt_id=transfer.transfer_id)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Deciding who gets told about something, and how."""

from typing import List

from ..clock import hour_of
from ..config import QUIET_HOURS_END, QUIET_HOURS_START, REMINDER_DAYS_BEFORE, SECONDS_PER_DAY
from ..loans.schedule import loan_schedule
from ..models import LOAN_ACTIVE, Member, Message
from ..money import format_cents
from ..store.repositories import get_member
from .templates import DEPOSIT_RECEIVED, INSTALMENT_DUE, LOAN_SENT, REPAYMENT_RECEIVED, render


def in_quiet_hours(moment: int) -> bool:
    hour = hour_of(moment)
    return hour >= QUIET_HOURS_START or hour < QUIET_HOURS_END


def messages_for(member: Member, template: str, amount_cents: int, reference: str, now: int) -> List[Message]:
    text = render(template, member.name, format_cents(amount_cents), reference)
    out = []
    if member.wants_email and member.email:
        out.append(Message(channel="email", to=member.email, subject=text["subject"], body=text["body"], sent_at=now))
    if member.wants_sms and member.phone and not in_quiet_hours(now):
        out.append(Message(channel="sms", to=member.phone, body=text["body"], sent_at=now))
    return out


def _notify(store, member_id: str, template: str, amount_cents: int, reference: str, sender, clock) -> int:
    member = get_member(store, member_id)
    return sum(1 for message in messages_for(member, template, amount_cents, reference, clock.now()) if sender.send(message))


def notify_deposit(store, member_id: str, amount_cents: int, reference: str, sender, clock) -> int:
    return _notify(store, member_id, DEPOSIT_RECEIVED, amount_cents, reference, sender, clock)


def notify_loan_sent(store, member_id: str, amount_cents: int, reference: str, sender, clock) -> int:
    return _notify(store, member_id, LOAN_SENT, amount_cents, reference, sender, clock)


def notify_repayment(store, member_id: str, amount_cents: int, reference: str, sender, clock) -> int:
    return _notify(store, member_id, REPAYMENT_RECEIVED, amount_cents, reference, sender, clock)


def send_reminders(store, sender, clock) -> int:
    """Reminds borrowers whose next instalment falls due soon and returns how many messages went out."""
    now = clock.now()
    sent = 0
    for loan in store.loans.values():
        if loan.status != LOAN_ACTIVE:
            continue
        for item in loan_schedule(loan):
            if now < item.due_at <= now + REMINDER_DAYS_BEFORE * SECONDS_PER_DAY:
                sent += _notify(store, loan.member_id, INSTALMENT_DUE, item.total_cents, loan.id, sender, clock)
    return sent

"""Splitting a payment between what a borrower owes."""

from dataclasses import dataclass

from ..models import Loan


@dataclass
class Allocation:
    penalty_cents: int
    interest_cents: int
    principal_cents: int
    excess_cents: int


def allocate(loan: Loan, amount_cents: int) -> Allocation:
    remaining = amount_cents
    penalty = min(remaining, loan.penalty_cents)
    remaining -= penalty
    interest = min(remaining, loan.accrued_interest_cents)
    remaining -= interest
    principal = min(remaining, loan.outstanding_principal_cents)
    remaining -= principal
    return Allocation(penalty, interest, principal, remaining)

"""Working out how far behind a borrower is."""

from ..config import GRACE_DAYS, SECONDS_PER_DAY
from ..loans.schedule import loan_schedule
from ..models import Loan
from ..store.repositories import repayments_for


def paid_towards_schedule(store, loan_id: str) -> int:
    return sum(item.amount_cents for item in repayments_for(store, loan_id))


def overdue_cents(store, loan: Loan, now: int) -> int:
    cutoff = now - GRACE_DAYS * SECONDS_PER_DAY
    due = sum(item.total_cents for item in loan_schedule(loan) if item.due_at <= cutoff)
    return max(0, due - paid_towards_schedule(store, loan.id))


def days_in_arrears(store, loan: Loan, now: int) -> int:
    paid = paid_towards_schedule(store, loan.id)
    for item in loan_schedule(loan):
        if paid >= item.total_cents:
            paid -= item.total_cents
            continue
        return max(0, (now - item.due_at) // SECONDS_PER_DAY)
    return 0

"""Applying a payment to a loan."""

from ..errors import InvalidLoanState, ValidationFailed
from ..loans.guarantors import release_guarantees
from ..models import ENTRY_DEPOSIT, LOAN_ACTIVE, LOAN_CLEARED, Repayment
from ..savings.ledger import post_entry

from .allocation import allocate


def apply_repayment(store, loan_id: str, amount_cents: int, clock, source: str = "branch", reference: str = "") -> Repayment:
    if amount_cents <= 0:
        raise ValidationFailed(["amount must be positive"])
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_ACTIVE:
        raise InvalidLoanState(loan_id, loan.status)
    split = allocate(loan, amount_cents)
    loan.penalty_cents -= split.penalty_cents
    loan.accrued_interest_cents -= split.interest_cents
    loan.outstanding_principal_cents -= split.principal_cents
    repayment = Repayment(id=store.next_id("RPY"), loan_id=loan.id, amount_cents=amount_cents, at=clock.now(),
                          penalty_cents=split.penalty_cents, interest_cents=split.interest_cents,
                          principal_cents=split.principal_cents, source=source, reference=reference)
    store.repayments.append(repayment)
    if split.excess_cents:
        account = account_of_member(store, loan.member_id)
        post_entry(store, account.id, ENTRY_DEPOSIT, split.excess_cents, clock, reference=loan.id, note="overpayment")
    if loan.outstanding_principal_cents == 0 and loan.accrued_interest_cents == 0 and loan.penalty_cents == 0:
        loan.status = LOAN_CLEARED
        loan.cleared_at = clock.now()
        release_guarantees(store, loan.id)
    save_loan(store, loan)
    return repayment

from typing import Dict

from ..errors import InvalidLoanState
from ..loans.guarantors import guarantees_for
from ..models import ENTRY_WITHDRAWAL, LOAN_DEFAULTED
from ..money import split_evenly
from ..savings.ledger import post_entry

def recover_from_guarantors(store, loan_id: str, clock) -> Dict[str, int]:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_DEFAULTED:
        raise InvalidLoanState(loan_id, loan.status)
    pledges = guarantees_for(store, loan_id)
    owed = loan.outstanding_principal_cents + loan.accrued_interest_cents + loan.penalty_cents
    taken = {}
    for pledge, share in zip(pledges, split_evenly(owed, len(pledges) or 1)):
        amount = min(share, pledge.amount_cents)
        account = account_of_member(store, pledge.member_id)
        post_entry(store, account.id, ENTRY_WITHDRAWAL, -amount, clock, reference=loan_id, note="guarantor recovery")
        taken[pledge.member_id] = amount
    return taken

def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)

PROCESSING_FEE_BASIS_POINTS = 150


def processing_fee(amount_cents: int) -> int:
    return basis_points_of(amount_cents, PROCESSING_FEE_BASIS_POINTS)


def _activate(store, loan: Loan, clock) -> Loan:
    loan.status = LOAN_ACTIVE
    loan.disbursed_at = clock.now()
    loan.outstanding_principal_cents = loan.principal_cents
    return save_loan(store, loan)


def disburse_loan(store, loan_id: str, gateway, clock) -> Disbursement:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPROVED:
        raise InvalidLoanState(loan_id, loan.status)
    member = get_member(store, loan.member_id)
    fee = processing_fee(loan.principal_cents)
    sent = loan.principal_cents - fee
    transfer = gateway.send(member.phone, sent, reference=loan.id)
    _activate(store, loan, clock)
    record_fee_income(store, loan.id, fee, clock)
    return Disbursement(loan.id, sent_cents=sent, fee_cents=fee, receipt_id=transfer.transfer_id)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Splitting a payment between what a borrower owes."""

from dataclasses import dataclass

from ..models import Loan


@dataclass
class Allocation:
    penalty_cents: int
    interest_cents: int
    principal_cents: int
    excess_cents: int


def allocate(loan: Loan, amount_cents: int) -> Allocation:
    remaining = amount_cents
    penalty = min(remaining, loan.penalty_cents)
    remaining -= penalty
    interest = min(remaining, loan.accrued_interest_cents)
    remaining -= interest
    principal = min(remaining, loan.outstanding_principal_cents)
    remaining -= principal
    return Allocation(penalty, interest, principal, remaining)

"""Working out how far behind a borrower is."""

from ..config import GRACE_DAYS, SECONDS_PER_DAY
from ..loans.schedule import loan_schedule
from ..models import Loan
from ..store.repositories import repayments_for


def paid_towards_schedule(store, loan_id: str) -> int:
    return sum(item.amount_cents for item in repayments_for(store, loan_id))


def overdue_cents(store, loan: Loan, now: int) -> int:
    cutoff = now - GRACE_DAYS * SECONDS_PER_DAY
    due = sum(item.total_cents for item in loan_schedule(loan) if item.due_at <= cutoff)
    return max(0, due - paid_towards_schedule(store, loan.id))


def days_in_arrears(store, loan: Loan, now: int) -> int:
    paid = paid_towards_schedule(store, loan.id)
    for item in loan_schedule(loan):
        if paid >= item.total_cents:
            paid -= item.total_cents
            continue
        return max(0, (now - item.due_at) // SECONDS_PER_DAY)
    return 0

"""Applying a payment to a loan."""

from ..errors import InvalidLoanState, ValidationFailed
from ..loans.guarantors import release_guarantees
from ..models import ENTRY_DEPOSIT, LOAN_ACTIVE, LOAN_CLEARED, Repayment

def apply_repayment(store, loan_id: str, amount_cents: int, clock, source: str = "branch", reference: str = "") -> Repayment:
    if amount_cents <= 0:
        raise ValidationFailed(["amount must be positive"])
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_ACTIVE:
        raise InvalidLoanState(loan_id, loan.status)
    split = allocate(loan, amount_cents)
    loan.penalty_cents -= split.penalty_cents
    loan.accrued_interest_cents -= split.interest_cents
    loan.outstanding_principal_cents -= split.principal_cents
    repayment = Repayment(id=store.next_id("RPY"), loan_id=loan.id, amount_cents=amount_cents, at=clock.now(),
                          penalty_cents=split.penalty_cents, interest_cents=split.interest_cents,
                          principal_cents=split.principal_cents, source=source, reference=reference)
    store.repayments.append(repayment)
    if split.excess_cents:
        account = account_of_member(store, loan.member_id)
        post_entry(store, account.id, ENTRY_DEPOSIT, split.excess_cents, clock, reference=loan.id, note="overpayment")
    if loan.outstanding_principal_cents == 0 and loan.accrued_interest_cents == 0 and loan.penalty_cents == 0:
        loan.status = LOAN_CLEARED
        loan.cleared_at = clock.now()
        release_guarantees(store, loan.id)
    save_loan(store, loan)
    return repayment

"""Collecting a defaulted loan from those who pledged for it."""

from typing import Dict

from ..errors import InvalidLoanState
from ..loans.guarantors import guarantees_for
from ..models import ENTRY_WITHDRAWAL, LOAN_DEFAULTED
from ..money import split_evenly
from ..savings.ledger import post_entry
from ..store.repositories import account_of_member, get_loan


def recover_from_guarantors(store, loan_id: str, clock) -> Dict[str, int]:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_DEFAULTED:
        raise InvalidLoanState(loan_id, loan.status)
    pledges = guarantees_for(store, loan_id)
    owed = loan.outstanding_principal_cents + loan.accrued_interest_cents + loan.penalty_cents
    taken = {}
    for pledge, share in zip(pledges, split_evenly(owed, len(pledges) or 1)):
        amount = min(share, pledge.amount_cents)
        account = account_of_member(store, pledge.member_id)
        post_entry(store, account.id, ENTRY_WITHDRAWAL, -amount, clock, reference=loan_id, note="guarantor recovery")
        taken[pledge.member_id] = amount
    return taken

from ..errors import NotFound

def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)

1: # Sacco
2: 
3: The back office of a small savings and credit co-operative in Kenya. It keeps members
4: and their shares, takes deposits, lends against savings, collects repayments, charges
5: interest and tells members what is happening.
6: 
7: ## Layout
8: 
9: - `sacco/members`: the member register, shares and identity checks.
10: - `sacco/savings`: savings accounts, the ledger, deposits, withdrawals and dividends.
11: - `sacco/loans`: products, eligibility, applications, guarantors, schedules, fees and sending money out.
12: - `sacco/interest`: monthly interest, accrual and late penalties.
13: - `sacco/repayments`: splitting a payment, applying it, arrears and recovery.
14: - `sacco/statements`: statements for savings and for loans.
15: - `sacco/notify`: message templates, delivery channels and who gets told what.
16: - `sacco/api`: request validation and the handlers the web layer calls.
17: - `sacco/reports`: portfolio and monthly figures for the board.
18: - `sacco/store`: an in-memory store with small repository helpers.
19: 
20: Money is held as whole cents. Times are whole seconds since the epoch, taken from
21: `sacco.clock` so tests can control them. Rates are basis points per month.

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Splitting a payment between what a borrower owes."""

from dataclasses import dataclass

from ..models import Loan


@dataclass
class Allocation:
    penalty_cents: int
    interest_cents: int
    principal_cents: int
    excess_cents: int


def allocate(loan: Loan, amount_cents: int) -> Allocation:
    remaining = amount_cents
    penalty = min(remaining, loan.penalty_cents)
    remaining -= penalty
    interest = min(remaining, loan.accrued_interest_cents)
    remaining -= interest
    principal = min(remaining, loan.outstanding_principal_cents)
    remaining -= principal
    return Allocation(penalty, interest, principal, remaining)

"""Working out how far behind a borrower is."""

from ..config import GRACE_DAYS, SECONDS_PER_DAY
from ..loans.schedule import loan_schedule
from ..models import Loan
from ..store.repositories import repayments_for


def paid_towards_schedule(store, loan_id: str) -> int:
    return sum(item.amount_cents for item in repayments_for(store, loan_id))


def overdue_cents(store, loan: Loan, now: int) -> int:
    cutoff = now - GRACE_DAYS * SECONDS_PER_DAY
    due = sum(item.total_cents for item in loan_schedule(loan) if item.due_at <= cutoff)
    return max(0, due - paid_towards_schedule(store, loan.id))


def days_in_arrears(store, loan: Loan, now: int) -> int:
    paid = paid_towards_schedule(store, loan.id)
    for item in loan_schedule(loan):
        if paid >= item.total_cents:
            paid -= item.total_cents
            continue
        return max(0, (now - item.due_at) // SECONDS_PER_DAY)
    return 0

from ..loans.guarantors import release_guarantees

def apply_repayment(store, loan_id: str, amount_cents: int, clock, source: str = "branch", reference: str = "") -> Repayment:
    if amount_cents <= 0:
        raise ValidationFailed(["amount must be positive"])
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_ACTIVE:
        raise InvalidLoanState(loan_id, loan.status)
    split = allocate(loan, amount_cents)
    loan.penalty_cents -= split.penalty_cents
    loan.accrued_interest_cents -= split.interest_cents
    loan.outstanding_principal_cents -= split.principal_cents
    repayment = Repayment(id=store.next_id("RPY"), loan_id=loan.id, amount_cents=amount_cents, at=clock.now(),
                          penalty_cents=split.penalty_cents, interest_cents=split.interest_cents,
                          principal_cents=split.principal_cents, source=source, reference=reference)
    store.repayments.append(repayment)
    if split.excess_cents:
        account = account_of_member(store, loan.member_id)
        post_entry(store, account.id, ENTRY_DEPOSIT, split.excess_cents, clock, reference=loan.id, note="overpayment")
    if loan.outstanding_principal_cents == 0 and loan.accrued_interest_cents == 0 and loan.penalty_cents == 0:
        loan.status = LOAN_CLEARED
        loan.cleared_at = clock.now()
        release_guarantees(store, loan.id)
    save_loan(store, loan)
    return repayment

def recover_from_guarantors(store, loan_id: str, clock) -> Dict[str, int]:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_DEFAULTED:
        raise InvalidLoanState(loan_id, loan.status)
    pledges = guarantees_for(store, loan_id)
    owed = loan.outstanding_principal_cents + loan.accrued_interest_cents + loan.penalty_cents
    taken = {}
    for pledge, share in zip(pledges, split_evenly(owed, len(pledges) or 1)):

        post_entry(store, account.id, ENTRY_WITHDRAWAL, -amount, clock, reference=loan_id, note="guarantor recovery")

4: from ..models import LOAN_ACTIVE, LOAN_APPROVED, Disbursement, Loan

9: PROCESSING_FEE_BASIS_POINTS = 150


def processing_fee(amount_cents: int) -> int:
    return basis_points_of(amount_cents, PROCESSING_FEE_BASIS_POINTS)


def _activate(store, loan: Loan, clock) -> Loan:
    loan.status = LOAN_ACTIVE
    loan.disbursed_at = clock.now()
    loan.outstanding_principal_cents = loan.principal_cents
    return save_loan(store, loan)


def disburse_loan(store, loan_id: str, gateway, clock) -> Disbursement:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPROVED:
        raise InvalidLoanState(loan_id, loan.status)
    member = get_member(store, loan.member_id)
    fee = processing_fee(loan.principal_cents)
    sent = loan.principal_cents - fee
    transfer = gateway.send(member.phone, sent, reference=loan.id)
    _activate(store, loan, clock)
    record_fee_income(store, loan.id, fee, clock)
    return Disbursement(loan.id, sent_cents=sent, fee_cents=fee, receipt_id=transfer.transfer_id)
