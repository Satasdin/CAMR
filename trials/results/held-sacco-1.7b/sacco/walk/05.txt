1: # Sacco

3: The back office of a small savings and credit co-operative in Kenya. It keeps members

8:

13: - `sacco/repayments`: splitting a payment, applying it, arrears and recovery.
14: - `sacco/statements`: statements for savings and for loans.

16: - `sacco/api`: request validation and the handlers the web layer calls.
17: - `sacco/reports`: portfolio and monthly figures for the board.
18: - `sacco/store`: an in-memory store with small repository helpers.

"""A member's savings statement for a period."""

from typing import Dict

from ..money import format_cents
from ..savings.ledger import entries_for
from ..store.repositories import account_of_member, get_member


def member_statement(store, member_id: str, start: int, end: int) -> Dict:
    """Every ledger line from start up to but not including end, with a running balance."""
    get_member(store, member_id)
    entries = entries_for(store, account_of_member(store, member_id).id)
    opening = sum(entry.amount_cents for entry in entries if entry.at < start)
    balance = opening
    lines = []
    for entry in entries:
        if not start <= entry.at < end:
            continue
        balance += entry.amount_cents
        lines.append({"entry_id": entry.id, "at": entry.at, "kind": entry.kind, "amount_cents": entry.amount_cents,
                      "balance_cents": balance, "note": entry.note})
    return {"member_id": member_id, "opening_balance_cents": opening, "lines": lines, "closing_balance_cents": balance}


def render_statement(statement: Dict) -> str:
    rows = [f"Statement for {statement['member_id']}", f"Opening balance: {format_cents(statement['opening_balance_cents'])}"]
    for line in statement["lines"]:
        rows.append(f"  {line['kind']}: {format_cents(line['amount_cents'])}  balance {format_cents(line['balance_cents'])}")
    rows.append(f"Closing balance: {format_cents(statement['closing_balance_cents'])}")
    return "\n".join(rows)

def deposit(store, member_id: str, amount_cents: int, clock, source: str = "branch", reference: str = "") -> LedgerEntry:
    if amount_cents < MIN_DEPOSIT_CENTS:
        raise ValidationFailed([f"deposits start at {format_cents(MIN_DEPOSIT_CENTS)}"])
    member = get_member(store, member_id)
    if not member.active:
        raise MemberInactive(member_id)
    account = account_of_member(store, member_id)
    if account.frozen:
        raise ValidationFailed(["this account is frozen"])
    return post_entry(store, account.id, ENTRY_DEPOSIT, amount_cents, clock, reference=reference, note=source)


def correct_deposit(store, entry_id: str, right_amount_cents: int, clock) -> int:
    if right_amount_cents <= 0:
        raise ValidationFailed(["amount must be positive"])
    entry = find_entry(store, entry_id)
    if entry.kind != ENTRY_DEPOSIT:
        raise ValidationFailed(["only a deposit can be corrected"])
    reverse_entry(store, entry_id, f"corrected to {format_cents(right_amount_cents)}", clock)
    post_entry(store, entry.account_id, ENTRY_DEPOSIT, right_amount_cents, clock, reference=entry.id, note="correction")
    return balance_of(store, entry.account_id)

1: # Sacco
2: 
3: The back office of a small savings and credit co-operative in Kenya. It keeps members
4: and their shares, takes deposits, lends against savings, collects repayments, charges
5: interest and tells members what is happening.
6: 
7: ## Layout
8: 
9: - `sacco/members`: the member register, shares and identity checks.
10: - `sacco/savings`: savings accounts, the ledger, deposits, withdrawals and dividends.
11: - `sacco/loans`: products, eligibility, applications, guarantors, schedules, fees and sending money out.
12: - `sacco/interest`: monthly interest, accrual and late penalties.
13: - `sacco/repayments`: splitting a payment, applying it, arrears and recovery.
14: - `sacco/statements`: statements for savings and for loans.
15: - `sacco/notify`: message templates, delivery channels and who gets told what.
16: - `sacco/api`: request validation and the handlers the web layer calls.
17: - `sacco/reports`: portfolio and monthly figures for the board.
18: - `sacco/store`: an in-memory store with small repository helpers.
19: 
20: Money is held as whole cents. Times are whole seconds since the epoch, taken from
21: `sacco.clock` so tests can control them. Rates are basis points per month.

"""A member's savings statement for a period."""

from typing import Dict

from ..money import format_cents
from ..savings.ledger import entries_for
from ..store.repositories import account_of_member, get_member


def member_statement(store, member_id: str, start: int, end: int) -> Dict:
    """Every ledger line from start up to but not including end, with a running balance."""
    get_member(store, member_id)
    entries = entries_for(store, account_of_member(store, member_id).id)
    opening = sum(entry.amount_cents for entry in entries if entry.at < start)
    balance = opening
    lines = []
    for entry in entries:
        if not start <= entry.at < end:
            continue
        balance += entry.amount_cents
        lines.append({"entry_id": entry.id, "at": entry.at, "kind": entry.kind, "amount_cents": entry.amount_cents,
                      "balance_cents": balance, "note": entry.note})
    return {"member_id": member_id, "opening_balance_cents": opening, "lines": lines, "closing_balance_cents": balance}


def render_statement(statement: Dict) -> str:
    rows = [f"Statement for {statement['member_id']}", f"Opening balance: {format_cents(statement['opening_balance_cents'])}"]
    for line in statement["lines"]:
        rows.append(f"  {line['kind']}: {format_cents(line['amount_cents'])}  balance {format_cents(line['balance_cents'])}")
    rows.append(f"Closing balance: {format_cents(statement['closing_balance_cents'])}")
    return "\n".join(rows)

def deposit(store, member_id: str, amount_cents: int, clock, source: str = "branch", reference: str = "") -> LedgerEntry:
    if amount_cents < MIN_DEPOSIT_CENTS:
        raise ValidationFailed([f"deposits start at {format_cents(MIN_DEPOSIT_CENTS)}"])
    member = get_member(store, member_id)
    if not member.active:

    account = account_of_member(store, member_id)
    if account.frozen:

    if entry.kind != ENTRY_DEPOSIT:

    reverse_entry(store, entry_id, f"corrected to {format_cents(right_amount_cents)}", clock)

def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)

4: and their shares, takes deposits, lends against savings, collects repayments, charges
5: interest and tells members what is happening.
6: 
7: ## Layout
8: 
9: - `sacco/members`: the member register, shares and identity checks.
10: - `sacco/savings`: savings accounts, the ledger, deposits, withdrawals and dividends.
11: - `sacco/loans`: products, eligibility, applications, guarantors, schedules, fees and sending money out.
12: - `sacco/interest`: monthly interest, accrual and late penalties.
13: - `sacco/repayments`: splitting a payment, applying it, arrears and recovery.
14: - `sacco/statements`: statements for savings and for loans.
15: - `sacco/notify`: message templates, delivery channels and who gets told what.
16: - `sacco/api`: request validation and the handlers the web layer calls.
17: - `sacco/reports`: portfolio and monthly figures for the board.
18: - `sacco/store`: an in-memory store with small repository helpers.
19: 
20: Money is held as whole cents. Times are whole seconds since the epoch, taken from
21: `sacco.clock` so tests can control them. Rates are basis points per month.

"""A member's savings statement for a period."""

from typing import Dict

from ..money import format_cents
from ..savings.ledger import entries_for
from ..store.repositories import account_of_member, get_member


def member_statement(store, member_id: str, start: int, end: int) -> Dict:
    """Every ledger line from start up to but not including end, with a running balance."""
    get_member(store, member_id)
    entries = entries_for(store, account_of_member(store, member_id).id)
    opening = sum(entry.amount_cents for entry in entries if entry.at < start)
    balance = opening
    lines = []
    for entry in entries:
        if not start <= entry.at < end:
            continue
        balance += entry.amount_cents
        lines.append({"entry_id": entry.id, "at": entry.at, "kind": entry.kind, "amount_cents": entry.amount_cents,
                      "balance_cents": balance, "note": entry.note})
    return {"member_id": member_id, "opening_balance_cents": opening, "lines": lines, "closing_balance_cents": balance}


def render_statement(statement: Dict) -> str:
    rows = [f"Statement for {statement['member_id']}", f"Opening balance: {format_cents(statement['opening_balance_cents'])}"]
    for line in statement["lines"]:
        rows.append(f"  {line['kind']}: {format_cents(line['amount_cents'])}  balance {format_cents(line['balance_cents'])}")
    rows.append(f"Closing balance: {format_cents(statement['closing_balance_cents'])}")
    return "\n".join(rows)

from ..config import MIN_DEPOSIT_CENTS
from ..errors import MemberInactive, ValidationFailed
from ..models import ENTRY_DEPOSIT, LedgerEntry
from ..money import format_cents
from ..store.repositories import account_of_member, get_member
from .ledger import balance_of, find_entry, post_entry
from .reversals import reverse_entry


def deposit(store, member_id: str, amount_cents: int, clock, source: str = "branch", reference: str = "") -> LedgerEntry:
    if amount_cents < MIN_DEPOSIT_CENTS:
        raise ValidationFailed([f"deposits start at {format_cents(MIN_DEPOSIT_CENTS)}"])
    member = get_member(store, member_id)
    if not member.active:
        raise MemberInactive(member_id)
    account = account_of_member(store, member_id)
    if account.frozen:
        raise ValidationFailed(["this account is frozen"])
    return post_entry(store, account.id, ENTRY_DEPOSIT, amount_cents, clock, reference=reference, note=source)


def correct_deposit(store, entry_id: str, right_amount_cents: int, clock) -> int:
    if right_amount_cents <= 0:
        raise ValidationFailed(["amount must be positive"])
    entry = find_entry(store, entry_id)
    if entry.kind != ENTRY_DEPOSIT:
        raise ValidationFailed(["only a deposit can be corrected"])
    reverse_entry(store, entry_id, f"corrected to {format_cents(right_amount_cents)}", clock)
    post_entry(store, entry.account_id, ENTRY_DEPOSIT, right_amount_cents, clock, reference=entry.id, note="correction")
    return balance_of(store, entry.account_id)

"""Every movement of money on a savings account."""

from typing import List

from ..errors import NotFound
from ..models import LedgerEntry


def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)

from ..errors import AlreadyReversed
from ..models import ENTRY_REVERSAL
from .ledger import balance_of, find_entry, post_entry


def reverse_entry(store, entry_id: str, reason: str, clock) -> int:
    entry = find_entry(store, entry_id)
    if any(item.kind == ENTRY_REVERSAL and item.reference == entry_id for item in store.ledger):
        raise AlreadyReversed(entry_id)
    post_entry(store, entry.account_id, ENTRY_REVERSAL, -entry.amount_cents, clock, reference=entry.id, note=reason)
    return balance_of(store, entry.account_id)
