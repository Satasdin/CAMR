# Tellers need to correct a mistyped deposit

Tellers keep mistyping the amount when they post a deposit, and the ledger balance and the member's printed statement are then wrong until someone at head office fixes the ledger by hand. A deposit of KES 500 posted as KES 5,000 is the usual story.

Add `correct_deposit(store, entry_id, right_amount_cents, clock)` to `sacco/savings/deposits.py`. It fixes the posted deposit so that the ledger balance and the member's statement show the amount that was really paid, and returns the account's new balance in cents. Only a deposit can be corrected; a ledger entry of any other kind, or an amount that is not positive, raises `ValidationFailed`. An unknown entry raises `NotFound`.

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

def available_to_withdraw(store, member_id: str) -> int:

def withdraw(store, member_id: str, amount_cents: int, clock, reference: str = "") -> LedgerEntry:
    if amount_cents <= 0:

def member_statement(store, member_id: str, start: int, end: int) -> Dict:
    """Every ledger line from start up to but not including end, with a running balance."""
    get_member(store, member_id)
    entries = entries_for(store, account_of_member(store, member_id).id)
    opening = sum(entry.amount_cents for entry in entries if entry.at < start)
    balance = opening
    lines = []
    for entry in entries:
        if not start <= entry.at < end:
            continue
        balance += entry.amount_cents
        lines.append({"entry_id": entry.id, "at": entry.at, "kind": entry.kind, "amount_cents": entry.amount_cents,
                      "balance_cents": balance, "note": entry.note})
    return {"member_id": member_id, "opening_balance_cents": opening, "lines": lines, "closing_balance_cents": balance}


def render_statement(statement: Dict) -> str:
    rows = [f"Statement for {statement['member_id']}", f"Opening balance: {format_cents(statement['opening_balance_cents'])}"]
    for line in statement["lines"]:
        rows.append(f"  {line['kind']}: {format_cents(line['amount_cents'])}  balance {format_cents(line['balance_cents'])}")
    rows.append(f"Closing balance: {format_cents(statement['closing_balance_cents'])}")
    return "\n".join(rows)

def correct_deposit(store, entry_id, right_amount_cents, clock) -> int:
    entry = find_entry(store, entry_id)
    if entry.kind != ENTRY_DEPOSIT:
        raise ValidationFailed(["only deposits can be corrected"])
    if right_amount_cents <= 0:
        raise ValidationFailed(["amount must be positive"])
    if entry.amount_cents != right_amount_cents:
        # Update the ledger entry
        store.ledger[entry.index] = LedgerEntry(
            id=entry.id,
            account_id=entry.account_id,
            kind=ENTRY_DEPOSIT,
            amount_cents=right_amount_cents,
            at=entry.at,
            reference=entry.reference,
            note=entry.note
        )
        # Update the account balance
        account = account_of_member(store, entry.member_id)
        account.balance = balance_of(store, account.id)
        save_account(store, account)
        return account.balance
    return entry.amount_cents

def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

def available_to_withdraw(store, member_id: str) -> int:
    balance = savings_balance(store, member_id)
    return max(0, balance - pledged_by(store, member_id) - MIN_SAVINGS_LEFT_CENTS)


def withdraw(store, member_id: str, amount_cents: int, clock, reference: str = "") -> LedgerEntry:
    if amount_cents <= 0:
        raise ValidationFailed(["amount must be positive"])

    if not member.active:

        raise InsufficientSavings(amount_cents, available)
    account = account_of_member(store, member_id)
    return post_entry(store, account.id, ENTRY_WITHDRAWAL, -amount_cents, clock, reference=reference)

"""A member's savings statement for a period."""

from typing import Dict

from ..money import format_cents
from ..savings.ledger import entries_for
from ..store.repositories import account_of_member, get_member


def member_statement(store, member_id: str, start: int, end: int) -> Dict:
    """Every ledger line from start up to but not including end, with a running balance."""
    get_member(store, member_id)
    entries = entries_for(store, account_of_member(store, member_id).id)
    opening = sum(entry.amount_cents for entry in entries if entry.at < start)
    balance = opening
    lines = []
    for entry in entries:
        if not start <= entry.at < end:
            continue
        balance += entry.amount_cents
        lines.append({"entry_id": entry.id, "at": entry.at, "kind": entry.kind, "amount_cents": entry.amount_cents,
                      "balance_cents": balance, "note": entry.note})
    return {"member_id": member_id, "opening_balance_cents": opening, "lines": lines, "closing_balance_cents": balance}


def render_statement(statement: Dict) -> str:
    rows = [f"Statement for {statement['member_id']}", f"Opening balance: {format_cents(statement['opening_balance_cents'])}"]
    for line in statement["lines"]:
        rows.append(f"  {line['kind']}: {format_cents(line['amount_cents'])}  balance {format_cents(line['balance_cents'])}")
    rows.append(f"Closing balance: {format_cents(statement['closing_balance_cents'])}")
    return "\n".join(rows)

def post_entry(store, account_id: str, kind: str, amount_cents: int, clock, reference: str = "", note: str = "") -> LedgerEntry:
    entry = LedgerEntry(id=store.next_id("ENT"), account_id=account_id, kind=kind, amount_cents=amount_cents,
                        at=clock.now(), reference=reference, note=note)
    store.ledger.append(entry)
    return entry


def find_entry(store, entry_id: str) -> LedgerEntry:
    for entry in store.ledger:
        if entry.id == entry_id:
            return entry
    raise NotFound("entry", entry_id)


def entries_for(store, account_id: str) -> List[LedgerEntry]:
    found = [entry for entry in store.ledger if entry.account_id == account_id]
    return sorted(found, key=lambda entry: entry.at)


def balance_of(store, account_id: str) -> int:
    return sum(entry.amount_cents for entry in store.ledger if entry.account_id == account_id)

from ..errors import AlreadyReversed
from ..models import ENTRY_REVERSAL
from .ledger import balance_of, find_entry, post_entry


def reverse_entry(store, entry_id: str, reason: str, clock) -> int:
    entry = find_entry(store, entry_id)
    if any(item.kind == ENTRY_REVERSAL and item.reference == entry_id for item in store.ledger):
        raise AlreadyReversed(entry_id)
    post_entry(store, entry.account_id, ENTRY_REVERSAL, -entry.amount_cents, clock, reference=entry.id, note=reason)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

from ..config import MIN_SAVINGS_LEFT_CENTS
from ..errors import InsufficientSavings, MemberInactive, ValidationFailed
from ..loans.guarantors import pledged_by
from ..models import ENTRY_WITHDRAWAL, LedgerEntry
from ..store.repositories import account_of_member, get_member
from .accounts import savings_balance
from .ledger import post_entry


def available_to_withdraw(store, member_id: str) -> int:
    balance = savings_balance(store, member_id)
    return max(0, balance - pledged_by(store, member_id) - MIN_SAVINGS_LEFT_CENTS)


def withdraw(store, member_id: str, amount_cents: int, clock, reference: str = "") -> LedgerEntry:
    if amount_cents <= 0:
        raise ValidationFailed(["amount must be positive"])
    member = get_member(store, member_id)
    if not member.active:
        raise MemberInactive(member_id)
    available = available_to_withdraw(store, member_id)
    if amount_cents > available:
        raise InsufficientSavings(amount_cents, available)
    account = account_of_member(store, member_id)
    return post_entry(store, account.id, ENTRY_WITHDRAWAL, -amount_cents, clock, reference=reference)

"""A member's savings statement for a period."""

from typing import Dict

from ..money import format_cents
from ..savings.ledger import entries_for
from ..store.repositories import account_of_member, get_member


def member_statement(store, member_id: str, start: int, end: int) -> Dict:
    """Every ledger line from start up to but not including end, with a running balance."""
    get_member(store, member_id)
    entries = entries_for(store, account_of_member(store, member_id).id)
    opening = sum(entry.amount_cents for entry in entries if entry.at < start)
    balance = opening
    lines = []
    for entry in entries:
        if not start <= entry.at < end:
            continue
        balance += entry.amount_cents
        lines.append({"entry_id": entry.id, "at": entry.at, "kind": entry.kind, "amount_cents": entry.amount_cents,
                      "balance_cents": balance, "note": entry.note})
    return {"member_id": member_id, "opening_balance_cents": opening, "lines": lines, "closing_balance_cents": balance}


def render_statement(statement: Dict) -> str:
    rows = [f"Statement for {statement['member_id']}", f"Opening balance: {format_cents(statement['opening_balance_cents'])}"]
    for line in statement["lines"]:
        rows.append(f"  {line['kind']}: {format_cents(line['amount_cents'])}  balance {format_cents(line['balance_cents'])}")
    rows.append(f"Closing balance: {format_cents(statement['closing_balance_cents'])}")
    return "\n".join(rows)
