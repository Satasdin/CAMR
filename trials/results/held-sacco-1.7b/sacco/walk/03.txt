9:     """Rounds half up."""
10:     if amount_cents < 0:
11:         raise ValueError("amount must not be negative")
12:     return (amount_cents * percent + 50) // 100
13: 
14: 
15: def basis_points_of(amount_cents: int, basis_points: int) -> int:
16:     """Rounds half up."""
17:     if amount_cents < 0:
18:         raise ValueError("amount must not be negative")
19:     return (amount_cents * basis_points + 5_000) // 10_000
20: 
21: 
22: def add(amounts: Iterable[int]) -> int:
23:     return sum(amounts)
24: 
25: 
26: def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
27:     sign = "-" if amount_cents < 0 else ""
28:     whole, part = divmod(abs(amount_cents), 100)
29:     return f"{sign}{currency} {whole:,}.{part:02d}"
30: 
31: 
32: def parse_amount(text: str) -> int:
33:     """Reads an amount typed by staff, such as '1,250.50', into cents."""
34:     cleaned = text.replace(",", "").strip()
35:     if not cleaned:
36:         raise ValueError("empty amount")
37:     negative = cleaned.startswith("-")
38:     if negative:
39:         cleaned = cleaned[1:]
40:     whole, _, part = cleaned.partition(".")
41:     if not whole.isdigit() and whole != "":
42:         raise ValueError(f"not an amount: {text}")
43:     if part and (not part.isdigit() or len(part) > 2):
44:         raise ValueError(f"not an amount: {text}")
45:     cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
46:     return -cents if negative else cents

<excerpt file="sacco/loans/guarantors.py" kind="code">
1: """Members who stand behind another member's loan."""
2: 
3: from typing import List
4: 
5: from ..config import MAX_GUARANTORS
6: from ..errors import ValidationFailed
7: from ..models import Guarantee, Loan
8: from ..savings.accounts import savings_balance
9: from ..store.repositories import get_loan, get_member
10: 
11: 
12: def guarantees_for(store, loan_id: str) -> List[Guarantee]:
13:     return [item for item in store.guarantees if item.loan_id == loan_id and not item.released]
14: 
15: 
16: def pledged_by(store, member_id: str) -> int:
17:     return sum(item.amount_cents for item in store.guarantees if item.member_id == member_id and not item.released)
18: 
19: 
20: def add_guarantee(store, loan_id: str, member_id: str, amount_cents: int, clock) -> Guarantee:
21:     loan = get_loan(store, loan_id)
22:     get_member(store, member_id)
23:     if member_id == loan.member_id:
24:         raise ValidationFailed(["a member cannot guarantee their own loan"])
25:     if len(guarantees_for(store, loan_id)) >= MAX_GUARANTORS:
26:         raise ValidationFailed([f"a loan can have at most {MAX_GUARANTORS} guarantors"])
27:     if amount_cents <= 0 or amount_cents > savings_balance(store, member_id) - pledged_by(store, member_id):
28:         raise ValidationFailed(["pledge is more than the guarantor has free"])
29:     guarantee = Guarantee(loan_id=loan_id, member_id=member_id, amount_cents=amount_cents, created_at=clock.now())
30:     store.guarantees.append(guarantee)
31:     return guarantee
32: 
33: 
34: def security_gap(store, loan: Loan) -> int:
35:     """How much of the loan is still not covered by savings and pledges."""
36:     pledged = sum(item.amount_cents for item in guarantees_for(store, loan.id))
37:     covered = savings_balance(store, loan.member_id) + pledged
38:     return max(0, loan.principal_cents - covered)
39: 
40: 
41: def release_guarantees(store, loan_id: str) -> int:
42:     released = guarantees_for(store, loan_id)
43:     for item in released:
44:         item.released = True
45:     return len(released)
