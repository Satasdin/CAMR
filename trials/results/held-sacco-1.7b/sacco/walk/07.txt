9:     """Rounds half up."""
10:     if amount_cents < 0:
11:         raise ValueError("amount must not be negative")
12:     return (amount_cents * percent + 50) // 100
13: 
14: 
15: def basis_points_of(amount_cents: int, basis_points: int) -> int:
16:     """Rounds half up."""
17:     if amount_cents < 0:
18:         raise ValueError("amount must not be negative")
19:     return (amount_cents * basis_points + 5_000) // 10_000
20: 
21: 
22: def add(amounts: Iterable[int]) -> int:
23:     return sum(amounts)
24: 
25: 
26: def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
27:     sign = "-" if amount_cents < 0 else ""
28:     whole, part = divmod(abs(amount_cents), 100)
29:     return f"{sign}{currency} {whole:,}.{part:02d}"
30: 
31: 
32: def parse_amount(text: str) -> int:
33:     """Reads an amount typed by staff, such as '1,250.50', into cents."""
34:     cleaned = text.replace(",", "").strip()
35:     if not cleaned:
36:         raise ValueError("empty amount")
37:     negative = cleaned.startswith("-")
38:     if negative:
39:         cleaned = cleaned[1:]
40:     whole, _, part = cleaned.partition(".")
41:     if not whole.isdigit() and whole != "":
42:         raise ValueError(f"not an amount: {text}")
43:     if part and (not part.isdigit() or len(part) > 2):
44:         raise ValueError(f"not an amount: {text}")
45:     cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
46:     return -cents if negative else cents

from ..errors import InvalidLoanState
from ..models import ENTRY_FEE, LOAN_ACTIVE, LOAN_APPROVED, Disbursement, Loan

from ..savings.ledger import post_entry

from .fees import record_fee_income

PROCESSING_FEE_BASIS_POINTS = 150


def processing_fee(amount_cents: int) -> int:
    return basis_points_of(amount_cents, PROCESSING_FEE_BASIS_POINTS)


def _activate(store, loan: Loan, clock) -> Loan:
    loan.status = LOAN_ACTIVE
    loan.disbursed_at = clock.now()
    loan.outstanding_principal_cents = loan.principal_cents
    return save_loan(store, loan)


def disburse_loan(store, loan_id: str, gateway, clock) -> Disbursement:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPROVED:
        raise InvalidLoanState(loan_id, loan.status)
    member = get_member(store, loan.member_id)
    fee = processing_fee(loan.principal_cents)
    transfer = gateway.send(member.phone, loan.principal_cents, reference=loan.id)
    _activate(store, loan, clock)
    account = account_of_member(store, loan.member_id)
    post_entry(store, account.id, ENTRY_FEE, -fee, clock, reference=loan.id, note="processing fee")
    record_fee_income(store, loan.id, fee, clock)
    return Disbursement(loan.id, sent_cents=loan.principal_cents, fee_cents=fee, receipt_id=transfer.transfer_id)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

from ..errors import InvalidLoanState
from ..models import ENTRY_FEE, LOAN_ACTIVE, LOAN_APPROVED, Disbursement, Loan
from ..money import basis_points_of
from ..savings.ledger import post_entry

from .fees import record_fee_income

PROCESSING_FEE_BASIS_POINTS = 150


def processing_fee(amount_cents: int) -> int:
    return basis_points_of(amount_cents, PROCESSING_FEE_BASIS_POINTS)


def _activate(store, loan: Loan, clock) -> Loan:
    loan.status = LOAN_ACTIVE
    loan.disbursed_at = clock.now()
    loan.outstanding_principal_cents = loan.principal_cents
    return save_loan(store, loan)


def disburse_loan(store, loan_id: str, gateway, clock) -> Disbursement:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPROVED:
        raise InvalidLoanState(loan_id, loan.status)
    member = get_member(store, loan.member_id)
    fee = processing_fee(loan.principal_cents)
    transfer = gateway.send(member.phone, loan.principal_cents, reference=loan.id)
    _activate(store, loan, clock)
    account = account_of_member(store, loan.member_id)
    post_entry(store, account.id, ENTRY_FEE, -fee, clock, reference=loan.id, note="processing fee")
    record_fee_income(store, loan.id, fee, clock)
    return Disbursement(loan.id, sent_cents=loan.principal_cents, fee_cents=fee, receipt_id=transfer.transfer_id)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]

"""Sending an approved loan to the member."""

from ..errors import InvalidLoanState
from ..models import ENTRY_FEE, LOAN_ACTIVE, LOAN_APPROVED, Disbursement, Loan
from ..money import basis_points_of
from ..savings.ledger import post_entry
from ..store.repositories import account_of_member, get_loan, get_member, save_loan
from .fees import record_fee_income

PROCESSING_FEE_BASIS_POINTS = 150


def processing_fee(amount_cents: int) -> int:
    return basis_points_of(amount_cents, PROCESSING_FEE_BASIS_POINTS)


def _activate(store, loan: Loan, clock) -> Loan:
    loan.status = LOAN_ACTIVE
    loan.disbursed_at = clock.now()
    loan.outstanding_principal_cents = loan.principal_cents
    return save_loan(store, loan)


def disburse_loan(store, loan_id: str, gateway, clock) -> Disbursement:
    loan = get_loan(store, loan_id)
    if loan.status != LOAN_APPROVED:
        raise InvalidLoanState(loan_id, loan.status)
    member = get_member(store, loan.member_id)
    fee = processing_fee(loan.principal_cents)
    transfer = gateway.send(member.phone, loan.principal_cents, reference=loan.id)
    _activate(store, loan, clock)
    account = account_of_member(store, loan.member_id)
    post_entry(store, account.id, ENTRY_FEE, -fee, clock, reference=loan.id, note="processing fee")
    record_fee_income(store, loan.id, fee, clock)
    return Disbursement(loan.id, sent_cents=loan.principal_cents, fee_cents=fee, receipt_id=transfer.transfer_id)

"""Helpers for working with whole cents."""

from typing import Iterable

from .config import CURRENCY


def percent_of(amount_cents: int, percent: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * percent + 50) // 100


def basis_points_of(amount_cents: int, basis_points: int) -> int:
    """Rounds half up."""
    if amount_cents < 0:
        raise ValueError("amount must not be negative")
    return (amount_cents * basis_points + 5_000) // 10_000


def add(amounts: Iterable[int]) -> int:
    return sum(amounts)


def format_cents(amount_cents: int, currency: str = CURRENCY) -> str:
    sign = "-" if amount_cents < 0 else ""
    whole, part = divmod(abs(amount_cents), 100)
    return f"{sign}{currency} {whole:,}.{part:02d}"


def parse_amount(text: str) -> int:
    """Reads an amount typed by staff, such as '1,250.50', into cents."""
    cleaned = text.replace(",", "").strip()
    if not cleaned:
        raise ValueError("empty amount")
    negative = cleaned.startswith("-")
    if negative:
        cleaned = cleaned[1:]
    whole, _, part = cleaned.partition(".")
    if not whole.isdigit() and whole != "":
        raise ValueError(f"not an amount: {text}")
    if part and (not part.isdigit() or len(part) > 2):
        raise ValueError(f"not an amount: {text}")
    cents = int(whole or "0") * 100 + int((part + "00")[:2] if part else "0")
    return -cents if negative else cents


def split_evenly(amount_cents: int, parts: int) -> list:
    """Splits an amount so the parts always add back to the amount."""
    if parts <= 0:
        raise ValueError("parts must be positive")
    share, remainder = divmod(amount_cents, parts)
    return [share + 1 if index < remainder else share for index in range(parts)]
