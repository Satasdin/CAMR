"""Delivering messages to a channel."""

from typing import Callable, Dict, List

from ..errors import DeliveryFailed
from ..models import Message


class Sender:
    """Hands each message to the function registered for its channel."""

    def __init__(self):
        self.channels: Dict[str, Callable[[Message], None]] = {}
        self.failed: List[Message] = []

    def register(self, channel: str, deliver: Callable[[Message], None]) -> None:
        self.channels[channel] = deliver

    def send(self, message: Message) -> bool:
        """True if the channel took the message."""
        deliver = self.channels.get(message.channel)
        if deliver is None:
            self.failed.append(message)
            return False
        try:
            deliver(message)
        except DeliveryFailed:
            self.failed.append(message)
            return False
        return True


def outbox_sender(store, channels=("sms", "email")) -> Sender:
    """A sender that keeps every message in the store's outbox."""
    sender = Sender()
    for channel in channels:
        sender.register(channel, store.outbox.append)
    return sender

"""Delivering messages to a channel."""

from typing import Callable, Dict, List

from ..errors import DeliveryFailed
from ..models import Message


class Sender:
    """Hands each message to the function registered for its channel."""

    def __init__(self):
        self.channels: Dict[str, Callable[[Message], None]] = {}
        self.failed: List[Message] = []

    def register(self, channel: str, deliver: Callable[[Message], None]) -> None:
        self.channels[channel] = deliver

    def send(self, message: Message) -> bool:
        """True if the channel took the message."""
        deliver = self.channels.get(message.channel)
        if deliver is None:
            self.failed.append(message)
            return False
        try:
            deliver(message)
        except DeliveryFailed:
            self.failed.append(message)
            return False
        return True


def outbox_sender(store, channels=("sms", "email")) -> Sender:
    """A sender that keeps every message in the store's outbox."""
    sender = Sender()
    for channel in channels:
        sender.register(channel, store.outbox.append)
    return sender

"""Chasing unpaid bills by sending SMS reminders to patients."""

from ..errors import ValidationFailed, NotFound
from ..models import INVOICE_OPEN, Invoice, Patient
from ..store.repositories import get_invoice
from ..money import format_cents
from ..clock import day_of


def chase_invoice(store, invoice_id: str, sender, clock) -> Message:
    """Sends one SMS to the patient's phone about an unpaid invoice.

    The text quotes the invoice number, the amount still owed, and the visit date.
    Raises ValidationFailed if the invoice is not open with something owing.
    Raises NotFound if the invoice does not exist.
    """
    invoice = get_invoice(store, invoice_id)
    if invoice.status != INVOICE_OPEN or invoice.paid_cents == invoice.total_cents:
        raise ValidationFailed([f"invoice {invoice_id} cannot be chased as it is not open or has no amount owing"])
    patient = get_patient(store, invoice.patient_id)
    if not patient.phone:
        raise NotFound("patient", invoice.patient_id)
    message_body = (
        f"Invoice {invoice.id}: {format_cents(invoice.total_cents - invoice.paid_cents)} "
        f"due on {day_of(invoice.appointment.start_at)}"
    )
    message = Message(
        channel="sms",
        to=patient.phone,
        body=message_body,
        sent_at=clock.now()
    )
    sender.send(message)
    return message
