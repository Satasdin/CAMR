Question: Task:
# Fewer invoices stuck in review after a timeout

Support says that most invoices sitting in the review queue with the reason `gateway timeout` were in fact paid: the customer's money was taken and only the answer got lost. Staff spend every morning matching these up by hand.

Change `charge_invoice` in `courier/billing/checkout.py` so that when the payment did go through, the invoice ends up paid with its receipt, the same way as after a normal successful charge, and nobody has to review it. If there is no sign of the payment, the invoice goes to review as it does now.

# Fewer invoices stuck in review after a timeout

Support says that most invoices sitting in the review queue with the reason `gateway timeout` were in fact paid: the customer's money was taken and only the answer got lost. Staff spend every morning matching these up by hand.

Change `charge_invoice` in `courier/billing/checkout.py` so that when the payment did go through, the invoice ends up paid with its receipt, the same way as after a normal successful charge, and nobody has to review it. """Taking payment for an invoice."""

from ..errors import GatewayDeclined, GatewayTimeout, InvoiceNotOpen, PaymentDeclined
from ..models import INVOICE_OPEN, INVOICE_PAID, INVOICE_PENDING_REVIEW, Invoice
from ..store.repositories import get_customer, get_invoice, save_invoice
from .review_queue import add_to_review


def _mark_paid(store, invoice: Invoice, receipt, clock) -> Invoice:
    invoice.status = INVOICE_PAID
    invoice.paid_at = clock.now()
    invoice.receipt_id = receipt.charge_id
    return save_invoice(store, invoice)


def _send_to_review(store, invoice: Invoice, clock) -> Invoice:
    invoice.status = INVOICE_PENDING_REVIEW
    add_to_review(store, invoice.id, "gateway timeout", clock)
    return save_invoice(store, invoice)


def charge_invoice(store, invoice_id: str, gateway, clock) -> Invoice:
    """Charges the customer's saved payment method for one invoice."""
    invoice = get_invoice(store, invoice_id)
    if invoice.status != INVOICE_OPEN:
        raise InvoiceNotOpen(invoice_id, invoice.status)
    customer = get_customer(store, invoice.customer_id)
    try:
        receipt = gateway.charge(customer.payment_token, invoice.total_cents, reference=invoice.id)
    except GatewayDeclined as declined:
        raise PaymentDeclined(declined.reason) from declined
    except GatewayTimeout:
        return _send_to_review(store, invoice, clock)
    return _mark_paid(store, invoice, receipt, clock)

"""The payment gateway, and a stand-in for local development."""

from dataclasses import dataclass
from typing import Dict, List, Optional

from ..errors import GatewayDeclined, GatewayTimeout


@dataclass
class Receipt:
    charge_id: str
    reference: str
    amount_cents: int


class PaymentGateway:
    """What the service needs from a payment provider."""

    def charge(self, token: str, amount_cents: int, reference: str) -> Receipt:
        """Takes payment. Raises GatewayTimeout or GatewayDeclined."""
        raise NotImplementedError

    def lookup(self, reference: str) -> Optional[Receipt]:
        """The receipt for a reference, or None if the gateway has no such payment."""
        raise NotImplementedError

    def refund(self, charge_id: str, amount_cents: int) -> str:
        """Returns the provider's refund identifier."""
        raise NotImplementedError


class SandboxGateway(PaymentGateway):
    """Keeps payments in memory. Tokens starting with 'declined' are refused."""

    def __init__(self):
        self.receipts: Dict[str, Receipt] = {}
        self.refunds: List[tuple] = []
        self.fail_next: Optional[Exception] = None
        self._counter = 0

    def charge(self, token: str, amount_cents: int, reference: str) -> Receipt:
        if self.fail_next is not None:
            problem, self.fail_next = self.fail_next, None
            raise problem
        if token.startswith("declined"):
            raise GatewayDeclined("card refused")
        if not token:
            raise GatewayDeclined("no payment method")
        self._counter += 1
        receipt = Receipt(f"ch_{self._counter:05d}", reference, amount_cents)
        self.receipts[reference] = receipt
        return receipt

    def lookup(self, reference: str) -> Optional[Receipt]:
        return self.receipts.get(reference)

    def refund(self, charge_id: str, amount_cents: int) -> str:
        self.refunds.append((charge_id, amount_cents))
        return f"re_{len(self.refunds):05d}"

    def simulate_timeout(self) -> None:
        self.fail_next = GatewayTimeout("gateway did not answer")

Question: Task:
# Fewer invoices stuck in review after a timeout

Support says that most invoices sitting in the review queue with the reason `gateway timeout` were in fact paid: the customer's money was taken and only the answer got lost. Staff spend every morning matching these up by hand.

Change `charge_invoice` in `courier/billing/checkout.py` so that when the payment did go through, the invoice ends up paid with its receipt, the same way as after a normal successful charge, and nobody has to review it. If there is no sign of the payment, the invoice goes to review as it does now.

"""Taking payment for an invoice."""

from ..errors import GatewayDeclined, GatewayTimeout, InvoiceNotOpen, PaymentDeclined
from ..models import INVOICE_OPEN, INVOICE_PAID, INVOICE_PENDING_REVIEW, Invoice
from ..store.repositories import get_customer, get_invoice, save_invoice
from .review_queue import add_to_review


def _mark_paid(store, invoice: Invoice, receipt, clock) -> Invoice:
    invoice.status = INVOICE_PAID
    invoice.paid_at = clock.now()
    invoice.receipt_id = receipt.charge_id
    return save_invoice(store, invoice)


def _send_to_review(store, invoice: Invoice, clock) -> Invoice:
    invoice.status = INVOICE_PENDING_REVIEW
    add_to_review(store, invoice.id, "gateway timeout", clock)
    return save_invoice(store, invoice)


def charge_invoice(store, invoice_id: str, gateway, clock) -> Invoice:
    """Charges the customer's saved payment method for one invoice."""
    invoice = get_invoice(store, invoice_id)
    if invoice.status != INVOICE_OPEN:
        raise InvoiceNotOpen(invoice_id, invoice.status)
    customer = get_customer(store, invoice.customer_id)
    try:
        receipt = gateway.charge(customer.payment_token, invoice.total_cents, reference=invoice.id)
    except GatewayDeclined as declined:
        raise PaymentDeclined(declined.reason) from declined
    except GatewayTimeout:
        return _send_to_review(store, invoice, clock)
    return _mark_paid(store, invoice, receipt, clock)

"""The payment gateway, and a stand-in for local development."""

from dataclasses import dataclass
from typing import Dict, List, Optional

from ..errors import GatewayDeclined, GatewayTimeout


@dataclass
class Receipt:
    charge_id: str
    reference: str
    amount_cents: int


class PaymentGateway:
    """What the service needs from a payment provider."""

    def charge(self, token: str, amount_cents: int, reference: str) -> Receipt:
        """Takes payment. Raises GatewayTimeout or GatewayDeclined."""
        raise NotImplementedError

    def lookup(self, reference: str) -> Optional[Receipt]:
        """The receipt for a reference, or None if the gateway has no such payment."""
        raise NotImplementedError

    def refund(self, charge_id: str, amount_cents: int) -> str:
        """Returns the provider's refund identifier."""
        raise NotImplementedError


class SandboxGateway(PaymentGateway):
    """Keeps payments in memory. Tokens starting with 'declined' are refused."""

    def __init__(self):
        self.receipts: Dict[str, Receipt] = {}
        self.refunds: List[tuple] = []
        self.fail_next: Optional[Exception] = None
        self._counter = 0

    def charge(self, token: str, amount_cents: int, reference: str) -> Receipt:
        if self.fail_next is not None:
            problem, self.fail_next = self.fail_next, None
            raise problem
        if token.startswith("declined"):
            raise GatewayDeclined("card refused")
        if not token:
            raise GatewayDeclined("no payment method")
        self._counter += 1
        receipt = Receipt(f"ch_{self._counter:05d}", reference, amount_cents)
        self.receipts[reference] = receipt
        return receipt

    def lookup(self, reference: str) -> Optional[Receipt]:
        return self.receipts.get(reference)

    def refund(self, charge_id: str, amount_cents: int) -> str:
        self.refunds.append((charge_id, amount_cents))
        return f"re_{len(self.refunds):05d}"

    def simulate_timeout(self) -> None:
        self.fail_next = GatewayTimeout("gateway did not answer")

"""The payment gateway, and a stand-in for local development."""

from dataclasses import dataclass
from typing import Dict, List, Optional

from ..errors import GatewayDeclined, GatewayTimeout


@dataclass
class Receipt:
    charge_id: str
    reference: str
    amount_cents: int


class PaymentGateway:
    """What the service needs from a payment provider."""

    def charge(self, token: str, amount_cents: int, reference: str) -> Receipt:
        """Takes payment. Raises GatewayTimeout or GatewayDeclined."""
        raise NotImplementedError

    def lookup(self, reference: str) -> Optional[Receipt]:
        """The receipt for a reference, or None if the gateway has no such payment."""
        raise NotImplementedError

    def refund(self, charge_id: str, amount_cents: int) -> str:
        """Returns the provider's refund identifier."""
        raise NotImplementedError


class SandboxGateway(PaymentGateway):
    """Keeps payments in memory. Tokens starting with 'declined' are refused."""

    def __init__(self):
        self.receipts: Dict[str, Receipt] = {}
        self.refunds: List[tuple] = []
        self.fail_next: Optional[Exception] = None
        self._counter = 0

    def charge(self, token: str, amount_cents: int, reference: str) -> Receipt:
        if self.fail_next is not None:
            problem, self.fail_next = self.fail_next, None
            raise problem
        if token.startswith("declined"):
            raise GatewayDeclined("card refused")
        if not token:
            raise GatewayDeclined("no payment method")
        self._counter += 1
        receipt = Receipt(f"ch_{self._counter:05d}", reference, amount_cents)
        self.receipts[reference] = receipt
        return receipt

    def lookup(self, reference: str) -> Optional[Receipt]:
        return self.receipts.get(reference)

    def refund(self, charge_id: str, amount_cents: int) -> str:
        self.refunds.append((charge_id, amount_cents))
        return f"re_{len(self.refunds):05d}"

    def simulate_timeout(self) -> None:
        self.fail_next = GatewayTimeout("gateway did not answer")

from ..models import INVOICE_OPEN, INVOICE_PAID, INVOICE_PENDING_REVIEW, Invoice

    invoice.paid_at = clock.now()
    invoice.receipt_id = receipt.charge_id
    return save_invoice(store, invoice)


def _send_to_review(store, invoice: Invoice, clock) -> Invoice:
    invoice.status = INVOICE_PENDING_REVIEW
    add_to_review(store, invoice.id, "gateway timeout", clock)
    return save_invoice(store, invoice)


def charge_invoice(store, invoice_id: str, gateway, clock) -> Invoice:
    """Charges the customer's saved payment method for one invoice."""
    invoice = get_invoice(store, invoice_id)
    if invoice.status != INVOICE_OPEN:
        raise InvoiceNotOpen(invoice_id, invoice.status)
    customer = get_customer(store, invoice.customer_id)
    try:
        receipt = gateway.charge(customer.payment_token, invoice.total_cents, reference=invoice.id)
    except GatewayDeclined as declined:
        raise PaymentDeclined(declined.reason) from declined
    except GatewayTimeout:
        # Check if the receipt exists after the timeout
        try:
            # Try to look up the receipt by reference
            lookup_receipt = gateway.lookup(invoice.id)
            if lookup_receipt is not None:
                # Payment was successful, mark as paid
                return _mark_paid(store, invoice, lookup_receipt, clock)
            else:
                # No receipt found, send to review
                return _send_to_review(store, invoice, clock)
        except Exception:
            # If lookup fails, send to review
            return _send_to_review(store, invoice, clock)
    return _mark_paid(store, invoice, receipt, clock)
